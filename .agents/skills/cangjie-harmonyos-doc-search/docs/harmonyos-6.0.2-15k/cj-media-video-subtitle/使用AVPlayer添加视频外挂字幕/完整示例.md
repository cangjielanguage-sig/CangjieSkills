## 完整示例

```cangjie
import kit.MediaKit.*
import kit.CoreFileKit.*
import kit.AudioKit.*
import ohos.base.*

var ctx = None<UIAbilityContext>

class AVPlayerSubtitleDemo {
    private var avPlayer: ?AVPlayer = None
    // 注册avplayer回调函数
    func setAVPlayerCallback(avPlayer: AVPlayer) {
        // error回调监听函数,当avPlayer在操作过程中出现错误时调用reset接口触发重置流程
        avPlayer.on(AVPlayerCallbackType.error, ErrCallBack(avPlayer))
        // 注册字幕回调函数
        avPlayer.on(AVPlayerCallbackType.subtitleUpdate, SubtitleUpdateCallback())
    }

    // 以下demo为使用资源管理接口获取打包在HAP内的媒体资源文件并通过url属性设置
    func avPlayerSubtitleUrlDemo() {
        // 创建avPlayer实例对象
        this.avPlayer = createAVPlayer()
        // 设置视频信息
        // 创建回调函数
        this.setAVPlayerCallback(this.avPlayer.getOrThrow())

        var fdUrl: String = "http://xxx.xxx.xxx.xxx:xx/xx/index.srt"

        this.avPlayer.getOrThrow().addSubtitleFromUrl(fdUrl)
    }

    // 以下示例为使用资源管理接口获取打包在HAP内的媒体资源文件并通过FromFd属性设置
    func avPlayerSubtitleFromFdDemo() {
        // 创建avPlayer实例对象
        this.avPlayer = createAVPlayer()
        // 设置视频信息
        // 创建回调函数
        this.setAVPlayerCallback(this.avPlayer.getOrThrow())
        var fileDescriptor = ctx.getOrThrow().resourceManager.getRawFd("xxx.srt")

        this.avPlayer.getOrThrow().addSubtitleFromFd(fileDescriptor.fd, offset: fileDescriptor.offset,
            length: fileDescriptor.length)
    }

    // 注销字幕回调函数
    func avPlayerSubtitleOffDemo() {
        if (this.avPlayer.isSome()) {
            this.avPlayer.getOrThrow().off(AVPlayerCallbackType.subtitleUpdate)
        }
    }
}

//注册字幕回调函数类
class SubtitleUpdateCallback <: Callback1Argument<SubtitleInfo> {
    public init() {}
    public open func invoke(info: SubtitleInfo) {
        var text = info.text
        var startTime = info.startTime
        var duration = info.duration
        AppLog.info("subtitleUpdate info: text= ${text} startTime= ${startTime} duration= ${duration}")
    }
}

// error监听回调函数类
class ErrCallBack <: Callback1Argument<BusinessException> {
    let avPlayer_: AVPlayer
    public init(avPlayer: AVPlayer) {
        this.avPlayer_ = avPlayer
    }
    public open func invoke(err: BusinessException): Unit {
        AppLog.error("Invoke avPlayer failed, code is ${err.code}, message is ${err.message}")
        avPlayer_.reset()
    }
}
```

// main_ability.cj
import kit.AbilityKit.{LaunchReason, LaunchParam, Want, UIAbility}
import kit.ArkUI.WindowStage
import kit.UIKit.AppLog

class MainAbility <: UIAbility {
    public init() {
        super()
        registerSelf()
    }

    public override func onCreate(want: Want, launchParam: LaunchParam): Unit {
        AppLog.info("MainAbility OnCreated.${want.abilityName}")
        match (launchParam.launchReason) {
            case LaunchReason.START_ABILITY => AppLog.info("START_ABILITY")
            case _ => ()
        }
    }

    public override func onWindowStageCreate(windowStage: WindowStage): Unit {
        AppLog.info("MainAbility onWindowStageCreate.")
        windowStage.loadContent("EntryView")
        // declared in index.cj
        ctx = this.context
    }
}
