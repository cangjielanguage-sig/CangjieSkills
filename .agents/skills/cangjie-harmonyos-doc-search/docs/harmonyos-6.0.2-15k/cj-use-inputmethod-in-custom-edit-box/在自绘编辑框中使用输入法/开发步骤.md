## 开发步骤

1. 开发者在自绘编辑框中使用输入法时，首先需要在 DevEco Studio 工程中新建一个 cj 文件，命名为自定义控件的名称。本示例中命名为 `CustomInput`，在文件中定义一个自定义控件，并从 `kit.IMEKit` 中导入 `inputMethod`。

    ```cangjie
    import kit.IMEKit.*
    import ohos.component.*
    import ohos.state_macro_manage.Component

    @Component
    class CustomInput {
        func build() {}
    }
    ```

2. 在控件中，使用 `Text` 组件作为自绘编辑框的文本显示组件，使用状态变量 `inputText` 作为 `Text` 组件要显示的内容。

    ```cangjie
    import kit.IMEKit.*
    import ohos.component.*
    import ohos.state_macro_manage.Component
    import ohos.state_macro_manage.State

    @Component
    class CustomInput {
        @State
        var inputText: String = "" // inputText作为Text组件要显示的内容。

        func build() {
            Text(this.inputText) // Text组件作为自绘编辑框的文本显示组件。
                .fontSize(16).width(100.percent).lineHeight(40).id('customInput').height(45).border(
                width: 1, radius: 30.vp, color: 0x554455).maxLines(1)
        }
    }
    ```

3. 在控件中获取inputMethodController实例，并在文本点击时调用controller示例的attach方法绑定和拉起软键盘，并注册监听输入法插入文本、删除等方法，本示例仅展示插入、删除。

    ```cangjie
    import kit.IMEKit.*
    import ohos.component.*
    import ohos.state_macro_manage.Component
    import ohos.state_macro_manage.State
    import kit.IMEKit.EnterKeyType as IMEEnterKeyType
    import ohos.state_manage.*

    @Component
    class CustomInput {
        @State
        var inputText: String = "" // inputText作为Text组件要显示的内容。

        var isAttach = false
        let inputController: InputMethodController = getController()

        func build() {
            Text(this.inputText) // Text组件作为自绘编辑框的文本显示组件。
                .fontSize(16).width(100.percent).lineHeight(40).id('customInput').onBlur {
                off()
            }.height(45).border(width: 1, radius: 30.vp, color: 0x554455).maxLines(1).onClick {
                evt => attachAndListener()
            }
        }

        func attachAndListener() { // 绑定和设置监听
            FocusControl.requestFocus('CustomInput')
            let textConfig = TextConfig(inputAttribute: InputAttribute(TextInputType.TEXT, IMEEnterKeyType.SEARCH))
            this.inputController.attach(true, textConfig)
            if (!this.isAttach) {
                let insertCallback = InsertTextCallback({
                    p: String => this.inputText = this.inputText + p
                })
                this.inputController.on(InsertText, insertCallback)
                let deleteCallback = DeleteCallback(
                    {
                        p: Int32 =>
                        let runes = inputText.toRuneArray()
                        this.inputText = String(runes[..runes.size - Int64(p)])
                    }
                )
                this.inputController.on(DeleteLeft, deleteCallback)
                this.isAttach = true
            }
        }

        func off() {
            this.isAttach = false
            this.inputController.off(InsertText)
            this.inputController.off(DeleteLeft)
        }
    }

    public class InsertTextCallback <: Callback1Argument<String> {
        public InsertTextCallback(let function: (String) -> Unit) {}

        public func invoke(arg1: String): Unit {
            function(arg1)
        }
    }

    public class DeleteCallback <: Callback1Argument<Int32> {
        public DeleteCallback(let function: (Int32) -> Unit) {}

        public func invoke(arg1: Int32): Unit {
            function(arg1)
        }
    }
    ```

4. 在应用界面布局中引入该控件即可。此处假设使用界面为 `index.cj` 和控件 `custominput.cj` 在同一目录下。

    ```cangjie
    import ohos.state_macro_manage.Entry
    import ohos.state_macro_manage.Component

    @Entry
    @Component
    class Index {
        func build() {
            Column {
                CustomInput() // 使用控件
            }
        }
    }
    ```
