## 完整示例

下面展示了使用AVScreenCaptureRecorder屏幕录屏存文件的完整示例代码。

```cangjie
import kit.MediaKit.*
import kit.CoreFileKit.*
import kit.AudioKit.*
import ohos.base.*

class AVScreenCaptureDemo {
    private var screenCapture: ?AVScreenCaptureRecorder = None
    var captureConfig: AVScreenCaptureRecordConfig = AVScreenCaptureRecordConfig(
        // 参考应用文件访问与管理开发示例新建并读写一个文件fd
        0,
        // 开发者可以根据自身的需要设置宽高
        768, // 宽度frameWidth
        1280, // 高度frameHeight
        // 可选参数及其默认值
        videoBitrate: 10000000,
        audioSampleRate: 48000,
        audioChannelCount: 2,
        audioBitrate: 96000,
        preset: AVScreenCaptureRecordPreset.SCREEN_RECORD_PRESET_H264_AAC_MP4
    )

    // 调用startRecording方法可以开始一次录屏存文件的流程，结束录屏可以通过点击录屏胶囊停止按钮进行操作。
    public func startRecording() {
        this.screenCapture = createAVScreenCaptureRecorder()
        if (this.screenCapture.isSome()) {
        // success
        } else {
            // failed
            return
        }
        var realScreenCapture = this.screenCapture.getOrThrow()
        realScreenCapture.on(ProfileCallbackType.CONNECTION_STATE_CHANGE, StateChangeCallback(realScreenCapture))
        realScreenCapture.on(ProfileCallbackType.CONNECTION_ERROR, ErrCallBack())
        realScreenCapture.initialize(this.captureConfig)

        // 豁免隐私窗口
        let windowIDs = [57u64, 86u64]
        realScreenCapture.skipPrivacyMode(windowIDs)

        realScreenCapture.startRecording()
        this.screenCapture = None
    }

    // 可以主动调用stopRecording方法来停止录屏。
    public func stopRecording() {
        if (this.screenCapture.isNone()) {
            // Error
            return
        }
        var realscreenCapture = this.screenCapture.getOrThrow()
        realscreenCapture.stopRecording()

        // 调用release()方法销毁实例，释放资源。
        realscreenCapture.release()

        // 最后需要关闭创建的录屏文件fd, fs.close(fd)
    }
}

class StateChangeCallback <: Callback1Argument<AVScreenCaptureStateCode> {
    let screenCapture: AVScreenCaptureRecorder
    public init(screenCapture: AVScreenCaptureRecorder) {
        this.screenCapture = screenCapture
    }
    public open func invoke(infoType: AVScreenCaptureStateCode): Unit {
        match (infoType) {
            case AVScreenCaptureStateCode.SCREENCAPTURE_STATE_STARTED => AppLog.info("录屏成功开始后会收到的回调")
            case AVScreenCaptureStateCode.SCREEN_CAPTURE_STATE_CANCELED =>
                this.screenCapture.release()
                AppLog.info("不允许使用录屏功能")
                return
            case AVScreenCaptureStateCode.SCREEN_CAPTURE_STATE_STOPPED_BY_USER =>
                this.screenCapture.release()
                AppLog.info("通过录屏胶囊结束录屏，底层录制会停止")
            case AVScreenCaptureStateCode.SCREEN_CAPTURE_STATE_INTERRUPTED_BY_OTHER => AppLog.info(
                "录屏因其他中断而停止，底层录制会停止")
            case AVScreenCaptureStateCode.SCREEN_CAPTURE_STATE_STOPPED_BY_CALL => AppLog.info(
                "录屏过程因通话中断，底层录制会停止")
            case AVScreenCaptureStateCode.SCREEN_CAPTURE_STATE_MIC_UNAVAILABLE => AppLog.info("录屏麦克风不可用")
            case AVScreenCaptureStateCode.SCREEN_CAPTURE_STATE_MIC_MUTED_BY_USER => AppLog.info("录屏麦克风被用户静音")
            case AVScreenCaptureStateCode.SCREEN_CAPTURE_STATE_MIC_UNMUTED_BY_USER => AppLog.info(
                "录屏麦克风被用户取消静音")
            case AVScreenCaptureStateCode.SCREEN_CAPTURE_STATE_ENTER_PRIVATE_SCENE =>
                // 目前可以从系统直接注册监听到进入隐私场景
                AppLog.info("录屏进入隐私场景")
            case AVScreenCaptureStateCode.SCREEN_CAPTURE_STATE_EXIT_PRIVATE_SCENE => AppLog.info("录屏退出隐私场景")
            case AVScreenCaptureStateCode.SCREEN_CAPTURE_STATE_STOPPED_BY_USER_SWITCHES => AppLog.info(
                "用户账号切换，底层录制会停止")
            case _ => ()
        }
    }
}
