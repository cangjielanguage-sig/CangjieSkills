## 事件调用

通过在Image组件上绑定onComplete事件，图片加载成功后可以获取图片的必要信息。如果图片加载失败，也可以通过绑定onError回调来获得结果。

<!-- run -->

```cangjie
package ohos_app_cangjie_entry

import kit.UIKit.*
import ohos.state_macro_manage.*
import ohos.resource_manager.*
import ohos.hilog.Hilog

@Entry
@Component
class EntryView {
    @State
    var widthValue: Float64 = 0.0
    @State
    var heightValue: Float64 = 0.0
    @State
    var componentWidth: Float64 = 0.0
    @State
    var componentHeight: Float64 = 0.0
    func build() {
        Column() {
            Row() {
                Image(@r(app.media.example)).width(200).height(150).margin(15).onComplete(
                    {
                        msg: CJImageComplete =>
                        this.widthValue = msg.width
                        this.heightValue = msg.height
                        this.componentWidth = msg.componentWidth
                        this.componentHeight = msg.componentHeight
                    }
                ).onError({
                    evt => AppLog.info("load image fail")
                }).overlay(
                    title: '\nwidth: ${this.widthValue}, height: ${this.heightValue}\ncomponentWidth: ${this.componentWidth}\ncomponentHeight: ${this.componentHeight}',
                    align: Alignment.Bottom,
                    offset: ContentOffset(x: 0.0, y: 60.0)
                )
            }
        }
    }
}
```
