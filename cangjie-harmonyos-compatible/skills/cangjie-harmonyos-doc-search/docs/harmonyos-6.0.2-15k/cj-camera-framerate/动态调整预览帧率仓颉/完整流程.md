## 完整流程

根据上述开发流程，完整的session配流及previewOutput在session.start前后调整帧率示例代码如下。

```cangjie
func sessionConfig(cameraManager: CameraManager, cameraInput: CameraInput, previewOutput: PreviewOutput): Unit {
    // 创建CaptureSession实例
    let session = createPhotoSession(cameraManager)
    // 开始配置会话
    session.beginConfig()
    // 把CameraInput加入到会话
    session.addInput(cameraInput)
    // 把previewOutput加入到会话
    session.addOutput(previewOutput)
    // 提交配置信息
    session.commitConfig()
    // 获取当前支持的帧率范围
    let supportFrameRateArray: Array<FrameRateRange> = getSupportedFrameRange(previewOutput)
    // 可在start前对帧率进行设置
    if (supportFrameRateArray.size != 0) {
        // 将帧率设置为第一组帧率范围的最大值
        setFrameRate(previewOutput, supportFrameRateArray[0].max, supportFrameRateArray[0].max)
    }
    // 获取当前生效的帧率设置
    var activeFrameRateArray: FrameRateRange = getActiveFrameRange(previewOutput)
    // 开始会话工作并启动预览
    session.start()
    // 可在start后对帧率进行设置
    if (supportFrameRateArray.size != 0) {
        // 可将帧率设置为最大值的一半（能否设置成功取决于平台是否支持）
        setFrameRate(previewOutput, supportFrameRateArray[0].max / 2, supportFrameRateArray[0].max / 2)
    }
    // 获取当前生效的帧率设置
    activeFrameRateArray = getActiveFrameRange(previewOutput)
}
```
