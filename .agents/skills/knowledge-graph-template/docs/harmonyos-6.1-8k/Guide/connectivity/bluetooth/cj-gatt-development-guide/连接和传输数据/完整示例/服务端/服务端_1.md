            this.myCharacteristicUuid,
            des,
            value
        )
        return descriptor
    }

        let characteristic: BleCharacteristic = BleCharacteristic(
            this.myServiceUuid,
            this.myCharacteristicUuid,
            charBuffer,
            descriptors
        )

    }
