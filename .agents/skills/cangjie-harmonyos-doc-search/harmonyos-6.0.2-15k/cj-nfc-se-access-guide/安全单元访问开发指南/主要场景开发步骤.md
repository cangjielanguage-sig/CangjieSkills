## 主要场景开发步骤

### 应用程序访问安全单元

1. import需要的安全单元模块。
2. 判断设备是否支持安全单元能力。
3. 访问安全单元，实现数据的读取或写入。

```cangjie
import kit.ConnectivityKit.*
import ohos.base.BusinessException

var seService: ?SEService = None
var seReaders: ?Array<Reader> = None
var seSession: ?Session = None
var seChannel: ?Channel = None
let aidArray: Array<Int32> = [0xA0, 0x00, 0x00, 0x00, 0x03, 0x10, 0x10]
let p2: Int32 = 0x00

class EntryAbility <: UIAbility {
    public func onCreate(want: Want, launchParam: LaunchParam): Unit {
        AppLog.info('Ability onCreate')

        // 判断设备是否支持NFC能力和HCE能力
        // ...
        AppLog.info('secure element available.')
        this.omaTest()
    }
    private func omaTest() {
        // get the service
        try {
            seService = createService()
        } catch (e: BusinessException) {
            AppLog.error('errCode: ${e.code}, errMessage: ' + e.message)
        }

        // get readers
        try {
            seReaders = seService?.getReaders()
        } catch (e: BusinessException) {
            AppLog.error('errCode: ${e.code}, errMessage: ' + e.message)
        }
        if (seReaders.isNone() || seReaders?.size == 0) {
            AppLog.error('no valid reader found.')
            seService?.shutdown()
            return
        }
        var reader: ?Reader = None
        for (i in 0..(seReaders?.size ?? 0)) {
            let r = seReaders.getOrThrow()[i]
            if (r.getName().contains("SIM")) {
                reader = r
            }
        }
        if (reader.isNone()) {
            AppLog.error('no valid sim reader.')
            return
        }
        AppLog.info(reader?.getName())

        // get session
        try {
            seSession = reader?.openSession() as Session
        } catch (e: BusinessException) {
            AppLog.error('errCode: ${e.code}, errMessage: ' + e.message)
        }
        if (seSession.isNone()) {
            AppLog.error('seSession invalid.')
            seService?.shutdown()
            return
        }

        // get channel
        try {
            // change the aid value for open logical channel.
            seChannel = seSession?.openLogicalChannel(aidArray, p2)
        } catch (e: BusinessException) {
            AppLog.error('errCode: ${e.code}, errMessage: ' + e.message)
        }

        if (seChannel.isNone()) {
            AppLog.error('seChannel invalid.')
            return
        }

        // transmit data
        let cmdData: Array<Int32> = [0x01, 0x02, 0x03, 0x04] // please change the raw data to be correct.
        try {
            let response: Array<Int32> = seChannel.getOrThrow().transmit(cmdData)
            AppLog.info('seChannel.transmit() response = ${response}')
        } catch (e: BusinessException) {
            AppLog.error('errCode: ${e.code}, errMessage: ' + e.message)
        }

        // close channel. must make sure the channel is closed at last.
        try {
            seChannel?.close()
        } catch (e: BusinessException) {
            AppLog.error('errCode: ${e.code}, errMessage: ' + e.message)
        }
    }
}
```
