## 目标方开发步骤

1. 在module.json5中配置[uris](../cj-start/basic-knowledge/module-configuration-file.md)：

    a. 设置linkFeature属性以声明当前应用支持的特性功能，从而系统可以从设备已安装应用中找到当前支持该特性的应用，取值范围如下：

   | 取值         | 含义                 |
   | ------------ | -------------------- |
   | QueryExpress | 声明应用支持快递查询。 |

    b. 设置scheme、host、port、path/pathStartWith属性，与Want中URI相匹配，以便区分不同功能。

    ```json
    {
        "abilities": [
            {
                "skills": [
                    {
                        "uris": [
                            {
                                "scheme": "express",
                                "host": "queryExpress",
                                "path": "",
                                "linkFeature": "QueryExpress"
                            }
                        ]
                    }
                ]
            }
        ]
    }
    ```

2. 解析参数并做对应处理。

    ```cangjie
    UIAbility.onCreate(want: Want, launchParam: LaunchParam): Unit
    ```

    在参数**want.uri**中会携带目标方配置的linkFeature对应的uri。

    在参数**want.parameters**中会携带Caller方传入的参数，如下所示：

   | 参数名    | 类型   | 必填 | 说明     |
   | --------- | ------ | ---- | -------- |
   | expressNo | String | 是   | 快递单号。 |

**完整示例：**

```cangjie
import ohos.state_manage.LocalStorage
import kit.AbilityKit.{UIAbility, Want, LaunchParam}
import kit.ArkUI.WindowStage
import kit.UIKit.AppLog
import std.collection.HashMap

let localStorage = LocalStorage()

class MainAbility <: UIAbility {
    var windowStage: ?WindowStage = None

    var uri: ?String = None
    var expressNo: ?String = None

    public override func onCreate(want: Want, launchParam: LaunchParam): Unit {
        this.uri = want.uri
        // want.parameters是一个json格式的字符串，用户可通过三方json库解析出expressNo字段的值
    }

    public override func onNewWant(want: Want, launchParam: LaunchParam): Unit {
        this.uri = want.uri
        // want.parameters是一个json格式的字符串，用户可通过三方json库解析出expressNo字段的值
        match (this.windowStage) {
            case Some(v) => loadPage(v)
            case None =>
                AppLog.error("windowStage is null")
                this.context.terminateSelf()
        }
    }

    public override func onWindowStageCreate(windowStage: WindowStage): Unit {
        AppLog.info("MainAbility onWindowStageCreate.")
        this.windowStage = windowStage
        loadPage(windowStage)
    }

    private func loadPage(windowStage: WindowStage): Unit {
        AppLog.info("loadPage, uri=${this.uri}")
        if (this.uri == "express://queryExpress") {
            // 构建快递查询参数
            localStorage.setOrCreate<?String>("expressNo", this.expressNo)
            // 拉起快递查询页面
            windowStage.loadContent("QueryExpressPage")
        } else {
            // 默认拉起首页
            windowStage.loadContent("EntryView")
        }
    }
}
```
