## 参考示例

```cangjie
import kit.AudioKit.*
import ohos.base.{BusinessException, Callback1Argument}

var audioRenderer: ?AudioRenderer = None
let audioStreamInfo = AudioStreamInfo(
    AudioChannel.CHANNEL_2, // 通道。
    AudioEncodingType.ENCODING_TYPE_RAW, // 编码格式。
    AudioSampleFormat.SAMPLE_FORMAT_S16LE, // 采样格式。
    AudioSamplingRate.SAMPLE_RATE_48000 // 采样率。
)
let audioRendererInfo = AudioRendererInfo(
    StreamUsage.STREAM_USAGE_MUSIC, // 音频流使用类型：音乐。根据业务场景配置，参考StreamUsage。
    0 // 音频渲染器标志。
)
let audioRendererOptions = AudioRendererOptions(audioRendererInfo, audioStreamInfo)

// 创建AudioRenderer实例。
func initialize() {
    try {
        audioRenderer = createAudioRenderer(audioRendererOptions)
        AppLog.info("Invoke createAudioRenderer succeeded.")
    } catch (e: BusinessException) {
        AppLog.error("Invoke createAudioRenderer failed, code is ${e.code}, message is ${e.message}")
    }
}

// 订阅监听音频流输出设备变化及原因。
func subscribe() {
    let callback: (AudioStreamDeviceChangeInfo) -> Unit = {
        deviceChangeInfo: AudioStreamDeviceChangeInfo => match (deviceChangeInfo.changeReason) {
            case AudioStreamDeviceChangeReason.REASON_OLD_DEVICE_UNAVAILABLE =>
                // 响应设备不可用事件，如果应用处于播放状态，应暂停播放，更新UX界面。
                // await audioRenderer.pause()
                ()
            case AudioStreamDeviceChangeReason.REASON_NEW_DEVICE_AVAILABLE =>
                // 应用根据业务情况响应设备可用事件。
                ()
            case AudioStreamDeviceChangeReason.REASON_OVERRODE =>
                // 应用根据业务情况响应设备强选事件。
                ()
            case AudioStreamDeviceChangeReason.REASON_UNKNOWN =>
                // 应用根据业务情况响应未知原因事件。
                ()
            case _ => ()
        }
    }
    audioRenderer?.on(AR_OUTPUT_DEVICE_CHANGE_WITH_INFO, AudioStreamDeviceChangeInfoCallback(callback))
}

class AudioStreamDeviceChangeInfoCallback <: Callback1Argument<AudioStreamDeviceChangeInfo> {
    public AudioStreamDeviceChangeInfoCallback(let f: (AudioStreamDeviceChangeInfo) -> Unit) {}
    public func invoke(info: AudioStreamDeviceChangeInfo): Unit {
        f(info)
    }
}
```
