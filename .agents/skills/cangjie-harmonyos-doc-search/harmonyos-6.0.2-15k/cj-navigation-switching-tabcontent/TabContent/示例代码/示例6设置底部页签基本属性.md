### 示例6（设置底部页签基本属性）

本示例通过padding、verticalAlign、layoutMode、symmetricExtensible实现了底部页签基本属性的展示。

<!-- run -->

```cangjie
package ohos_app_cangjie_entry

import kit.UIKit.*
import std.collection.*
import kit.LocalizationKit.*
import ohos.state_macro_manage.*

@Entry
@Component
class EntryView {
    var controller: TabsController = TabsController()
    @State
    var text: String = "2"
    @State
    var tabPadding: Int64 = 0
    @State
    var symmetricExtensible: Bool = false
    @State
    var layoutMode: LayoutMode = LayoutMode.VERTICAL
    @State
    var verticalAlign: VerticalAlign = VerticalAlign.Center

    func build() {
        Column() {
            Row() {
                Button("padding+10 ${this.tabPadding.value}").width(47.percent).height(50).margin(top: 5).onClick(
                    {
                    => this.tabPadding += 10
                }).margin(right: 6.percent, bottom: 12.vp)
                Button("padding-10 ${this.tabPadding.value}").width(47.percent).height(50).margin(top: 5).onClick(
                    {
                    => this.tabPadding -= 10
                }).margin(bottom: 12.vp)
            }

            Row() {
                Button("文本增加").width(47.percent).height(50).margin(top: 5).onClick({
                    => this.text += "文本增加"
                }).margin(right: 6.percent, bottom: 12.vp)
                Button("文本重置").width(47.percent).height(50).margin(top: 5).onClick({
                    => this.text = "2"
                }).margin(bottom: 12.vp)
            }

            Row() {
                Button("symmetricExtensible改变 ${this.symmetricExtensible}").width(100.percent).height(50).margin(
                    top: 5).onClick({
                    => this.symmetricExtensible = !this.symmetricExtensible
                }).margin(bottom: 12.vp)
            }

            Row() {
                Button("layoutMode垂直").width(47.percent).height(50).margin(top: 5).onClick(
                    {
                    => this.layoutMode = LayoutMode.VERTICAL
                }).margin(right: 6.percent, bottom: 12.vp)
                Button("layoutMode水平 ").width(47.percent).height(50).margin(top: 5).onClick(
                    {
                    => this.layoutMode = LayoutMode.HORIZONTAL
                }).margin(bottom: 12.vp)
            }

            Row() {
                Button("verticalAlign朝上").width(100.percent).height(50).margin(top: 5).onClick(
                    {
                    => this.verticalAlign = VerticalAlign.Top
                }).margin(bottom: 12.vp)
            }

            Row() {
                Button("verticalAlign居中").width(100.percent).height(50).margin(top: 5).onClick(
                    {
                    => this.verticalAlign = VerticalAlign.Center
                }).margin(bottom: 12.vp)
            }

            Row() {
                Button("verticalAlign朝下").width(100.percent).height(50).margin(top: 5).onClick(
                    {
                    => this.verticalAlign = VerticalAlign.Bottom
                }).margin(bottom: 12.vp)
            }

            Tabs(BarPosition.End, this.controller) {
                TabContent() {
                    Column().width(100.percent).height(100.percent).backgroundColor(0xE67C92)
                }.tabBar(BottomTabBarStyle.of(@r(app.media.startIcon), "1"))

                TabContent() {
                    Column().width(100.percent).height(100.percent).backgroundColor(0x00CB87)
                }.tabBar(
                    BottomTabBarStyle.of(@r(app.media.startIcon), this.text).padding(this.tabPadding).verticalAlign(
                    this.verticalAlign).layoutMode(this.layoutMode).symmetricExtensible(this.symmetricExtensible))

                TabContent() {
                    Column().width(100.percent).height(100.percent).backgroundColor(0x007DFF)
                }.tabBar(BottomTabBarStyle.of(@r(app.media.startIcon), "3"))
            }.animationDuration(300).height(60.percent).backgroundColor(0xf1f3f5).barMode(BarMode.Fixed)
        }.width(100.percent).height(500).margin(top: 5).padding(24.vp)
    }
}
```
