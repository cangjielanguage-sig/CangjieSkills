### 示例代码

#### @LocalStorageProp

<!-- run -->

```cangjie
package ohos_app_cangjie_entry

import kit.UIKit.*
import ohos.state_macro_manage.*

let storage = LocalStorage()
let tmp = storage.setOrCreate<Int64>("PropA", 47)

// 使LocalStorage可以从@Component组件访问
@Entry[storage]
@Component
class EntryView {
    @LocalStorageProp["PropA"]
    let storageProp1: Int64 = 1
    func build() {
        Column(15) {
            Button("Parent from LocalStorage ${storageProp1}").onClick {
                =>
                // 只能通过操作storage来改变PropA的值
                storage.set<Int64>("PropA", 48)
            }
        }
    }
}
```

#### @LocalStorageLink

<!-- run -->

```cangjie
package ohos_app_cangjie_entry

import kit.UIKit.*
import ohos.state_macro_manage.*

let storage = LocalStorage()

@Entry[storage]
@Component
class EntryView {
    @LocalStorageLink["count"]
    var a: Int64 = 0
    func build() {
        Row {
            Column {
                Text("${a}")
                // a的改动会同步到各个子组件
                Button("Click").onClick {
                    => a += 1
                }
                Son()
                Son2()
            }.width(100.percent)
        }.height(100.percent)
    }
}

@Component
class Son {
    @LocalStorageLink["count"]
    var a1: Int64 = 0
    func build() {
        Column() {
            Text("son: ${a1}")
            // a1的改动会同步到父组件和兄弟组件
            Button("Son click").onClick {
                => a1 -= 1
            }
        }
    }
}

@Component
class Son2 {
    @LocalStorageLink["count"]
    var a2: Int64 = 0
    func build() {
        Column() {
            Text("son2: ${a2}")
            // a2的改动会同步到父组件和兄弟组件
            Button("Son2 click").onClick {
                => a2 += 2
            }
        }
    }
}
```
