// main_ability.cj
import kit.AbilityKit.*
import ohos.base.AppLog
internal import ohos.ability.UIAbilityContext
internal import ohos.ability.AbilityStage
public class Global {
    public static var _abilityContext: Option<UIAbilityContext> = None

    public static prop abilityContext: UIAbilityContext {
        get() {
            Global._abilityContext.getOrThrow()
        }
    }

    public static func getAbilityContext() {
        _abilityContext.getOrThrow()
    }

        getStageContext(getAbilityContext())
    }

    }
}

        super()

    }

            case _ => ()
        }
    }

        AppLog.info("MainAbility onWindowStageCreate.")

    }

import kit.AudioKit.*
import ohos.base.{BusinessException, Callback1ArgumentWithReturn}
import kit.CoreFileKit.*

class WriteDataCallback <: Callback1ArgumentWithReturn<Array<Byte>, AudioDataCallbackResult> {
    public WriteDataCallback(let f: (Array<Byte>) -> AudioDataCallbackResult) {}
    public func invoke(buffer: Array<Byte>): AudioDataCallbackResult {
        f(buffer)
    }
}

const TAG: String = "AudioRendererDemo"

let abilityContext = Global.getAbilityContext()
let path = abilityContext.cacheDir
// 确保该沙箱路径下存在该资源。
let filePath = path + '/StarWars10s-2C-48000-4SW.wav'
let file = FileFs.open(filePath, mode: READ_ONLY.mode)
var bufferSize = 0
var audioRenderer: ?AudioRenderer = None

let audioStreamInfo = AudioStreamInfo(
    AudioChannel.CHANNEL_2, // 通道。
    AudioEncodingType.ENCODING_TYPE_RAW, // 编码格式。
    AudioSampleFormat.SAMPLE_FORMAT_S16LE, // 采样格式。
    AudioSamplingRate.SAMPLE_RATE_48000 // 采样率。
)

let audioRendererInfo = AudioRendererInfo(
    StreamUsage.STREAM_USAGE_MUSIC, // 音频流使用类型：音乐。根据业务场景配置，参考StreamUsage。
    0 // 音频渲染器标志。
)

// 初始化，创建实例，设置监听事件。
func initialize() {
    try {
        audioRenderer = createAudioRenderer(audioRendererOptions)
        audioRenderer?.on(AudioRendererCallbackType.AR_WRITE_DATA, WriteDataCallback(callback))
        AppLog.info("Invoke createAudioRenderer succeeded.")
    } catch (e: BusinessException) {
        AppLog.error("Invoke createAudioRenderer failed, code is ${e.code}, message is ${e.message}")
    }
}

// 开始一次音频渲染。
func start() {
    if (let Some(renderer) <- audioRenderer) {
        match (renderer.state) {
            // 当且仅当状态为prepared、paused和stopped之一时才能启动渲染。
            case AudioState.STATE_PREPARED | AudioState.STATE_PAUSED | AudioState.STATE_STOPPED =>
                try {
                    // 启动渲染。
                    renderer.start()
                    AppLog.info("Renderer start success.")
                } catch (e: BusinessException) {
                    AppLog.error("Renderer start failed, code is ${e.code}, message is ${e.message}")
                }
            case _ => AppLog.error(TAG + "start failed")
        }
    }
}

// 暂停渲染。
func pause() {
    if (let Some(renderer) <- audioRenderer) {
        match (renderer.state) {
            // 只有渲染器状态为running的时候才能暂停。
            case AudioState.STATE_RUNNING =>
                try {
                    // 暂停渲染
                    renderer.pause()
                    AppLog.info("Renderer pause success.")
                } catch (e: BusinessException) {
                    AppLog.error("Renderer pause failed")
                }
            case _ => AppLog.error("Renderer is not running")
        }
    }
}
