            ColorSpace.BT2020_HLG_LIMIT
        } else {
            ColorSpace.BT709_LIMIT
        }
        let colorSpaces: Array<ColorSpace> = getSupportedColorSpaces(session)
        var isSupportedColorSpaces: Bool = false
        for (color in colorSpaces) {
            match ((color, colorSpace)) {
                case (ColorSpace.BT2020_HLG_LIMIT, ColorSpace.BT2020_HLG_LIMIT) =>
                    isSupportedColorSpaces = true
                    break
                case (ColorSpace.BT709_LIMIT, ColorSpace.BT709_LIMIT) =>
                    isSupportedColorSpaces = true
                    break
                case _ => continue
            }
        }
        if (isSupportedColorSpaces) {
            AppLog.info("setColorSpace: ${colorSpace}")
            session.setColorSpace(colorSpace)
            let activeColorSpace: ColorSpace = session.getActiveColorSpace()
            AppLog.info("activeColorSpace: ${activeColorSpace}")
        } else {
            AppLog.info("colorSpace: ${colorSpace} is not support")
        }
    }
