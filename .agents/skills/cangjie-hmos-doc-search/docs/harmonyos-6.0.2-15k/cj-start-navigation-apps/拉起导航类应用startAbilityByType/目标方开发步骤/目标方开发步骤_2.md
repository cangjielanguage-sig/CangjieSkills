public override func onCreate(want: Want, launchParam: LaunchParam): Unit {
        this.uri = want.uri
        // want.parameters是一个json格式的字符串，用户可通过三方json库解析出
        // destinationLatitude、destinationLongitude、destinationName、originName、
        // originLatitude、originLongitude、vehicleType、destinationPoiId、originPoiId字段的值
    }

    public override func onNewWant(want: Want, launchParam: LaunchParam): Unit {
        this.uri = want.uri
        // want.parameters是一个json格式的字符串，用户可通过三方json库解析出
        // destinationLatitude、destinationLongitude、destinationName、originName、
        // originLatitude、originLongitude、vehicleType、destinationPoiId、originPoiId字段的值
        match (this.windowStage) {
            case Some(v) => loadPage(v)
            case None =>
                AppLog.error("windowStage is null")
                this.context.terminateSelf()
        }
    }

    public override func onWindowStageCreate(windowStage: WindowStage): Unit {
        AppLog.info("MainAbility onWindowStageCreate.")
        this.windowStage = windowStage
        loadPage(windowStage)
    }

    private func loadPage(windowStage: WindowStage): Unit {
        AppLog.info("loadPage, uri=${this.uri}")
        if (this.uri == "maps://navigation") {
            // 构建导航场景参数
            localStorage.setOrCreate<?Float64>("destinationLatitude", this.destinationLatitude)
            localStorage.setOrCreate<?Float64>("destinationLongitude", this.destinationLongitude)
            localStorage.setOrCreate<?String>("destinationPoiId", this.destinationPoiId)
            // 拉起导航页面
            windowStage.loadContent("NavigationPage")
        } else if (this.uri == "maps://routePlan") {
            // 构建路径规划场景参数
            localStorage.setOrCreate<?Float64>("destinationLatitude", this.destinationLatitude)
            localStorage.setOrCreate<?Float64>("destinationLongitude", this.destinationLongitude)
            localStorage.setOrCreate<?String>("destinationName", this.destinationName)
            localStorage.setOrCreate<?Float64>("originLatitude", this.originLatitude)
            localStorage.setOrCreate<?Float64>("originLongitude", this.originLongitude)
            localStorage.setOrCreate<?String>("originName", this.originName)
            localStorage.setOrCreate<?Int64>("vehicleType", this.vehicleType)
            localStorage.setOrCreate<?String>("destinationPoiId", this.destinationPoiId)
            localStorage.setOrCreate<?String>("originPoiId", this.originPoiId)
            // 拉起路径规划页面
            windowStage.loadContent("RoutePlanPage")
        } else if (this.uri == 'maps://search') {
            // 构建位置搜索场景参数
            localStorage.setOrCreate<?String>("destinationName", this.destinationName)
            // 拉起位置搜索页面
            windowStage.loadContent("PlaceSearchPage")
        } else {
            // 默认拉起首页
            windowStage.loadContent("EntryView")
        }
    }
}
