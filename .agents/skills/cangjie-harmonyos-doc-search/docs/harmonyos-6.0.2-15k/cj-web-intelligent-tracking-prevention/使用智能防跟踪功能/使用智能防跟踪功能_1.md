# 使用智能防跟踪功能

Web组件支持智能防跟踪功能，即跟踪型网站作为三方插入别的网页时，其发送的网络请求禁止携带cookie。

- 通过调用[enableIntelligentTrackingPrevention](../../API_Reference/source_zh_cn/apis/ArkWeb/cj-apis-webview.md#func-enableintelligenttrackingpreventionbool)接口使能或者关闭相应Web组件的智能防跟踪功能，默认情况下该功能未启用。

    ```cangjie
    // index.cj
    import ohos.state_macro_manage.*
    import kit.ArkWeb.WebviewController
    import kit.UIKit.{Web, BusinessException}

    @Entry
    @Component
    class EntryView {
        let webController = WebviewController()

        func build() {
            Column {
                Button("enableIntelligentTrackingPrevention").onClick {
                    evt => try {
                        webController.enableIntelligentTrackingPrevention(true)
                        AppLog.info("enableIntelligentTrackingPrevention: true")
                    } catch (e: BusinessException) {
                        AppLog.error("enableIntelligentTrackingPrevention ErrorCode: ${e.code},  Message: ${e.message}")
                    }
                }
                Web(src: 'www.example.com', controller: webController)
            }
        }
    }
    ```

- 通过调用[isIntelligentTrackingPreventionEnabled](../../API_Reference/source_zh_cn/apis/ArkWeb/cj-apis-webview.md#func-isintelligenttrackingpreventionenabled)接口判断当前Web组件是否开启了智能防跟踪功能。

    ```cangjie
    // index.cj
    import ohos.state_macro_manage.*
    import kit.ArkWeb.WebviewController
    import kit.UIKit.{Web, BusinessException}

    @Entry
    @Component
    class EntryView {
        let webController = WebviewController()

        func build() {
            Column {
                Button("isIntelligentTrackingPreventionEnabled").onClick {
                    evt => try {
                        let result = webController.isIntelligentTrackingPreventionEnabled()
                        AppLog.info("res: ${result}")
                    } catch (e: BusinessException) {
                        AppLog.error("isIntelligentTrackingPreventionEnabled ErrorCode: ${e.code},  Message: ${e.message}")
                    }
                }
                Web(src: 'www.example.com', controller: webController)
            }
        }
    }
    ```

    同时，智能防跟踪功能提供了一组接口，用于设置需要绕过智能防跟踪功能的域名列表。这些接口设置的域名列表是整个应用生效，而非某个Web组件。

- 通过调用[addIntelligentTrackingPreventionBypassingList](../../API_Reference/source_zh_cn/apis/ArkWeb/cj-apis-webview.md#static-func-addintelligenttrackingpreventionbypassinglistarraystring)接口设置需要绕过智能防跟踪功能的域名列表。

    ```cangjie
    // index.cj
    import ohos.state_macro_manage.*
    import kit.ArkWeb.WebviewController
    import kit.UIKit.{Web, BusinessException}

    @Entry
    @Component
    class EntryView {
        let webController = WebviewController()

        func build() {
            Column {
                Button("addIntelligentTrackingPreventionBypassingList").onClick {
                    evt => try {
                        let hostList: Array<String> = ["www.test1.com", "www.test2.com", "www.test3.com"]
                        WebviewController.addIntelligentTrackingPreventionBypassingList(hostList)
                    } catch (e: BusinessException) {
                        AppLog.error(
                            "addIntelligentTrackingPreventionBypassingList ErrorCode: ${e.code},  Message: ${e.message}")
                    }
                }
                Web(src: 'www.example.com', controller: webController)
            }
        }
    }
    ```
