```cangjie
    func getVideoOutput(cameraManager: CameraManager, videoSurfaceId: String, cameraOutputCapability: CameraOutputCapability): Option<VideoOutput> {
        let videoProfilesArray: Array<VideoProfile> = cameraOutputCapability.videoProfiles
        if (videoProfilesArray.size == 0) {
            AppLog.error("createOutput videoProfilesArray is empty");
            return None
        }
        // AVRecorderProfile.
        let aVRecorderProfile: AVRecorderProfile = AVRecorderProfile(
            ContainerFormatType.CFT_MPEG_4, // 视频文件封装格式，只支持MP4。
            videoBitrate: 100000i32, // 视频比特率。
            videoCodec: CodecMimeType.VIDEO_AVC, // 视频文件编码格式，支持avc格式。
            videoFrameWidth: 640i32, // 视频分辨率的宽。
            videoFrameHeight: 480i32, // 视频分辨率的高。
            videoFrameRate: 30i32 // 视频帧率。
        )
        // 创建视频录制的参数，预览流与录像输出流的分辨率的宽(videoFrameWidth)高(videoFrameHeight)比要保持一致。
        let aVRecorderConfig: AVRecorderConfig = AVRecorderConfig(
            aVRecorderProfile,
            'fd://35',
            videoSourceType: VideoSourceType.VIDEO_SOURCE_TYPE_SURFACE_YUV
        )
        // 创建avRecorder。
        let avRecorder = createAVRecorder()
        // 设置视频录制的参数。
        avRecorder.prepare(aVRecorderConfig)
        // createVideoOutput传入的videoProfile对象的宽高需要和aVRecorderProfile保持一致。
        for (profile in videoProfilesArray) {
            if (Int32(profile.size.width) == aVRecorderProfile.videoFrameWidth.getOrThrow() && Int32(profile.size.height) ==
                aVRecorderProfile.videoFrameHeight.getOrThrow()) {
                // 创建VideoOutput对象。
                let videoOutput = cameraManager.createVideoOutput(profile, videoSurfaceId)
                return videoOutput
            }
        }
        AppLog.error("videoProfile is not found")
        return None
    }
    ```

4. 开始录像。

    先通过videoOutput的[start](../../../API_Reference/source_zh_cn/apis/CameraKit/cj-apis-multimedia-camera.md#func-start-2)方法启动录像输出流，再通过avRecorder的[start](../../../API_Reference/source_zh_cn/apis/MediaKit/cj-apis-multimedia_media.md#func-start)方法开始录像。

    ```cangjie
    func startVideo(videoOutput: VideoOutput, avRecorder: AVRecorder): Unit {
        videoOutput.start()
        avRecorder.start()
    }
    ```

    ```cangjie
    func stopVideo(videoOutput: VideoOutput, avRecorder: AVRecorder): Unit {

    }
    ```
