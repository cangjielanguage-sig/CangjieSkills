### 开发步骤及注意事项

1. 配置音频采集参数并创建AudioCapturer实例，音频采集参数的详细信息可以查看[AudioCapturerOptions](../../../API_Reference/source_zh_cn/apis/AudioKit/cj-apis-multimedia-audio.md#class-audiocaptureroptions)。

   > **说明：**
   >
   > 当设置麦克风音频源（即[SourceType](../../../API_Reference/source_zh_cn/apis/AudioKit/cj-apis-multimedia-audio.md#enum-sourcetype)为SOURCE_TYPE_MIC、SOURCE_TYPE_VOICE_RECOGNITION、SOURCE_TYPE_VOICE_COMMUNICATION、SOURCE_TYPE_VOICE_MESSAGE）时，需要申请麦克风权限ohos.permission.MICROPHONE，申请方式请参见[向用户申请授权](../../security/AccessToken/cj-request-user-authorization.md)。

    ```cangjie
    import ohos.base.*
    import kit.AudioKit.*

    let audioStreamInfo = AudioStreamInfo(
        AudioChannel.CHANNEL_2, // 通道
        AudioEncodingType.ENCODING_TYPE_RAW, // 编码格式
        AudioSampleFormat.SAMPLE_FORMAT_S16LE, // 采样格式
        AudioSamplingRate.SAMPLE_RATE_48000 // 采样率
    )
    let audioCapturerInfo = AudioCapturerInfo(
        SourceType.SOURCE_TYPE_MIC, // 音源类型：麦克风音频源。根据业务场景配置，参考SourceType
        0 // 音频采集器标志
    )
    let audioCapturerOptions = AudioCapturerOptions(audioCapturerInfo, audioStreamInfo)
    let audioCapturer = createAudioCapturer(audioCapturerOptions)
    ```

2. 调用on(AudioCapturerCallbackType.READ_DATA)方法，订阅监听音频数据读入回调。

    请参见[获取UIAbility的上下文信息](../../application-models/cj-uiability-usage.md#获取uiability的上下文信息)获取globalContext。

    ```cangjie
    import ohos.base.*
    import kit.CoreFileKit.*
    import kit.AbilityKit.*

    class ReadDataCallback <: Callback1Argument<Array<Byte>> {
        public func invoke(arg: Array<Byte>) {
            AppLog.info("AudioCapturer callback: ${arg.size}")
            // globalContext在Ability中获取
            let path = globalContext.getOrThrow().cacheDir
            let filePath = path + '/StarWars10s-2C-48000-4SW.pcm'
            let file: File = FileFs.open(filePath, mode: READ_WRITE.mode | CREATE.mode)
            AppLog.info("AudioCapturer ReadDataCallback open success, filePath: ${filePath}")
            let len = FileFs.write(file.fd, arg, options: WriteOptions(length: UIntNative(arg.size), offset: bufferSize))
            AppLog.info("AudioCapturer ReadDataCallback write len: ${len}")
            bufferSize += len
        }
    }

    audioCapturer.on(AudioCapturerCallbackType.READ_DATA, readDataCb)
    ```

3. 调用start()方法进入running状态，开始录制音频。

   ```cangjie
    import ohos.base.*

    try {
        audioCapturer.start()
        AppLog.info("AudioCapturer started")
    } catch (e: BusinessException) {
        AppLog.error("AudioCapturer start errCode: ${e.code}, errMessage: ${e.message}")
    }
    ```

4. 调用stop()方法停止录制。

    ```cangjie
    import ohos.base.*

    try {
        audioCapturer.stop()
        AppLog.info("AudioCapturer stopped")
    } catch (e: BusinessException) {
        AppLog.error("AudioCapturer stop errCode: ${e.code}, errMessage: ${e.message}")
    }
    ```

5. 调用release()方法销毁实例，释放资源。

    ```cangjie
    import ohos.base.*

    try {
        audioCapturer.release()
        AppLog.info("AudioCapturer release success")
    } catch (e: BusinessException) {
        AppLog.error("AudioCapturer release errCode: ${e.code}, errMessage: ${e.message}")
    }
    ```
