### 开发步骤及注意事项

1. 配置音频渲染参数并创建AudioRenderer实例，音频渲染参数的详细信息请参见[AudioRendererOptions](../../../API_Reference/source_zh_cn/apis/AudioKit/cj-apis-multimedia-audio.md#class-audiorendereroptions)。

    ```cangjie
    import kit.AudioKit.*
    import ohos.base.BusinessException

    let audioStreamInfo = AudioStreamInfo(
        AudioChannel.CHANNEL_2, // 通道。
        AudioEncodingType.ENCODING_TYPE_RAW, // 编码格式。
        AudioSampleFormat.SAMPLE_FORMAT_S16LE, // 采样格式。
        AudioSamplingRate.SAMPLE_RATE_48000 // 采样率。
    )

    let audioRendererInfo = AudioRendererInfo(
        StreamUsage.STREAM_USAGE_MUSIC, // 音频流使用类型：音乐。根据业务场景配置，参考StreamUsage。
        0 // 音频渲染器标志。
    )

    let audioRendererOptions = AudioRendererOptions(audioRendererInfo, audioStreamInfo)

    try {
        let audioRenderer = createAudioRenderer(audioRendererOptions)
        AppLog.info("Invoke createAudioRenderer succeeded.")
    } catch (e: BusinessException) {
        AppLog.error("Invoke createAudioRenderer failed, code is ${e.code}, message is ${e.message}")
    }
    ```

2. 调用[on(AudioRendererCallbackType, Callback1ArgumentWithReturn\<Array\<Byte>, AudioDataCallbackResult>)](../../../API_Reference/source_zh_cn/apis/AudioKit/cj-apis-multimedia-audio.md#func-onaudiorenderercallbacktype-callback1argumentwithreturnarraybyte-audiodatacallbackresult)方法，订阅监听音频数据写入回调。该方法支持返回回调结果，系统可以根据开发者返回的值来决定此次回调中的数据是否播放。

    > **注意：**
    >
    > - 能填满回调所需长度数据的情况下，返回AudioDataCallbackResult.VALID，系统会取用完整长度的数据缓冲进行播放。请不要在未填满数据的情况下返回AudioDataCallbackResult.VALID，否则会导致杂音、卡顿等现象。
    >
    > - 在无法填满回调所需长度数据的情况下，建议开发者返回AudioDataCallbackResult.INVALID，系统不会处理该段音频数据，然后会再次向应用请求数据，确认数据填满后返回AudioDataCallbackResult.VALID。
    >
    > - 回调函数结束后，音频服务会把缓冲中数据放入队列里等待播放，因此请勿在回调外再次更改缓冲中的数据。对于最后一帧，如果数据不够填满缓冲长度，开发者需要使用剩余数据拼接空数据的方式，将缓冲填满，避免缓冲内的历史脏数据对播放效果产生不良的影响。
    >
    > - 示例中获取AbilityContext的实现方法，请参见[完整示例](#完整示例)。

    ```cangjie
    import kit.AudioKit.*
    import ohos.base.{BusinessException, Callback1ArgumentWithReturn}
    import kit.CoreFileKit.*

    // 定义监听音频数据写入回调函数
    class WriteDataCallback <: Callback1ArgumentWithReturn<Array<Byte>, AudioDataCallbackResult> {
        public WriteDataCallback(let f: (Array<Byte>) -> AudioDataCallbackResult) {}
        public func invoke(buffer: Array<Byte>): AudioDataCallbackResult {
            f(buffer)
        }
    }

    // 请在工程内实现Global类型，在MainAbility中初始化获取context以获取AbilityContext
    let abilityContext = Global.getAbilityContext()
    let path = abilityContext.cacheDir
    // 确保该沙箱路径下存在该资源。
    let filePath = path + '/StarWars10s-2C-48000-4SW.wav'
    let file = FileFs.open(filePath, mode: READ_ONLY.mode)
    var bufferSize = 0

    let callback: (Array<Byte>) -> AudioDataCallbackResult = {
        buffer: Array<Byte> => try {
            FileFs.read(file.fd, buffer, options: ReadOptions())
            bufferSize += buffer.size
            // 系统会判定buffer有效，正常播放。
            return AudioDataCallbackResult.VALID
        } catch (e: BusinessException) {
            AppLog.error("Error reading file: ${e.toString()}")
            // 系统会判定buffer无效，不播放。
            return AudioDataCallbackResult.INVALID
        }
    }
    audioRenderer.on(AudioRendererCallbackType.AR_WRITE_DATA, WriteDataCallback(callback))
    ```

3. 调用start()方法进入running状态，开始渲染音频。
