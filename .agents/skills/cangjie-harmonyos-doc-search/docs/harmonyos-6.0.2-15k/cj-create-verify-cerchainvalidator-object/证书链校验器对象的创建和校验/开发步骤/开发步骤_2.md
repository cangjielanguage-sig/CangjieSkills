    func certChainValidatorSample(): Unit {
        // 证书链校验器算法。目前仅支持PKIX。
        let algorithm = 'PKIX'

        // 创建一个证书链校验器实例。
        let validator = createCertChainValidator(algorithm)

        let bs: Array<UInt8> = [183, 4, 45, 45, 45, 45] //example data
        let certChainData = CertChainData(
            // Uint8Array类型：L-V格式（证书数据长度-证书数据）。
            bs,
            // 证书的数量。本例中为2。
            2,
            // 证书格式。仅支持 PEM 和 DER。在此示例中，证书为 PEM 格式。
            EncodingFormat.FORMAT_PEM
        )
        try {
            // 验证证书链。
            validator.validate(certChainData)
            AppLog.info('validate success')
        } catch (e: BusinessException) {
            // 校验失败。
            AppLog.error('validate failed, errCode: ${e.code}, errMsg: ${e.message}')
        }
    }
