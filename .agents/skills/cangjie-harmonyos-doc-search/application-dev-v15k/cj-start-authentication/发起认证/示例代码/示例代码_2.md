// 设置认证参数
let reuseUnlockResult = ReuseUnlockResult(ReuseMode.AUTH_TYPE_IRRELEVANT, 300000)
try {
    let rand = createRandom()
    let len: Int32 = 16 // Generate a 16-byte random number
    let randData = rand.generateRandom(len).data
    // 设置认证参数
    let authParam = AuthParam(randData, [UserAuthType.FACE], AuthTrustLevel.ATL3, reuseUnlockResult: reuseUnlockResult)
    // 配置认证界面
    let widgetParam = WidgetParam('请进行身份认证','')
    // 获取认证对象
    let userAuthInstance = getUserAuthInstance(authParam, widgetParam)
    AppLog.info('get userAuth instance success')
    // 订阅认证结果
    userAuthInstance.on('result', {
        userAuthResult: UserAuthResult =>
        AppLog.info('userAuthInstance callback result: ${userAuthResult.result}')
        // 可在认证结束或其他业务需要场景，取消订阅认证结果
        userAuthInstance.off('result')
    })
    AppLog.info('auth on success')
    userAuthInstance.start()
    AppLog.info('auth start success')
} catch (error:BusinessException) {
    AppLog.error('auth catch error. Code is ${error.code}, message is ${error.message}')
}
