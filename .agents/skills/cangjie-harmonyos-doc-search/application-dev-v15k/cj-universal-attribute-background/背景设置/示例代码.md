```cangjie
package ohos_app_cangjie_entry

import kit.UIKit.*
import ohos.state_macro_manage.*
import kit.LocalizationKit.*

@Entry
@Component
class EntryView {
    func build() {
        Column(5) {
            Text("background color").fontSize(9).width(90.percent).fontColor(0xCCCCCC)
            Row().width(90.percent).height(50).backgroundColor(0xE5E5E5).border(width: 1.vp)

            Text("background image repeat along X").fontSize(9).width(90.percent).fontColor(0xCCCCCC)
            Row().backgroundImage(src: @r(app.media.background), repeat: ImageRepeat.X).backgroundImageSize(
                width: 250.px, height: 140.px).width(90.percent).height(70).border(width: 1.vp)

            Text("background image repeat along Y").fontSize(9).width(90.percent).fontColor(0xCCCCCC)
            Row().backgroundImage(src: @r(app.media.background), repeat: ImageRepeat.Y).backgroundImageSize(
                width: 500.px, height: 120.px).width(90.percent).height(100).border(width: 1.vp)

            Text("background image size").fontSize(9).width(90.percent).fontColor(0xCCCCCC)
            Row().width(90.percent).height(150).backgroundImage(src: @r(app.media.background),
                repeat: ImageRepeat.NoRepeat).backgroundImageSize(width: 1000, height: 500).border(width: 1.vp)

            Text("background fill the box(Cover)").fontSize(9).width(90.percent).fontColor(0xCCCCCC)
            // 不保准图片完整的情况下占满盒子
            Row().width(200).height(50).backgroundImage(src: @r(app.media.background), repeat: ImageRepeat.NoRepeat).
                backgroundImageSize(ImageSize.Cover).border(width: 1.vp)

            Text("background fill the box(Contain)").fontSize(9).width(90.percent).fontColor(0xCCCCCC)
            // 保准图片完整的情况下放到最大
            Row().width(200).height(50).backgroundImage(src: @r(app.media.background), repeat: ImageRepeat.NoRepeat).
                backgroundImageSize(ImageSize.Contain).border(width: 1.vp)

            Text("background image position").fontSize(9).width(90.percent).fontColor(0xCCCCCC)
            Row().width(100).height(50).backgroundImage(src: @r(app.media.background), repeat: ImageRepeat.NoRepeat).
                backgroundImageSize(width: 1000, height: 560).backgroundImagePosition(x: -500, y: -300).border(
                width: 1.vp)
        }.width(100.percent).height(100.percent).padding(top: 5)
    }
}
```
