## 示例代码

```cangjie
package ohos_app_cangjie_entry

import kit.UIKit.*
import ohos.state_macro_manage.*

extend LocationButtonOnClickResult {
    func toString(): String {
        return match (this) {
            case Success => "Success"
            case _ => "TemporaryAuthorizationFailed"
        }
    }
}

@Entry
@Component
class EntryView {
    func build() {
        Row() {
            Column(10) {
                LocationButton().onClick(
                    {
                    event: ClickEvent, result: LocationButtonOnClickResult => AppLog.info("result:" + result.toString())
                })
                LocationButton(
                    icon: LocationIconStyle.Lines,
                    text: LocationDescription.CurrentLocation,
                    buttonType: ButtonType.Capsule
                ).backgroundColor(0x10007dff)
                LocationButton(
                    icon: LocationIconStyle.Lines,
                    text: LocationDescription.PunchIn,
                    buttonType: ButtonType.Capsule
                ).fontSize(16).width(100).backgroundColor(Color.GREEN)
                LocationButton(
                    icon: LocationIconStyle.Lines,
                    text: LocationDescription.Locating,
                    buttonType: ButtonType.Capsule
                ).fontSize(16).size(width: 80, height: 30).backgroundColor(Color.RED)
                LocationButton(
                    icon: LocationIconStyle.Lines,
                    text: LocationDescription.AddLocation,
                    buttonType: ButtonType.Capsule
                ).fontSize(16).constraintSize(
                    minWidth: 0,
                    maxWidth: 100,
                    minHeight: 0,
                    maxHeight: 30
                )
                LocationButton(
                    icon: LocationIconStyle.Lines,
                    text: LocationDescription.ShareLocation,
                    buttonType: ButtonType.Capsule
                ).fontSize(16).size(width: 100, height: 35).backgroundColor(Color.ORANGE)
                LocationButton(
                    icon: LocationIconStyle.Lines,
                    text: LocationDescription.SendLocation,
                    buttonType: ButtonType.Capsule
                )
                LocationButton(
                    icon: LocationIconStyle.Lines,
                    text: LocationDescription.SelectLocation,
                    buttonType: ButtonType.Capsule
                )
                LocationButton(
                    icon: LocationIconStyle.Lines,
                    text: LocationDescription.SendCurrentLocation,
                    buttonType: ButtonType.Capsule
                )
            }.width(100.percent)
        }.height(100.percent)
    }
}
```
