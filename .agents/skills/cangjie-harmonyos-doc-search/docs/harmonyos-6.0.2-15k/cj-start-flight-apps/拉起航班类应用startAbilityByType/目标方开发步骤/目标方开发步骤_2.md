private func loadPage(windowStage: WindowStage): Unit {
        AppLog.info("loadPage, uri=${this.uri}")
        if (this.uri == "flight://queryByFlightNo") {
            // 构建按航班号查询场景参数
            localStorage.setOrCreate<?String>("flightNo", this.flightNo)
            localStorage.setOrCreate<?String>("departureDate", this.departureDate)
            // 拉起按航班号查询页面
            windowStage.loadContent("QueryByFlightNoPage")
        } else if (this.uri == "finance://queryByLocation") {
            // 构建按起降地查询场景参数
            localStorage.setOrCreate<?String>("originLocation", this.originLocation)
            localStorage.setOrCreate<?String>("destinationLocation", this.destinationLocation)
            localStorage.setOrCreate<?String>("departureDate", this.departureDate)
            // 拉起按起降地查询页面
            windowStage.loadContent("QueryByLocationPage")
        } else {
            // 默认拉起首页
            windowStage.loadContent("EntryView")
        }
    }
}
