## 完整示例

参考以下示例，完成“开始录制-暂停录制-恢复录制-停止录制”的完整流程。

```cangjie
import kit.MediaKit.*
import kit.CoreFileKit.*
import kit.AudioKit.*
import ohos.base.*

//初始化context，可以通过MainAbility获取AbilityContext
var ctx = None<UIAbilityContext>

class AudioRecorderDemo {
    private var avRecorder: ?AVRecorder = None
    private var avProfile: AVRecorderProfile = AVRecorderProfile(
        ContainerFormatType.CFT_MPEG_4A, // 封装格式，当前支持MP4，M4A，MP3，WAV
        audioBitrate: 100000, // 音频比特率
        audioChannels: 2, // 音频声道数
        audioCodec: CodecMimeType.AUDIO_AAC, // 音频编码格式，当前支持ACC，MP3，G711MU
        audioSampleRate: 48000 // 音频采样率
    )
    private var avConfig: AVRecorderConfig = AVRecorderConfig(
        this.avProfile,
        "fd://35", // 参考应用文件访问与管理开发示例新建并读写一个文件
        audioSourceType: AudioSourceType.AUDIO_SOURCE_TYPE_MIC // 音频输入源，这里设置为麦克风
    )
    private var uriPath: String = ""
    private var filePath: String = ""

    // 创建文件以及设置avConfig.url
    func createAndSetFd(): Unit {
        let path: String = ctx.getOrThrow().filesDirectory + "/example.mp3" // 文件沙箱路径，文件后缀名应与封装格式对应
        let audioFile: File = FileFs.open(path, mode: READ_WRITE.mode | CREATE.mode)
        this.avConfig.url = "fd://" + audioFile.fd.toString() // 更新url
        this.filePath = path
    }

    // 注册audioRecorder回调函数
    func setAudioRecorderCallback() {
        if (this.avRecorder.isSome()) {
            // 用户可以在此补充状态发生切换后想要进行的动作
            this.avRecorder.getOrThrow().on(AVRecorderCallbackType.AVRECORDER_STATE_CHANGE, StateChangeCallback())
            // 错误上报回调函数
            this.avRecorder.on(AVRecorderCallbackType.AVRECORDER_ERROR, ErrCallBack())
        }
    }

    // 开始录制对应的流程
    func startRecordingProcess() {
        if (this.avRecorder.isSome()) {
            this.avRecorder.getOrThrow().release()
            this.avRecorder = None
        }
        // 1.创建录制实例
        this.avRecorder = createAVRecorder()
        this.setAudioRecorderCallback()
        // 2.获取录制文件fd赋予avConfig里的url；参考FilePicker文档
        match (this.avRecorder) {
            case Some(v) =>
                // 3.配置录制参数完成准备工作
                v.prepare(this.avConfig)
                // 4.开始录制
                v.start()
            case _ => ()
        }
    }

    // 暂停录制对应的流程
    func pauseRecordingProcess() {
        match (this.avRecorder) {
            case Some(v) => if (v.state == AVRecorderState.STARTED) {
                v.pause()
            }
            case _ => ()
        }
    }

    // 恢复录制对应的流程
    func resumeRecordingProcess() {
        match (this.avRecorder) {
            case Some(v) => if (v.state == AVRecorderState.PAUSED) {
                v.resume()
            }
            case _ => ()
        }
    }

    // 停止录制对应的流程
    func stopRecordingProcess() {
        match (this.avRecorder) {
            case Some(v) =>
                // 仅在started或者paused状态下调用stop为合理状态切换
                if (v.state == AVRecorderState.STARTED || v.state == AVRecorderState.PAUSE) {
                    v.stop()
                }
                v.reset()
                v.release()
            case _ => ()
        }
        this.avRecorder = None
    }

    // 一个完整的【开始录制-暂停录制-恢复录制-停止录制】示例
    func audioRecorderDemo() {
        this.startRecordingProcess() // 开始录制

        // 用户此处可以自行设置录制时长，例如通过设置休眠阻止代码执行
        this.pauseRecordingProcess() //暂停录制
        this.resumeRecordingProcess() // 恢复录制
        this.stopRecordingProcess() // 停止录制
    }
}

class StateChangeCallback <: OnAVPlayerStateChangeHandle {
    public init() {}
    public open func invoke(state: AVPlayerState, reason: StateChangeReason): Unit {
        AppLog.info("current state is ${state.toString()}")
        // 用户可以在此补充状态发生切换后想要进行的动作
    }
}

class ErrCallBack <: Callback1Argument<BusinessException> {
    public init(avPlayer: AVPlayer) {}
    public open func invoke(err: BusinessException): Unit {
        AppLog.error("Invoke avPlayer failed, code is ${err.code}, message is ${err.message}")
        // 自行实现错误上报回调逻辑
    }
}
```
