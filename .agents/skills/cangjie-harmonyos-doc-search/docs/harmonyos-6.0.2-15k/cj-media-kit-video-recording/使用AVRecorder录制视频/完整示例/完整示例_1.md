import kit.CoreFileKit.*
import kit.AudioKit.*
import ohos.base.*

const TAG = "VideoRecorderDemo:"

//初始化context，可以通过MainAbility获取AbilityContext
var ctx = None<UIAbilityContext>

class VideoRecorderDemo {
    private var avRecorder: ?AVRecorder = None
    private var videoOutSurfaceId: String = ""
    private var avProfile: AVRecorderProfile = AVRecorderProfile(
        ContainerFormatType.CFT_MPEG_4, // 视频文件封装格式，只支持MP4
        videoBitrate: 100000, // 视频比特率
        videoCodec: CodecMimeType.VIDEO_AVC, // 视频文件编码格式，支持avc格式
        videoFrameWidth: 640, // 视频分辨率的宽
        videoFrameHeight: 480, // 视频分辨率的高
        videoFrameRate: 30 // 视频帧率
    )
    private var avConfig: AVRecorderConfig = AVRecorderConfig(
        this.avProfile,
        "fd://35", //  参考应用文件访问与管理开发示例新建并读写一个文件
        videoSourceType: VideoSourceType.VIDEO_SOURCE_TYPE_SURFACE_YUV, // 视频源类型，支持YUV和ES两种格式
    )

    private var uriPath: String = "" // 文件uri，可用于安全控件保存媒体资源
    private var filePath: String = "" // 文件路径
    private var fileFd: Int32 = 0

    // 创建文件以及设置avConfig.url
    func createAndSetFd() {
        var path: String = ctx.getOrThrow().filesDirectory + "/example.mp4" // 文件沙箱路径，文件后缀名应与封装格式对应
        var videoFile: File = FileFs.open(path, mode: READ_WRITE.mode | CREATE.mode)
        this.avConfig.url = "fd://" + videoFile.fd.toString() // 设置url
        this.fileFd = videoFile.fd // 文件fd
        this.filePath = path
    }

    // 注册avRecorder回调函数
    func setAvRecorderCallback() {
        if (this.avRecorder.isSome()) {
            var realAvRecorder = this.avRecorder.getOrThrow()
            // 状态机变化回调函数
            realAvRecorder.on(AVRecorderCallbackType.AVRECORDER_STATE_CHANGE, StateChangeCallback())
            // 错误上报回调函数
            realAvRecorder.on(AVRecorderCallbackType.AVRECORDER_ERROR, ErrCallBack())
        }
    }

    // 相机相关准备工作
    func prepareCamera() {
        // 具体实现查看相机资料
    }

    // 启动相机出流
    func startCameraOutput() {
        // 调用VideoOutput的start接口开始录像输出
    }

    // 停止相机出流
    func stopCameraOutput() {
        // 调用VideoOutput的stop接口停止录像输出
    }

    // 释放相机实例
    func releaseCamera() {
        // 释放相机准备阶段创建出的实例
    }

    // 开始录制对应的流程
    func startRecordingProcess() {
        if (this.avRecorder.isNone()) {
            // 1.创建录制实例；
            this.avRecorder = createAVRecorder()
            this.setAvRecorderCallback()
        }
        var realAvRecorder = this.avRecorder.getOrThrow()
        // 2. 获取录制文件fd；获取到的值传递给avConfig里的url，实现略
        // 3.配置录制参数完成准备工作
        realAvRecorder.prepare(this.avConfig)
        this.videoOutSurfaceId = realAvRecorder.getInputSurface()
        // 4.完成相机相关准备工作
        this.prepareCamera()
        // 5.启动相机出流
        this.startCameraOutput()
        // 6. 启动录制
        realAvRecorder.start()
    }

    // 暂停录制对应的流程
    func pauseRecordingProcess() {
        if (this.avRecorder.isSome()) {
            var realAvRecorder = this.avRecorder.getOrThrow()
            if (realAvRecorder.state.toString() == "started") {
                realAvRecorder.pause()
                this.stopCameraOutput() // 停止相机出流
            }
        }
    }

    // 恢复录制对应的流程
    func resumeRecordingProcess() {
        if (this.avRecorder.isSome()) {
            var realAvRecorder = this.avRecorder.getOrThrow()
            if (realAvRecorder.state.toString() == "paused") {
                this.stopCameraOutput() // 停止相机出流
                realAvRecorder.resume()
            }
        }
    }
