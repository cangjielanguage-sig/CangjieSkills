本节以使用ApplicationContext获取filesDir为例，介绍如何获取应用文件路径，并在对应文件路径下新建文件和读写文件。示例中的context的获取方式见[获取UIAbility的上下文信息](cj-uiability-usage.md#获取uiability的上下文信息)。示例代码如下：

```cangjie
import kit.AbilityKit.UIAbilityContext
import kit.UIKit.{AppLog, Button}
import kit.CoreFileKit.{FileFs, ReadOptions, OpenMode}

// 见获取UIAbility的上下文信息章节
func getContext(): UIAbilityContext {
    return globalContext.getOrThrow()
}

@Entry
@Component
class EntryView {
    @State
    var message: String = "Hello World"

    func build() {
        Row {
            Column {
                Text(this.message)
                // ...
                Button("create file")
                    // ...
                    .onClick(
                    {
                        evt =>
                        let applicationContext = getContext().getApplicationContext()
                        // 获取应用文件路径
                        let filesDir = applicationContext.filesDirectory
                        AppLog.info("filePath: ${filesDir}")

                        // 文件不存在时创建并打开文件，文件存在时打开文件
                        let file = FileFs.open(
                            filesDir + "/test.txt",
                            mode: OpenMode.READ_WRITE.mode | OpenMode.CREATE.mode
                        )

                        // 写入一段内容至文件
                        let writeLen = FileFs.write(file.fd, "Try to write str.")

                        AppLog.info("The length of str is: ${writeLen}")
                        // 创建一个大小为1024字节的ArrayBuffer对象，用于存储从文件中读取的数据
                        let arrayBuffer = Array<Byte>(1024, {_ => 0})
                        // 设置读取的偏移量和长度
                        let readOptions = ReadOptions(offset: 0, length: 1024)

                        // 读取文件内容到ArrayBuffer对象中，并返回实际读取的字节数
                        let readLen = FileFs.read(file.fd, arrayBuffer, options: readOptions)
                        // 将ArrayBuffer对象转换为Buffer对象，并转换为字符串输出

                        AppLog.info("the content of file: ${String.fromUtf8(arrayBuffer[..readLen])}")
                        // 关闭文件
                        FileFs.close(file)
                    }
                )
            }.width(100.percent)
        }.height(100.percent)
    }
}
```
