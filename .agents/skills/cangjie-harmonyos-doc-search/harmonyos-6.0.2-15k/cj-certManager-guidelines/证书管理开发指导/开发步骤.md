## 开发步骤

1. 申请权限。调用证书管理相关接口，需要在module.json5文件的requestPermissions对象中配置ohos.permission.ACCESS_CERT_MANAGER权限。更多配置信息请参见[Stage模型应用程序包结构](../../cj-start/basic-knowledge/application-package-structure-stage.md)。

2. 导入相关模块。

    ```cangjie
    import kit.DeviceCertificateKit.*
    import ohos.base.BusinessException
    ```

3. 安装应用私有凭据，获取应用私有凭据，并使用应用私有凭据进行签名、验签，最后删除应用私有凭据。

    ```cangjie
    import kit.DeviceCertificateKit.*
    import ohos.base.BusinessException

    func certificateManagerSample() {
        /* 安装的凭据数据需要业务赋值，本例数据非凭据数据 */
        let keystore: Array<UInt8> = [0x30, 0x82, 0x04, 0x6a, 0x02, 0x01]

        /* 安装凭据对应的密码，业务赋值 */
        let keystorePwd: String = '123456'
        let appKeyUri: String = ''
        try {
            /* 安装私有凭据 */
            let keyUri = installPrivateCertificate(keystore, keystorePwd, appKeyUri)
        } catch (e: BusinessException) {
            AppLog.error('Failed to install private certificate.Code: ${e.code}, message: ${e.message}')
        }

        try {
            /* srcData为待签名、验签的数据，业务自行赋值 */
            let srcData: Array<UInt8> = [0x86, 0xf7, 0x0d, 0x01, 0x07, 0x01]

            /* 构造签名的属性参数 */
            let signSpec: CMSignatureSpec = CMSignatureSpec(CmKeyPurpose.CM_KEY_PURPOSE_SIGN, CmKeyPadding.CM_PADDING_PSS,
                CmKeyDigest.CM_DIGEST_SHA256)

            /* 签名 */
            let signHandle: CMHandle = `init`(appKeyUri, signSpec)
            update(signHandle, srcData)
            let signRes = finish(signHandle)

            /* 构造验签的的属性参数 */
            let verifySpec: CMSignatureSpec = CMSignatureSpec(CmKeyPurpose.CM_KEY_PURPOSE_VERIFY,
                CmKeyPadding.CM_PADDING_PSS, CmKeyDigest.CM_DIGEST_SHA256)

            /* 验签 */
            let verifyHandle: CMHandle = `init`(appKeyUri, verifySpec)
            update(verifyHandle, srcData)
            finish(verifyHandle, signRes)
            AppLog.info('Succeeded in signing and verifying.')
        } catch (e: BusinessException) {
            AppLog.error('Failed to sign or verify. Code: ${e.code},message: ${e.message}')
        }

        try {
            let keyUri = installPrivateCertificate(keystore, keystorePwd, appKeyUri)
            /* 卸载私有凭据 */
            uninstallPrivateCertificate(keyUri)
        } catch (e: BusinessException) {
            AppLog.error('Failed to uninstall private certificate. Code: ${e.code}, message: ${e.message}')
        }
    }
    ```
