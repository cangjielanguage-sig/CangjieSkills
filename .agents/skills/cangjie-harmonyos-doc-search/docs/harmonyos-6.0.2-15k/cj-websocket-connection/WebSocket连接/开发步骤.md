## 开发步骤

1. 导入需要的webSocket模块。

2. 创建一个WebSocket连接，返回一个WebSocket对象。

3. （可选）订阅WebSocket的打开、消息接收、关闭、Error事件。

4. 根据URL地址，发起WebSocket连接。

5. 使用完WebSocket连接之后，主动断开连接。

```cangjie
import kit.NetworkKit.*
import ohos.base.*

let defaultIpAddress = "ws://"
let ws = createWebSocket()

let openCallback = WebSocketAsyncCallback<HashMap<UInt32, String>>({
    errorCode: Option<AsyncError>, data: Option<HashMap<UInt32, String>> => match (errorCode) {
        case Some(e) => AppLog.error("on open, error: ${e.code}")
        case _ =>
            // 当收到on('open')事件时，可以通过send()方法与服务器进行通信
            ws.send(MessageData.STRING_DATA("Hello, server!"), { err, value=>
                if (let Some(v) <- err) {
                    AppLog.error("Failed to send the message. Err: ${v.code}")
                }
                if (let Some(v) <- value) {
                    AppLog.info("Message send successfully")
                }
            })
    }
})

ws.on(OnOffType.OPEN, openCallback)

let messageCallBack = WebSocketAsyncCallback<MessageData>({
    errorCode: Option<AsyncError>, data: Option<MessageData> => match (errorCode) {
        case Some(e) => AppLog.error("on message, error: ${e.code}")
        case _ => match (data) {
            case Some(value) =>
                AppLog.info("on message, message is ${value}")
                if (value.toString() == "bye") {
                    ws.close({
                        err, value =>
                            if (let Some(v) <- err) {
                                AppLog.error("Failed to close the connection. Err: ${v.code}")
                            }
                            if (let Some(v) <- value) {
                                AppLog.info("Connection closed successfully")
                            }
                    })
                }
            case _ => AppLog.error("on message, error")
        }
    }
})

ws.on(OnOffType.MESSAGE, messageCallBack)

let closeCallBack = WebSocketAsyncCallback<CloseResult>({
    errorCode: Option<AsyncError>, data: Option<CloseResult> => match (errorCode) {
        case Some(e) => AppLog.error("on close, error: ${e.code}")
        case _ => match (data) {
            case Some(value) => AppLog.info("on close, code is ${value.code}, reason is ${value.reason}")
            case _ => AppLog.error("on close, error")
        }
    }
})

ws.on(OnOffType.CLOSE, closeCallBack)

let errorCallBack = WebSocketErrorCallback({
    err: BusinessException => AppLog.error("on error, error: ${err.code}")
})

ws.on(OnOffType.ERROR, errorCallBack)
ws.connect(defaultIpAddress, {
    err, value =>
        if (let Some(v) <- err) {
            AppLog.error("Connection failed. Err: ${v.code}")
        }
        if (let Some(v) <- value) {
            AppLog.info("Connected successfully")
        }
})
```
