### 示例2（自定义样式）

该示例通过配置disappearTextStyle、textStyle、selectedTextStyle、acceptButtonStyle、cancelButtonStyle实现了自定义文本以及按钮样式。

<!-- run -->

```cangjie
package ohos_app_cangjie_entry

import kit.UIKit.*
import ohos.state_macro_manage.*
import std.time.*

@Entry
@Component
class EntryView {
    @State
    var selectedDate: DateTime = DateTime.of(year: 2010, month: Month.of(1), dayOfMonth: 1)

    func build() {
        Column {
            Button("DatePickerDialog").margin(20).onClick {
                DatePickerDialog.show(
                    options: DatePickerDialogOptions(
                        start: DateTime.of(year: 2000, month: Month.of(1), dayOfMonth: 1),
                        end: DateTime.of(year: 2100, month: Month.of(12), dayOfMonth: 31),
                        selected: this.selectedDate,
                        disappearTextStyle: PickerTextStyle(0x297bec, MyFont(size: 20.fp, weight: FontWeight.Bold)),
                        textStyle: PickerTextStyle(Color.BLACK, MyFont(size: 18.fp, weight: FontWeight.Normal)),
                        selectedTextStyle: PickerTextStyle(Color.BLUE, MyFont(size: 26.fp, weight: FontWeight.Regular)),
                        acceptButtonStyle: PickerDialogButtonStyle(
                            `type`: ButtonType.Normal,
                            style: ButtonStyleMode.NORMAL,
                            role: ButtonRole.NORMAL,
                            fontColor: 0xFF5151D8,
                            fontSize: 26.fp,
                            fontWeight: FontWeight.Bolder,
                            fontStyle: FontStyle.Normal,
                            fontFamily: "sans-serif",
                            backgroundColor: 0xA6ACAF,
                            borderRadius: BorderRadiuses(topLeft: 20, topRight: 20, bottomLeft: 20, bottomRight: 20)
                        ),
                        cancelButtonStyle: PickerDialogButtonStyle(
                            `type`: ButtonType.Normal,
                            style: ButtonStyleMode.NORMAL,
                            role: ButtonRole.NORMAL,
                            fontColor: Color.BLUE,
                            fontSize: 16.fp,
                            fontWeight: FontWeight.Normal,
                            fontStyle: FontStyle.Italic,
                            fontFamily: "sans-serif",
                            backgroundColor: 0x50182431,
                            borderRadius: BorderRadiuses(topLeft: 10, topRight: 10, bottomLeft: 10, bottomRight: 10)
                        ),
                        onDateAccept: {
                            value =>
                            this.selectedDate = value
                            AppLog.info(
                                "DatePickerDialog:onDateAccept(): ${value.year}-${value.monthValue}-${value.dayOfMonth}"
                            )
                        },
                        onCancel: {=> AppLog.info("DatePickerDialog:onCancel()")},
                        onDateChange: {
                            value => AppLog.info(
                                "DatePickerDialog:onDateChange(): ${value.year}-${value.monthValue}-${value.dayOfMonth}"
                            )
                        },
                        onDidAppear: {=> AppLog.info("DatePickerDialog:onDidAppear()")},
                        onDidDisappear: {=> AppLog.info("DatePickerDialog:onDidDisappear()")},
                        onWillAppear: {=> AppLog.info("DatePickerDialog:onWillAppear()")},
                        onWillDisappear: {=> AppLog.info("DatePickerDialog:onWillDisappear()")}
                    )
                )
            }
        }.width(100.percent)
    }
}
```
