### 示例4（设置循环滚动）

该示例通过配置loop设置TimePicker是否循环滚动。

<!-- run -->

```cangjie
package ohos_app_cangjie_entry

import kit.UIKit.*
import ohos.state_macro_manage.*
import std.time.*
import ohos.hilog.Hilog

@Entry
@Component
class EntryView {
    @State
    var isLoop: Bool = true
    @State
    var selectedTime: DateTime = DateTime.of(year: 2022, month: Month.of(7), dayOfMonth: 22, hour: 12)
    func build() {
        Column() {
            TimePicker(selected: this.selectedTime).loop(this.isLoop).onChange(
                {
                value => if (value.hour >= 0) {
                    this.selectedTime = DateTime.of(year: 2022, month: Month.of(7), dayOfMonth: 22, hour: value.hour,
                        minute: value.minute)
                    Hilog.info(0, "test", "select current date is: ${value.minute}")
                }
            })

            Row() {
                Text("循环滚动").fontSize(20)
                Toggle(ToggleType.SwitchType, isOn: true).onChange({
                    isOn: Bool => this.isLoop = isOn
                })
            }.position(x: 60.percent, y: 40.percent)
        }
    }
}
```
