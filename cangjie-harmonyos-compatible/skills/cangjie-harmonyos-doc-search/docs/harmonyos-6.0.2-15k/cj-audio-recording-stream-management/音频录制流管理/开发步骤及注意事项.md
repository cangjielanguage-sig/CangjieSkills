## 开发步骤及注意事项

1. 创建AudioStreamManager实例。

    在使用AudioStreamManager的API前，需要使用getStreamManager()创建一个AudioStreamManager实例。

    ```cangjie
    import kit.AudioKit.*
    import ohos.base.*

    let audioManager = getAudioManager()
    let audioStreamManager = audioManager.getStreamManager()
    ```

2. 使用on(AudioStreamManagerCallbackType.CAPTURER_CHANGE)监听音频录制流更改事件。 如果音频流监听应用需要在音频录制流状态变化、设备变化时获取通知，可以订阅该事件。

    ```cangjie
    class Callback <: Callback1Argument<AudioCapturerChangeInfoArray> {
        public func invoke(arg: AudioCapturerChangeInfoArray) {
            AppLog.info("callback array size: ${arg.size}")
            for (i in (0..arg.size)) {
                AppLog.info("## CapChange on is called for element ${i} ##")
                AppLog.info("StreamId for ${i} is: ${arg[i].streamId}")
                AppLog.info("Source for ${i} is: ${arg[i].capturerInfo.source}")
                AppLog.info("Flag  ${i} is: ${arg[i].capturerInfo.capturerFlags}")
                let devDescriptorArr: AudioDeviceDescriptors = arg[i].deviceDescriptors
                for (j in (0..devDescriptorArr.size)) {
                    AppLog.info("Id: ${j} : ${devDescriptorArr[j].id}")
                    AppLog.info("Type: ${j} : ${devDescriptorArr[j].deviceType}")
                    AppLog.info("Role: ${j} : ${devDescriptorArr[j].deviceRole}")
                    AppLog.info("Name: ${j} : ${devDescriptorArr[j].name}")
                    AppLog.info("Address: ${j} : ${devDescriptorArr[j].address}")
                    AppLog.info("SampleRates: ${j} : ${devDescriptorArr[j].sampleRates[0]}")
                    AppLog.info("ChannelCounts ${j} : ${devDescriptorArr[j].channelCounts[0]}")
                    AppLog.info("ChannelMask: ${j} : ${devDescriptorArr[j].channelMasks}")
                }
            }
        }
    }

    audioStreamManager.on(AudioStreamManagerCallbackType.CAPTURER_CHANGE, Callback())
    AppLog.info("on CAPTURER_CHANGE success")
    ```

3. （可选）使用off(AudioStreamManagerCallbackType.CAPTURER_CHANGE)取消监听音频录制流变化。

    ```cangjie
    audioStreamManager.off(AudioStreamManagerCallbackType.CAPTURER_CHANGE)
    AppLog.info("off CAPTURER_CHANGE success")
    ```

4. （可选）使用getCurrentAudioCapturerInfoArray()获取当前音频录制流的信息。

    该接口可获取音频录制流唯一ID、音频录制客户端的UID、音频状态以及音频捕获器的其他信息。

    > **说明：**
    >
    > 对所有音频流状态进行监听的应用需要[声明权限](../../security/AccessToken/cj-declare-permissions.md)：`ohos.permission.USE_BLUETOOTH`，否则无法获得实际的设备名称和设备地址信息，查询到的设备名称和设备地址（蓝牙设备的相关属性）将为空字符串。

    ```cangjie
    func getCurrentAudioCapturerInfoArray() {
        try {
            let infoarr = audioStreamManager.getCurrentAudioCapturerInfoArray()
            for (i in (0..infoarr.size)) {
                AppLog.info("ttt StreamId for ${i} is: ${infoarr[i].streamId}")
                AppLog.info("ttt Source for ${i} is: ${infoarr[i].capturerInfo.source}")
                AppLog.info("ttt Flag  ${i} is: ${infoarr[i].capturerInfo.capturerFlags}")
                let devDescriptorArr = infoarr[i].deviceDescriptors
                for (j in (0..devDescriptorArr.size)) {
                    AppLog.info("ttt Id: ${j} : ${devDescriptorArr[j].id}")
                    AppLog.info("ttt Type: ${j} : ${devDescriptorArr[j].deviceType}")
                    AppLog.info("ttt Role: ${j} : ${devDescriptorArr[j].deviceRole}")
                    AppLog.info("ttt Name: ${j} : ${devDescriptorArr[j].name}")
                    AppLog.info("ttt Address: ${j} : ${devDescriptorArr[j].address}")
                    AppLog.info("ttt SampleRates: ${j} : ${devDescriptorArr[j].sampleRates[0]}")
                    AppLog.info("ttt ChannelCounts ${j} : ${devDescriptorArr[j].channelCounts[0]}")
                    AppLog.info("ttt ChannelMask: ${j} : ${devDescriptorArr[j].channelMasks}")
                }
            }
        } catch (e: BusinessException) {
            AppLog.error("getCurrentAudioCapturerInfoArray failed, errCode: ${e.code}, errMessage: ${e.message}")
        }
    }
    ```
