## 示例代码

### 示例1（注册自定义字体）

<!-- run -->

```cangjie
package ohos_app_cangjie_entry

import kit.UIKit.*
import ohos.state_macro_manage.*
import kit.LocalizationKit.*

@Entry
@Component
class EntryView {
    protected func onAppear() {
        Font.registerFont(
            familyName: "Deyihei",
            familySrc: "/resources/rawfile/SmileySans-Oblique.ttf"
        )
    }

    func build() {
        Row {
            Column {
                Text("HelloWorld").fontFamily("Deyihei")
                Text("HelloWorld")
            }.width(100.percent)
        }.height(100.percent)
    }
}
```

### 示例3（获取字体详细信息）

<!-- run -->

```cangjie
package ohos_app_cangjie_entry

import kit.UIKit.*
import ohos.state_macro_manage.*

@Entry
@Component
class EntryView {
    func build() {
        Row {
            Column {
                Button("HelloWorld").onClick {
                    evt =>
                    let info = Font.getFontByName("HarmonyOS Sans Italic")
                    match (info) {
                        case Some(v) =>
                            AppLog.info("${v.path}")
                            AppLog.info("${v.postScriptName}")
                            AppLog.info("${v.fullName}")
                            AppLog.info("${v.family}")
                            AppLog.info("${v.subfamily}")
                            AppLog.info("${v.weight}")
                            AppLog.info("${v.width}")
                            AppLog.info("${v.italic}")
                            AppLog.info("${v.monoSpace}")
                            AppLog.info("${v.symbolic}")
                        case None => AppLog.info("None")
                    }
                }
            }.width(100.percent)
        }.height(100.percent)
    }
}
```

### 示例4（获取系统的UI字体配置）

<!-- run -->

```cangjie
package ohos_app_cangjie_entry

import kit.UIKit.*
import ohos.state_macro_manage.*

@Entry
@Component
class EntryView {
    func build() {
        Row {
            Column {
                Button("HelloWorld").onClick {
                    evt =>
                    let config = Font.getUIFontConfig()
                    for (fontDir in config.fontDir) {
                        AppLog.info("getUIFontConfig: fontDir = ${fontDir}")
                    }
                    // generic
                    for (generic in config.generic) {
                        AppLog.info("getUIFontConfig: generic.family = ${generic.family}")
                        let aliasList = generic.alias
                        for (alias in aliasList) {
                            AppLog.info("getUIFontConfig: alias.name = ${alias.name}")
                            AppLog.info("getUIFontConfig: alias.weight = ${alias.weight.toString()}")
                        }
                        let adjustList = generic.adjust
                        for (adjust in adjustList) {
                            AppLog.info("getUIFontConfig: adjust.weight = ${adjust.weight.toString()}")
                            AppLog.info("getUIFontConfig: adjust.to = ${adjust.to.toString()}")
                        }
                    }
                    // fallbackGroups
                    for (fallbackGroupInfo in config.fallbackGroups) {
                        AppLog.info("getUIFontConfig: fallbackGroupInfo.fontSetName = ${fallbackGroupInfo.fontSetName}")
                        let fallback = fallbackGroupInfo.fallback
                        for (fallbackInfo in fallback) {
                            AppLog.info("getUIFontConfig: fallback.language = ${fallbackInfo.language}")
                            AppLog.info("getUIFontConfig: fallback.family = ${fallbackInfo.family}")
                        }
                    }
                }
            }.width(100.percent)
        }.height(100.percent)
    }
}
```
