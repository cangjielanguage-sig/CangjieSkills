// 向会话中添加拍照输出流
    photoSession.addOutput(photoOutput)

    // 提交会话配置
    photoSession.commitConfig()

    // 启动会话
    photoSession.start()
    // 判断设备是否支持闪光灯
    let flashStatus: Bool = photoSession.hasFlash()
    AppLog.info("Returned with the flash light support status: ${flashStatus}")

    if (flashStatus) {
        // 判断是否支持自动闪光灯模式
        let flashModeStatus: Bool = photoSession.isFlashModeSupported(FlashMode.FLASH_MODE_AUTO)
        if (flashModeStatus) {
            // 设置自动闪光灯模式
            photoSession.setFlashMode(FlashMode.FLASH_MODE_AUTO)
        }
    }

    // 判断是否支持连续自动变焦模式
    let focusModeStatus: Bool = photoSession.isFocusModeSupported(FocusMode.FOCOS_MODE_CONTINUOUS_AUTO)

    if (focusModeStatus) {
        // 设置连续自动变焦模式
        photoSession.setFocusMode(FocusMode.FOCOS_MODE_CONTINUOUS_AUTO)
    }

    // 获取相机支持的可变焦距比范围
    let zoomRatioRange = photoSession.getZoomRatioRange()
    if (zoomRatioRange.size <= 0) {
        return;
    }
    // 设置可变焦距比
    photoSession.setZoomRatio(zoomRatioRange[0])
    let photoCaptureSetting: PhotoCaptureSetting = PhotoCaptureSetting(
        quality: QualityLevel.QUALITY_LEVEL_HIGH, // 设置图片质量高
        rotation: ImageRotation.ROTATION_0 // 设置图片旋转角度0
    )
    // 使用当前拍照设置进行拍照
    photoOutput.capture(photoCaptureSetting)

    // 需要在拍照结束之后调用以下关闭摄像头和释放会话流程，避免拍照未结束就将会话释放。
    // 停止当前会话
    photoSession.stop()

    // 释放相机输入流
    cameraInput.close()

    // 释放预览输出流
    previewOutput.release()

    // 释放拍照输出流
    photoOutput.release()

    // 释放会话
    photoSession.release()
}
