```cangjie
package ohos_app_cangjie_entry

import kit.UIKit.*
import ohos.state_macro_manage.*

@Entry
@Component
class EntryView {
    @State
    var hoverText: String = "no hover"
    @State
    var mouseText: String = "MouseText"
    @State
    var color: Color = Color.BLUE
    func build() {
        Column(20) {
            Button(this.hoverText).onHover(
                {
                isHover => if (isHover) {
                    this.hoverText = "on hover"
                    this.color = Color.RED
                } else {
                    this.hoverText = "no hover"
                    this.color = Color.BLUE
                }
            }).backgroundColor(this.color)

            Button("onMouse").onMouse(
                {
                    event =>
                    this.mouseText = """
                    onMouse:
                    Button = ${event.button.toString()}
                    Action = ${event.action.toString()}
                    localXY = (${event.x} , ${event.y})
                    screenXY = (${event.screenX} , ${event.screenY})
                    """
                    AppLog.info(this.mouseText)
                }
            )
            Text(this.mouseText)
        }.width(100.percent)
    }
}
```
