### 开启广告过滤

应用可以通过AdsBlockManager提供的[setAdsBlockRules()](../../API_Reference/source_zh_cn/apis/ArkWeb/cj-apis-webview.md#static-func-setadsblockrulesstring-bool)接口设置自定义的easylist过滤规则，并通过Web组件的[enableAdsBlock()](../../API_Reference/source_zh_cn/apis/ArkWeb/cj-apis-webview.md#func-enableadsblockbool)接口使能广告过滤特性。

在下面的示例中，演示了一个应用通过文件选择器选择easylist规则文件，并开启广告过滤功能。

```cangjie
// index.cj
import ohos.state_macro_manage.*
import kit.ArkWeb.{WebviewController, AdsBlockManager}
import kit.UIKit.{Web, BusinessException, Flex, ButtonOptions, ButtonType, AsyncError}
import kit.CoreFileKit.{DocumentSelectOptions, DocumentViewPicker, FileUri}
import kit.AbilityKit.UIAbilityContext

@Entry
@Component
class EntryView {
    let webController = WebviewController()
    let mainUrl: String = 'https://www.example.com'

    func getContext(): UIAbilityContext {
        match (globalAbilityContext) {
            case Some(context) => context
            case _ => throw Exception("can not get globalcontext")
        }
    }

    // 演示点击按钮，通过filepicker打开一个easylist规则文件并设置到Web组件中
    func build() {
        Column {
            Row {
                Flex {
                    // test_case_1
                    Button(ButtonOptions(shape: ButtonType.Capsule)) {
                        Text("setAdsBlockRules")
                    }.onClick {
                        evt => try {
                            let documentSelectionOptions = DocumentSelectOptions()
                            let picker = DocumentViewPicker(getContext())
                            let documentSelectCallback = {
                                errorCode: Option<AsyncError>, data: Option<Array<String>> => match (errorCode) {
                                    case Some(e) => AppLog.info("document select error: errcode is ${e.code}")
                                    case _ => match (data) {
                                        case Some(value) =>
                                            AppLog.info("documentUris is ${value}")
                                            if (value.size > 0) {
                                                let fileRealPath = FileUri(value[0])
                                                AppLog.info(
                                                    'DocumentViewPicker.select successfully, uri: ${fileRealPath}')
                                                AdsBlockManager.setAdsBlockRules(fileRealPath.path, true)
                                            }
                                        case _ => AppLog.info("document select error: data is null")
                                    }
                                }
                            }
                            picker.select(documentSelectCallback, option: documentSelectionOptions)
                        } catch (e: BusinessException) {
                            AppLog.error(
                                "enableIntelligentTrackingPrevention ErrorCode: ${e.code},  Message: ${e.message}")
                        }
                    }
                }
            }
            Web(src: mainUrl, controller: webController).onControllerAttached {
                => try {
                    webController.enableAdsBlock(true)
                } catch (e: BusinessException) {
                    AppLog.error("ErrorCode: ${e.code},  Message: ${e.message}")
                }
            }
        }
    }
}
```

如果存在内置的easylist规则文件，[setAdsBlockRules()](../../API_Reference/source_zh_cn/apis/ArkWeb/cj-apis-webview.md#static-func-setadsblockrulesstring-bool)接口的replace参数可用于设置规则文件的使用策略，replace为true表示不使用内置的easylist规则文件，replace为false表示自定义规则和内置的规则将会同时工作，如果发现内置规则与自定义规则冲突，可使用replace=true禁用内置规则效果。

设置的自定义规则文件将在应用进程内对所有的Web组件生效，是一个应用级全局配置文件，并将持久化，应用重启后可继续工作。
