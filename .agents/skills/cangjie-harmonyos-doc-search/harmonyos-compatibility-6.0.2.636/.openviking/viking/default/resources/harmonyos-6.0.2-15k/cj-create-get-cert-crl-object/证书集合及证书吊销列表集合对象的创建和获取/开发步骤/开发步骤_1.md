## 开发步骤

1. 导入[证书算法库框架模块](../../../API_Reference/source_zh_cn/apis/DeviceCertificateKit/cj-apis-cert.md)。

    ```cangjie
    import kit.DeviceCertificateKit.*
    ```

2. 基于已有的证书数据，调用[createX509Cert](../../../API_Reference/source_zh_cn/apis/DeviceCertificateKit/cj-apis-cert.md#func-createx509certencodingblob)创建X509证书的对象。

3. 基于已有的CRL数据，调用[createX509CRL](../../../API_Reference/source_zh_cn/apis/DeviceCertificateKit/cj-apis-cert.md#func-createx509crlencodingblob)创建X509证书吊销列表的对象。

4. 调用[createCertCRLCollection](../../../API_Reference/source_zh_cn/apis/DeviceCertificateKit/cj-apis-cert.md#func-createcertcrlcollectionarrayx509cert-arrayx509crl)创建[CertCRLCollection](../../../API_Reference/source_zh_cn/apis/DeviceCertificateKit/cj-apis-cert.md#class-certcrlcollection)的对象，并返回相应的结果。

5. 调用[collection.selectCerts](../../../API_Reference/source_zh_cn/apis/DeviceCertificateKit/cj-apis-cert.md#func-selectcertsx509certmatchparameters)查找所有与[X509CertMatchParameters](../../../API_Reference/source_zh_cn/apis/DeviceCertificateKit/cj-apis-cert.md#class-x509certmatchparameters)匹配的证书对象数组，并返回结果。

6. 调用[collection.selectCRLs](../../../API_Reference/source_zh_cn/apis/DeviceCertificateKit/cj-apis-cert.md#func-selectcrlsx509crlmatchparameters)查找所有与[X509CRLMatchParameters](../../../API_Reference/source_zh_cn/apis/DeviceCertificateKit/cj-apis-cert.md#class-x509crlmatchparameters)匹配的证书吊销列表数组，并返回结果。

    ```cangjie
    import kit.DeviceCertificateKit.*
    import ohos.base.BusinessException

    func createX509CRL() {
        let crlData = """
                -----BEGIN X509 CRL-----
                MIHzMF4CAQMwDQYJKoZIhvcNAQEEBQAwFTETMBEGA1UEAxMKQ1JMIGlz c3VlchcN
                MTcwODA3MTExOTU1WhcNMzIxMjE0MDA1MzIwWjAVMBMCAgPoFw0z MjEyMTQwMDUz
                MjBaMA0GCSqGSIb3DQEBBAUAA4GBACEPHhlaCTWA42ykeaOyR0SG QIHIOUR3gcDH
                J1LaNwiL+gDxI9rMQmlhsUGJmPIPdRs9uYyI +f854lsWYisD2PUEpn3DbEvzwYeQ
                5SqQoPDoM+YfZZa23hoTLsu52toXobP74sf/9K501p/ +8hm4ROMLBoRT86GQKY6g
                eavsH0Q3
                -----END  X509    CRL-----
                """

        // 证书吊销列表二进制数据，需业务自行赋值。
        let encodingBlob = EncodingBlob(
            crlData.toArray(),
            // 根据encodingData的格式进行赋值，支持FORMAT_PEM和 FORMAT_DER。
            EncodingFormat.FORMAT_PEM
        )
        var x509CRL: X509CRL
        try {
            x509CRL = createX509CRL(encodingBlob)
            return x509CRL
        } catch (e: BusinessException) {
            AppLog.error('createX509CRL failed, errCode: ${e. code}, errMsg: ${e.message}')
        }
    }

    func createX509Cert(): X509Cert {
        let certData = """

        let encodingBlob = EncodingBlob(
            certData.toArray(),
            // 根据encodingData的格式进行赋值，支持FORMAT_PEM和 FORMAT_DER。
