func build() {
        Column() {
            Button("下划线颜色变化").width(100.percent).margin(bottom: 12.vp).onClick(
                {
                    =>
                    // 对Button组件的宽高属性进行动画配置
                    if (this.colorFlag) {
                        animateTo(
                            AnimateParam(
                                duration: 1000, // 动画时长
                                curve: Curve.Linear, // 动画曲线
                                delay: 200, // 动画延迟
                                iterations: 1, // 播放次数
                                playMode: PlayMode.Normal, // 动画模式
                                onFinish: {=> AppLog.info("color change play end, colorFlag:${this.colorFlag}")}
                            ),
                            {
                                => this.indicatorColor = 0xff0000
                            }
                        )
                    } else {
                        (AnimateParam(
                                duration: 1000, // 动画时长
                                curve: Curve.Linear, // 动画曲线
                                delay: 200, // 动画延迟
                                iterations: 1, // 播放次数
                                playMode: PlayMode.Normal, // 动画模式
                                onFinish: {=> AppLog.info("color change play end, colorFlag:${this.colorFlag}")}
                            ), {
                                => this.indicatorColor = 0xFFBF00
                            })
                    }
                    this.colorFlag = !this.colorFlag
                }
            )
            Button("下划线高度变化").width(100.percent).margin(bottom: 12.vp).onClick(
                {
                    =>
                    // 对Button组件的宽高属性进行动画配置
                    if (this.heightFlag) {
                        animateTo(
                            AnimateParam(
                                duration: 1000, // 动画时长
                                curve: Curve.Linear, // 动画曲线
                                delay: 200, // 动画延迟
                                iterations: 1, // 播放次数
                                playMode: PlayMode.Normal, // 动画模式
                                onFinish: {=> AppLog.info("height change play end, colorFlag:${this.heightFlag}")}
                            ),
                            {
                                => this.indicatorHeight = 20
                            }
                        )
                    } else {
                        animateTo(
                            AnimateParam(
                                duration: 1000, // 动画时长
                                curve: Curve.Linear, // 动画曲线
                                delay: 200, // 动画延迟
                                iterations: 1, // 播放次数
                                playMode: PlayMode.Normal, // 动画模式
                                onFinish: {=> AppLog.info("height change play end, colorFlag:${this.heightFlag}")}
