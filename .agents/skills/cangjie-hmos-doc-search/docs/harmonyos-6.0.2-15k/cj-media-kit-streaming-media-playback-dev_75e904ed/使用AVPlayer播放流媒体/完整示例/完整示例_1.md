## 完整示例

参考以下示例，完整地播放一个流媒体视频。

```cangjie
import kit.MediaKit.*
import kit.CoreFileKit.*
import kit.AudioKit.*
import ohos.base.*
import std.collection.*
import std.convert.*

class AVPlayerDemo {
    private var count: Int32 = 0
    private var surfaceID: String = "" // surfaceID用于播放画面显示，具体的值需要通过Xcomponent接口获取，相关文档链接见上面Xcomponent创建方法
    private var isSeek: Bool = true // 用于区分模式是否支持seek操作
    public var audioTrackList: ArrayList<Int32> = ArrayList<Int32>()
    public var videoTrackList: ArrayList<Int32> = ArrayList<Int32>()

    func constructor(surfaceID: String) {
        this.surfaceID = surfaceID
    }

    // 注册avplayer回调函数
    func setAVPlayerCallback(avPlayer: AVPlayer) {
        // startRenderFrame首帧渲染回调函数
        avPlayer.on(AVPlayerCallbackType.startRenderFrame, StartRenderFrameCallback())
        // seek操作结果回调函数
        avPlayer.on(AVPlayerCallbackType.seekDone, SeekDoneCallBack())
        // error回调监听函数,当avPlayer在操作过程中出现错误时调用 reset接口触发重置流程
        avPlayer.on(AVPlayerCallbackType.error, ErrCallBack(avPlayer))
        // 状态机变化回调函数
        avPlayer.on(AVPlayerCallbackType.stateChange, StateChangeCallback(avPlayer, surfaceID))

        // 监听流媒体缓冲状态、缓冲百分比、已缓冲数据预估可播放时长
        avPlayer.on(AVPlayerCallbackType.bufferingUpdate, BufferingUpdateCallback())
    }

    // 以下demo为通过url设置网络地址来实现播放流媒体HLS点播视频
    func avPlayerVodDemo() {
        // 创建avPlayer实例对象
        var avPlayer: AVPlayer = createAVPlayer()
        // 创建状态机变化回调函数
        this.setAVPlayerCallback(avPlayer)
        this.isSeek = true // 点播支持seek操作
        avPlayer.url = "http://xxx.xxx.xxx.xxx:xx/xx/index.m3u8"
    }

    // 以下demo为通过url设置网络地址来实现播放流媒体HLS直播视频
    func avPlayerLiveDemo() {
        // 创建avPlayer实例对象
        var avPlayer: AVPlayer = createAVPlayer()
        // 创建状态机变化回调函数
        this.setAVPlayerCallback(avPlayer)
        this.isSeek = false // 直播不支持seek操作
        avPlayer.url = "http://xxx.xxx.xxx.xxx:xx/xx/index.m3u8"
    }

    func value2String(vt: ValueType): String {
        match (vt) {
            case INT(v) => v.toString()
            case INT64(v) => v.toString()
            case DOUBLE(v) => v.toString()
            case STRING(v) => v.toString()
        }
    }

    // 以下demo为通过url设置网络地址来实现播放Dash流媒体视频
    func avPlayerDashDemo() {
        // 创建avPlayer实例对象
        var avPlayer = createAVPlayer()
        // 创建状态机变化回调函数
        this.setAVPlayerCallback(avPlayer)
        this.isSeek = true // 表示支持seek操作
        avPlayer.url = "http://test.cn/dash/aaa.mpd" //须替换为DASH资源实际地址

        // 通过selectTrack设置音频/视频轨道，通过deselectTrack取消上次设置的音频/视频轨道并恢复到默认音频/视频轨道
        var arrList = avPlayer.getTrackDescription()
        for (i in 0..arrList.size) {
            var propertyIndex: ValueType = arrList[i][MediaDescriptionKey.MD_KEY_TRACK_INDEX]
            var propertyType: ValueType = arrList[i][MediaDescriptionKey.MD_KEY_TRACK_TYPE]
            if (value2String(propertyType) == "0") {
                this.audioTrackList.add(Int32.parse(value2String(propertyIndex))) // 获取音频轨道列表
            } else if (value2String(propertyType) == "1") {
                this.videoTrackList.add(Int32.parse(value2String(propertyIndex))) // 获取视频轨道列表
            }
        }
        //选择其中一个视频轨道
        avPlayer.selectTrack(this.videoTrackList[0])
        //取消选择的视频轨道
        avPlayer.deselectTrack(this.videoTrackList[0])
    }

    // 以下demo为通过setMediaSource设置自定义头域及媒体播放优选参数实现初始播放参数设置，以流媒体Https点播为例
    func preDownloadDemo() {
        // 创建avPlayer实例对象
        var avPlayer = createAVPlayer()
        // 创建状态机变化回调函数
        this.setAVPlayerCallback(avPlayer)
        this.isSeek = true // 点播支持seek操作
