func previewRotation(previewOutput: PreviewOutput): Unit {
    var initDisplayRotation = getDefaultDisplaySync().rotation
    var initPreviewRotation = previewOutput.getPreviewRotation(Int32(initDisplayRotation * 90))
    previewOutput.setPreviewRotation(initPreviewRotation, isDisplayLocked: false)
    off(LISTNER_TYPE_CAPTURE_STATUS_CHANGE)
    let cb = {
        value: Bool =>
        let rotation = getDefaultDisplaySync().rotation
        let imageRotation = rotation * 90
        let previewRotation = previewOutput.getPreviewRotation(Int32(imageRotation))
        previewOutput.setPreviewRotation(previewRotation, isDisplayLocked: false)
    }
    on(LISTNER_TYPE_CAPTURE_STATUS_CHANGE, TestCallback(cb))
}
