```cangjie
import kit.UIKit.*
import kit.UIKit.Color as UIColor
import kit.ArkGraphics2D.*
import ohos.state_macro_manage.Builder

// 定义两个DisplaySync变量，未初始化
var backDisplaySyncSlow: Option<DisplaySync> = Option<DisplaySync>.None
var backDisplaySyncFast: Option<DisplaySync> = Option<DisplaySync>.None
var isBigger_30: Bool = true
var isBigger_60: Bool = true
var drawFirstSize: Int64 = 25
var drawSecondSize: Int64 = 25

class Callback1 <: Callback1Argument<IntervalInfo> {
    public Callback1() {}
    public open func invoke(arg: IntervalInfo): Unit {
        if (isBigger_30) {
            drawFirstSize += 1
            if (drawFirstSize > 150) {
                isBigger_30 = false
            }
        } else {
            drawFirstSize -= 1
            if (drawFirstSize < 25) {
                isBigger_30 = true
            }
        }
    }
}

class Callback2 <: Callback1Argument<IntervalInfo> {
    public Callback2() {}
    public open func invoke(arg: IntervalInfo): Unit {
        if (isBigger_60) {
            drawSecondSize += 1
            if (drawSecondSize > 150) {
                isBigger_60 = false
            }
        } else {
            drawSecondSize -= 1
            if (drawSecondSize < 25) {
                isBigger_60 = true
            }
        }
    }
}

@Entry
@Component
class EntryView {
    @State
    var drawFirstSize: Int64 = 25
    @State
    var drawSecondSize: Int64 = 25
    @Builder
    func doSomeRenderFirst() {
        Text('30').fontSize(this.drawFirstSize)
    }
    @Builder
    func doSomeRenderSecond() {
        Text('60').fontSize(this.drawSecondSize)
    }
    protected func CreateDisplaySyncSlow() {
        // 定义期望绘制帧率
        let range: ExpectedFrameRateRange = ExpectedFrameRateRange( // 创建和配置帧率参数
            expected: 30,
            min: 0,
            max: 120
        )
        let draw30 = Callback1()
        backDisplaySyncSlow = DisplaySync.create() // 创建DisplaySync实例
        backDisplaySyncSlow.getOrThrow().setExpectedFrameRateRange(range) // 设置帧率
        backDisplaySyncSlow.getOrThrow().on(OnOffType.FRAME, draw30) // 订阅frame事件和注册订阅函数

    }
    func CreateDisplaySyncFast() {
        // 定义期望绘制帧率
        let range: ExpectedFrameRateRange = ExpectedFrameRateRange(
            expected: 60,
            min: 0,
            max: 120
        )

        let draw60 = Callback2()
        backDisplaySyncFast = DisplaySync.create() // 创建DisplaySync实例
        backDisplaySyncFast.getOrThrow().setExpectedFrameRateRange(range) // 设置帧率
        backDisplaySyncFast.getOrThrow().on(OnOffType.FRAME, draw60) // 订阅frame事件和注册订阅函数
    }

    public func aboutToDisappear() {
        if (!backDisplaySyncSlow.isNone()) {
            backDisplaySyncSlow.getOrThrow().stop() // DisplaySync失能关闭
            backDisplaySyncSlow = None // 实例置空
        }
        if (!backDisplaySyncFast.isNone()) {
            backDisplaySyncFast.getOrThrow().stop() // DisplaySync失能关闭
            backDisplaySyncFast = None // 实例置空
        }
    }

    func build() {
        Column() {
            Row {
                this.doSomeRenderFirst()
            }.height(40.percent)
            Row {
                this.doSomeRenderSecond()
            }.height(40.percent)
            Row() {
                Button('Start').id('CustomDrawStart').fontSize(14).fontWeight(FontWeight.W500).margin(bottom: 10,
                    left: 5).fontColor(UIColor.WHITE).onClick(
                    {
                        evt =>
                        if (backDisplaySyncSlow.isNone()) {
                            CreateDisplaySyncSlow()
                        }
                        if (backDisplaySyncFast.isNone()) {
                            CreateDisplaySyncFast()
                        }
                        if (!backDisplaySyncSlow.isNone()) {
                            backDisplaySyncSlow.getOrThrow().start() // DisplaySync使能开启
                        }
                        if (!backDisplaySyncFast.isNone()) {
                            backDisplaySyncFast.getOrThrow().start() // DisplaySync使能开启
                        }
                    }
                ).width(20.percent).height(40).shadow(radius: 10, color: UIColor(0x909399), offsetX: 1, offsetY: 1)

                Button('Stop').id('CustomDrawStop').fontSize(14).fontWeight(FontWeight.W500).margin(bottom: 10, left: 5).
                    fontColor(UIColor.WHITE).onClick(
                    {
                        evt =>
                        if (!backDisplaySyncSlow.isNone()) {
                            backDisplaySyncSlow.getOrThrow().stop() // DisplaySync失能关闭
                        }
                        if (!backDisplaySyncFast.isNone()) {
                            backDisplaySyncFast.getOrThrow().stop() // DisplaySync失能关闭
                        }
                    }
                ).width(20.percent).height(40).shadow(radius: 10, color: UIColor(0x909399), offsetX: 1, offsetY: 1)
            }
        }
    }
}
```
