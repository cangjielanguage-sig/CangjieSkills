# 建立应用侧与前端页面数据通道

前端页面和应用侧之间可以用[createWebMessagePorts()](../../API_Reference/source_zh_cn/apis/ArkWeb/cj-apis-webview.md#func-createwebmessageportsbool)接口创建消息端口来实现两端的通信。

在下面的示例中，应用侧页面中通过createWebMessagePorts方法创建消息端口，再把其中一个端口通过[postMessage()](../../API_Reference/source_zh_cn/apis/ArkWeb/cj-apis-webview.md#func-postmessagestring-arraywebmessageport-string)接口发送到前端页面，便可以在前端页面和应用侧之间互相发送消息。

- 应用侧代码：

    ```cangjie
    // index.cj
    import ohos.state_macro_manage.*
    import kit.LocalizationKit.{__GenerateResource__}
    import kit.UIKit.{Web, AsyncError, AsyncCallback, BusinessException, TextInput}
    import kit.ArkWeb.{WebviewController, WebMessagePort, WebMessage}

    let webController = WebviewController()
    var ports = Array<WebMessagePort>()

    @Entry
    @Component
    class EntryView {
        @State
        var sendFromCangjie: String = 'Send this message from Cangjie to HTML'
        @State
        var receivedFromHtml: String = 'Display received message send from HTML'

        func build() {
            Column() {
                // 展示接收到的来自HTML的内容
                Text(this.receivedFromHtml)
                // 输入框的内容发送到HTML
                TextInput(placeholder: "Send this message from Cangjie to HTML").onChange(
                    {
                    value: String => this.sendFromCangjie = value
                })
                Button("postMessage").onClick {
                    evt =>
                    AppLog.info("postMessage")
                    try {
                        // 1、创建两个消息端口。
                        ports = webController.createWebMessagePorts()
                        // 2、在应用侧的消息端口(如端口1)上注册回调事件。
                        ports[1].onMessageEvent(
                            {
                                message: WebMessage =>
                                var msg = "Got msg from HTML:"
                                match (message) {
                                    case STRING(s) =>
                                        AppLog.info("received string message from html5: " + s)
                                        msg = msg + s
                                    case ARRAY_BUFFER(arr) =>
                                        AppLog.info("received arraybuffer from html5: ${arr}")
                                        msg = msg + "${arr}"
                                    case _ => throw IllegalArgumentException("The type is not supported.")
                                }
                                this.receivedFromHtml = msg
                            }
                        )
                        // 3、将另一个消息端口(如端口0)发送到HTML侧，由HTML侧保存并使用。
                        webController.postMessage("__init_port__", [ports[0]], "*")
                    } catch (e: BusinessException) {
                        AppLog.error("ErrorCode: ${e.code},  Message: ${e.message}")
                    }
                }
                // 4、使用应用侧的端口给另一个已经发送到html的端口发送消息。
                Button("SendDataToHTML").onClick {
                    evt =>
                    AppLog.info("SendDataToHTML")
                    if (ports.size != 0) {
                        ports[1].postMessageEvent(WebMessage.STRING(this.sendFromCangjie))
                    } else {
                        AppLog.error("ports is null, Please initialize first")
                    }
                }

                Web(src: @rawfile("test_page_data_channel.html"), controller: webController).onPageBegin(
                    {
                    evt => AppLog.info("page begin url: ${evt.url}")
                }).onPageEnd({
                    evt => AppLog.info("page end url: ${evt.url}")
                })
            }
        }
    }
    ```
