## 完整示例

参考以下示例，设置文件描述符，获取一个音频的元数据和专辑封面。

```cangjie
// index.cj
import kit.MediaKit.*
import kit.ImageKit.*
import kit.AbilityKit.*
import kit.CoreFileKit.*

var ctx = None<UIAbilityContext>

@Entry
@Component
class Index {
    @State message: String = 'Hello World'
    // pixelMap对象声明，用于图片显示
    @State pixelMap: ?PixelMap = None
    let rootPath = ctx.getOrThrow().getApplicationContext().filesDirectory
    let testFilename = '/cover.mp3'

    func build() {
        Row() {
            Column() {
                Text(this.message).fontSize(50).fontWeight(FontWeight.Bold)
                Button() {
                    Text('TestButton')
                        .fontSize(30)
                        .fontWeight(FontWeight.Bold)
                    }
                .type(ButtonType.Capsule)
                .margin(top: 20.px)
                .backgroundColor(0x0D9FFB)
                .width(60.percent)
                .height(5.percent)
                .onClick {
                    evt =>
                    // 设置fdSrc, 获取音频元数据和专辑封面
                    testFetchMetadataFromFdSrc()
                    // 通过fdSrc获取沙箱路径下音频元数据和专辑封面（文件必须在沙箱路径里存在）
                    testFetchMetadataFromFdSrcPath()
                    // 设置dataSrc, 获取沙箱路径下音频元数据和专辑封面（文件必须在沙箱路径里存在）
                    testFetchMetadataFromDataSrc()
                }

                Image(this.pixelMap).width(300).height(300)
                .margin(top: 20)
            }
            .width(100.percent)
        }
        .height(100.percent)
    }

    // 在以下demo中，使用资源管理接口获取打包在HAP内的媒体资源文件，通过设置fdSrc属性，获取音频元数据并打印，
    // 获取音频专辑封面并通过Image控件显示在屏幕上。
    func testFetchMetadataFromFdSrc() {
        // 创建AVMetadataExtractor对象
        let avMetadataExtractor = createAVMetadataExtractor()

        // 设置fdSrc
        avMetadataExtractor.fdSrc = ctx.getOrThrow().resourceManager.getRawFd('cover.mp3')

        // 获取元数据
        let data = avMetadataExtractor.fetchMetadata()

        //获取专辑封面
        let pMap = avMetadataExtractor.fetchAlbumCover()
        this.pixelMap = pMap
        // 释放资源
        avMetadataExtractor.release()
    }

    // 在以下demo中，使用fs文件系统打开沙箱地址获取媒体文件地址，设置fdSrc属性，获取音频元数据并打印，
    // 获取音频专辑封面并通过Image控件显示在屏幕上。
    func testFetchMetadataFromFdSrcPath() {
        // 创建AVMetadataExtractor对象
        let avMetadataExtractor = createAVMetadataExtractor()

        // 设置fdSrc
        avMetadataExtractor.fdSrc = FileFs.open(this.rootPath + this.testFilename)

        // 获取元数据
        let metadata = fetchMetadata()
        AppLog.info("get meta data, mimeType: ${metadata.mimeType}")

        // 获取专辑封面
        this.pixelMap = fetchAlbumCover()

        // 释放资源
        avMetadataExtractor.release()
        AppLog.info("release data source success.")
    }

    // 在以下demo中，使用fs文件系统打开沙箱地址获取媒体文件地址，设置dataSrc属性，获取音频元数据并打印，
    // 获取音频专辑封面并通过Image控件显示在屏幕上。
    func testFetchMetadataFromDataSrc() {
        let context = ctx.getOrThrow()
        // 通过UIAbilityContext获取沙箱地址filesDir（以Stage模型为例）
        let fd = FileFs.open(this.rootPath + this.testFilename).fd
        let fileSize = FileFs.stat(this.rootPath + this.testFilename).size
        // 设置dataSrc描述符，通过callback从文件中获取资源，写入buffer中
        let dataSrc = AVDataSrcDescriptor(fileSize, DataSrcCallback(fd))

        // 创建AVMetadataExtractor对象
        let avMetadataExtractor = createAVMetadataExtractor()
        // 设置dataSrc
        avMetadataExtractor.dataSrc = dataSrc

        // 获取元数据
        let metadata = fetchMetadata()
        AppLog.info("get meta data, mimeType: ${metadata.mimeType}")

        // 获取专辑封面
        this.pixelMap = fetchAlbumCover()

        // 释放资源
        avMetadataExtractor.release()
        AppLog.info("release data source success.")
    }
}
