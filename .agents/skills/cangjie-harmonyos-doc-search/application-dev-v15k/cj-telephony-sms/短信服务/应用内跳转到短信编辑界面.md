## 应用内跳转到短信编辑界面

发送短信的接口需要系统权限才可调用，三方应用如果有发送短信需求，需要在应用内实现跳转到短信编辑的功能，并且需要携带编辑内容和收件人号码，可以通过调用元能力startAbility接口指定号码并跳转到发送短信页面的方式实现。

```cangjie
// index.cj
import kit.AbilityKit.*
import std.collection.*
import encoding.json.*

const MMS_BUNDLE_NAME = "com.ohos.mms"
const MMS_ABILITY_NAME = "com.ohos.mms.MainAbility"
const MMS_ENTITIES = "entity.system.home"

var ctx = None<UIAbilityContext>

@Entry
@Component
class JumpMessage {

    func startMMSAbilityExplicit() {
        let paramMap = HashMap<String, JsonValue>()
        paramMap.add("pageFlag", JsonString("conversation"))
        // 这里填写短信内容。
        paramMap.add("content", JsonString("我是短信具体内容"))
        // 这里完善联系人和号码；姓名主要是通过手机号来查询实际联系人名称，因此这种方式还是以手机号码为主。
        paramMap.add("contactObjects", JsonString("{\"contactsName\":\"张三\",\"telephone\":13311111111}"))
        let want = Want(
            bundleName: "com.ohos.mms",
            abilityName: "com.ohos.mms.MainAbility",
            parameters: JsonObject(paramMap).toString()
        )

        ctx.getOrThrow().startAbilityForResult(want) {
            err, data =>
                if (let Some(e) <- err>) {
                    AppLog.error("error, code: ${e}")
                }
                if (let Some(d) <- data) {
                    AppLog.info(d.resultCode)
                }
        }
    }

    func build() {
        Row() {
            Column() {
                Button('发送短信')
                    .onClick {
                        evt =>
                        startMMSAbilityExplicit()
                    }
            }
            .width(100.percent)
        }
        .height(100.percent)
    }
}
```

```cangjie
// main_ability.cj
import kit.AbilityKit.{LaunchReason, LaunchParam, Want, UIAbility}
import kit.ArkUI.WindowStage
import kit.UIKit.AppLog

class MainAbility <: UIAbility {
    public init() {
        super()
        registerSelf()
    }

    public override func onCreate(want: Want, launchParam: LaunchParam): Unit {
        AppLog.info("MainAbility OnCreated.${want.abilityName}")
        match (launchParam.launchReason) {
            case LaunchReason.START_ABILITY => AppLog.info("START_ABILITY")
            case _ => ()
        }
    }

    public override func onWindowStageCreate(windowStage: WindowStage): Unit {
        AppLog.info("MainAbility onWindowStageCreate.")
        windowStage.loadContent("EntryView")
        // declared in index.cj
        ctx = this.context
    }
}
```
