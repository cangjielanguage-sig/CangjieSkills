import kit.CryptoArchitectureKit.*

func signMessageBySegment(priKey: PriKey, plainText: Array<UInt8>) {
    let signAlg = "RSA1024|PKCS1|SHA256"
    let signer = createSign(signAlg)
    signer.initialize(priKey)
    let textSplitLen = 64 // 自定义的数据拆分长度，此处取64。
    let size = plainText.size
    for (i in 0..size : textSplitLen) {
        let len = if (i + textSplitLen > size) {
            size
        } else {
            i + textSplitLen
        }
        let updateMessage = plainText[i..len]
        let updateMessageBlob: DataBlob = DataBlob(updateMessage)
        // 分段update。
        signer.update(updateMessageBlob)
    }
    // 已通过分段传入所有明文，故此处sign传入None。
    let signData = signer.sign(None)
    return signData
}

func verifyMessagBySegment(pubKey: PubKey, plainText: Array<UInt8>, signMessageBlob: DataBlob) {
    let verifyAlg = "RSA1024|PKCS1|SHA256"
    let verifier = createVerify(verifyAlg)
    verifier.initialize(pubKey)
    let textSplitLen = 64 // 自定义的数据拆分长度，此处取64。
    let size = plainText.size
    for (i in 0..size : textSplitLen) {
        let len = if (i + textSplitLen > size) {
            size
        } else {
            i + textSplitLen
        }
        let updateMessage = plainText[i..len]
        let updateMessageBlob: DataBlob = DataBlob(updateMessage)
        // 分段update。
        verifier.update(updateMessageBlob)
    }
    // 已通过分段传入所有明文，故此处verify第一个参数传入None。
    let res = verifier.verify(None, signMessageBlob)
    AppLog.info("verify result is ${res}")
    return res
}
