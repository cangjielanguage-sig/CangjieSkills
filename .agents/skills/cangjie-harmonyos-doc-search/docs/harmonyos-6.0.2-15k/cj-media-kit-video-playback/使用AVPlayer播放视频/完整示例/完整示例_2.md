    func preDownloadDemo() {
        // 创建avPlayer实例对象
        var avPlayer: AVPlayer = createAVPlayer()
        // 创建mediaSource实例对象，设置媒体来源，定制HTTP请求，如需要，可以键值对的形式设置User-Agent、Cookie、Referer等字段
        var mediaSource: MediaSource = createMediaSourceWithUrl("http://xxx",
            headers: HashMap<String, String>([("User-Agent", "User-Agent-Value")]))
        var playbackStrategy: PlaybackStrategy = PlaybackStrategy(preferredWidth: 1, preferredHeight: 2,
            preferredBufferDuration: 3, preferredHdr: false)
        // 设置媒体来源和播放策略
        avPlayer.setMediaSource(mediaSource, strategy: playbackStrategy)
    }

    func value2String(vt: ValueType): String {
        match (vt) {
            case INT(v) => v.toString()
            case INT64(v) => v.toString()
            case DOUBLE(v) => v.toString()
            case STRING(v) => v.toString()
        }
    }

    func multiTrackDemo() {
        // 创建avPlayer实例对象
        var avPlayer: AVPlayer = createAVPlayer()
        var audioTrackIndex: ValueType
        var arrList = avPlayer.getTrackDescription()
        for (i in 0..arrList.size) {
            if (i != 0) {
                // 获取音频轨道列表
                audioTrackIndex = arrList[i][MediaDescriptionKey.MD_KEY_TRACK_INDEX]
                // 选择其中一个音频轨道
                avPlayer.selectTrack(Int32.parse(value2String(audioTrackIndex)))
                // 取消选择上次选中的音频轨道，并恢复到默认音频轨道。
                avPlayer.deselectTrack(Int32.parse(value2String(audioTrackIndex)))
            }
        }
    }

// error监听回调函数类
class ErrCallBack <: Callback1Argument<BusinessException> {
    let avPlayer_: AVPlayer
    public init(avPlayer: AVPlayer) {
        this.avPlayer_ = avPlayer
    }
    public open func invoke(err: BusinessException): Unit {
        AppLog.error("Invoke avPlayer failed, code is ${err.code}, message is ${err.message}")
        avPlayer_.reset()
    }
}

// 初始化AVDataSrcDescriptor使用的回调函数类
class avPlayerDataSrcSeekDemoCallback <: Callback3ArgumentWithReturn<Array<UInt8>, UInt32, Int64, Int32> {
    let fd: Int32
    let fileSize: Int64
    public init(fd: Int32, fileSize: Int64) {
        this.fd = fd
        this.fileSize = fileSize
    }
    public open func invoke(buf: Array<Byte>, length: UInt32, pos: Int64): Int32 {
        var num = 0
        num = FileFs.read(this.fd, buf, options: ReadOptions(offset: pos, length: UIntNative(length)))
        if (num > 0 && (this.fileSize >= pos)) {
            return Int32(num)
        }
        return -1
    }
}
