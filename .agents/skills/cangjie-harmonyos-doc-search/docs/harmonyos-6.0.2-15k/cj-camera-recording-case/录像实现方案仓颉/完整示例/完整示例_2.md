// videoProfile的宽高需要与AVRecorderProfile的宽高保持一致，并且需要使用AVRecorderProfile所支持的宽高。
    // 示例代码默认选择第一个videoProfile，实际开发需根据所需筛选videoProfile。
    let videoProfile: VideoProfile = videoProfilesArray[0]
    let isHdr = videoProfile.format == CameraFormat.CAMERA_FORMAT_YCBCR_P010 || videoProfile.format == CameraFormat.
        CAMERA_FORMAT_YCRCB_P010
    // 配置参数以实际硬件设备支持的范围为准。
    let aVRecorderProfile: AVRecorderProfile = AVRecorderProfile(
        ContainerFormatType.CFT_MPEG_4,
        audioBitrate: 48000,
        audioChannels: 2,
        audioCodec: CodecMimeType.AUDIO_AAC,
        audioSampleRate: 48000,
        videoBitrate: 2000000,
        videoCodec: if (isHdr) {
            CodecMimeType.VIDEO_HEVC
        } else {
            CodecMimeType.VIDEO_AVC
        },
        videoFrameWidth: Int32(videoProfile.size.width),
        videoFrameHeight: Int32(videoProfile.size.height),
        videoFrameRate: 30,
        isHdr: isHdr
    )
    let videoUri: String = "file://${context.filesDirectory}/${DateTime.now()}.mp4" // 本地沙箱路径。
    let file: File = FileFs.open(videoUri, mode: OpenMode.READ_WRITE.mode | OpenMode.CREATE.mode)
    let aVRecorderConfig: AVRecorderConfig = AVRecorderConfig(
        aVRecorderProfile,
        "fd://${file.fd.toString()}", // 文件需先由调用者创建，赋予读写权限，将文件fd传给此参数，eg.fd://45--file:///data/media/01.mp4
        audioSourceType: AudioSourceType.AUDIO_SOURCE_TYPE_MIC,
        videoSourceType: VideoSourceType.VIDEO_SOURCE_TYPE_SURFACE_YUV
    )

    let avRecorder: AVRecorder = createAVRecorder()
    avRecorder.prepare(aVRecorderConfig)

    let videoSurfaceId: String = avRecorder.getInputSurface()
    // 创建VideoOutput对象。
    let videoOutput: VideoOutput = cameraManager.createVideoOutput(videoProfile, videoSurfaceId)
    // 监听视频输出错误信息。
    videoOutput.on(CameraCallbackType.CameraError, VideoOutputErrorCallBack())

    //创建会话。
    let videoSession: VideoSession = (cameraManager.createSession(SceneMode.NORMAL_VIDEO) as VideoSession).getOrThrow()
    // 监听session错误信息。
    videoSession.on(CameraCallbackType.CameraError, VideoSessionErrorCallBack())

    // 开始配置会话。
    videoSession.beginConfig()

    // 创建相机输入流。
    let cameraInput: CameraInput = cameraManager.createCameraInput(cameraArray[0])
    // 监听cameraInput错误信息。
    let cameraDevice: CameraDevice = cameraArray[0]
    cameraInput.on(CameraCallbackType.CameraError, cameraDevice, CameraInputErrorCallBack())

    // 打开相机。
    cameraInput.open()

    // 向会话中添加相机输入流。
    videoSession.addInput(cameraInput)

    // 创建预览输出流，其中参数 surfaceId 参考下面 XComponent 组件，预览流为XComponent组件提供的surface。
    var previewProfile: Option<Profile> = None
    for (profile in previewProfilesArray) {
        if (abs(
            (Float64(profile.size.width) / Float64(profile.size.height)) - (Float64(videoProfile.size.width) / Float64(videoProfile.
            size.height))) < 2.2204460492503130808472633361816e-16) {
            previewProfile = profile
            break
        }
    } // 筛选与录像分辨率宽高比一致的预览分辨率。
    if (previewProfile.isNone()) {
        return
    }
    let previewOutput = cameraManager.createPreviewOutput(previewProfile.getOrThrow(), surfaceId)

    // 向会话中添加预览输出流。
    videoSession.addOutput(previewOutput)

    // 向会话中添加录像输出流。
    videoSession.addOutput(videoOutput)

    // 提交会话配置。
    videoSession.commitConfig()

    // 启动会话。
    videoSession.start()

    // 启动录像输出流。
    videoOutput.start()

    // 开始录像。
    avRecorder.start()

    // 停止录像输出流。
    videoOutput.stop()

    // 停止录像。
    avRecorder.stop()

    // 停止当前会话。
    videoSession.stop()

    // 关闭文件。
    FileFs.close(file)

    // 释放相机输入流。
    cameraInput.close()

    // 释放预览输出流。
    previewOutput.release()

    // 释放录像输出流。
    videoOutput.release()
