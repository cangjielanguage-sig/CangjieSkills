@Entry
@Component
class EntryView {
    @State
    var message: String = "selectImg"
    @State
    var originalImage: String = ""
    @State
    var editedImage: ?PixelMap = Option<PixelMap>.None
    private var filePath = ""

    // 根据uri读取图片内容
    func readImageByUri(uri: String): ?PixelMap {
        let file = try {
            FileFs.open(
                uri,
                mode: OpenMode.READ_ONLY.mode
            )
        } catch (e: BusinessException) {
            AppLog.error("Open file failed: ${e.message}")
            return None
        }

        try {
            AppLog.info("Original image file id: ${file.fd}")
            let imageSourceApi = createImageSource(file.fd)
            let pixmap = imageSourceApi.createPixelMap()
            editedImage = pixmap
            return pixmap
        } catch (e: BusinessException) {
            AppLog.error("ReadImage failed: ${e.message}")
        } finally {
            FileFs.close(file)
        }
        return None
    }

    // 图库中选取图片

    func build() {
        Row {
            Column {
                Button("selectImg").onClick(
                    {
                        evt =>
                        let context = getContext()
                        let photoResultCallback = {
                            errorCode: Option<AsyncError>, data: Option<PhotoSelectResult> => match (errorCode) {
                                case Some(e) => AppLog.error("photo select error: errcode is ${e.code}")
                                case _ => match (data) {
                                    case Some(value) =>
                                        let uri = value.photoUris[0]
                                        let file = try {
                                            FileFs.open(
                                                uri,
                                                mode: OpenMode.READ_ONLY.mode
                                            )
                                        } catch (e: BusinessException) {
                                            AppLog.error("Open file failed: ${e.message}")
                                            return
                                        }
                                        try {
                                            let timeStamp = DateTime.now()
                                            // 将用户图片拷贝到应用沙箱路径
                                            FileFs.copyFile(file.fd,
                                                context.filesDirectory + "/original-${timeStamp}.jpg")
                                            this.filePath = context.filesDirectory + "/original-${timeStamp}.jpg"
                                            this.originalImage = FileUri.getUriFromPath(this.filePath)
                                        } catch (e: BusinessException) {
                                            AppLog.error("readImage failed:${e.message}")
                                        } finally {
                                            FileFs.close(file.fd)
                                        }

                                    case _ => AppLog.error("photo select error: data is null")
                                }
                            }
                        }

                        var photoSelectOptions = PhotoSelectOptions()
                        photoSelectOptions.MIMEType = PhotoViewMIMETypes.IMAGE_TYPE
                        photoSelectOptions.maxSelectNumber = 1
                        let photoPicker = PhotoViewPicker(context)
                        photoPicker.select(photoResultCallback, option: photoSelectOptions)
                    }
                ).margin(20)

                Button("editImg").onClick(
                    {
                        evt =>
                        let context = getContext()
                        let callBack = AbilityStartCallback(
                            {
                                code, name, message => AppLog.info(
                                    "onError code ${code} name: ${name} message: ${message}")
                            },
                            onResult: {
                                // 获取到回调结果中编辑后的图片uri并做对应的处理
                                result => editedImage = readImageByUri(result.want.uri)
                            }
                        )
                        // 将图片转换为图片uri，并调用startAbilityByType拉起图片编辑应用面板
                        let uri = FileUri.getUriFromPath(this.filePath)

                        let flag = Int64(Flags.FLAG_AUTH_READ_URI_PERMISSION.getValue())

                        try {
                            context.startAbilityByType(
                                "photoEditor",
                                "{\"ability.params.stream\":[\"${uri}\"], \"ability.want.params.uriPermissionFlag\":${flag}}",
                                callBack
                            )
                        } catch (e: BusinessException) {
                            AppLog.error("startAbilityByType fail, err: ${e.message}")
                        }
                    }
                ).margin(20)

                Image(this.originalImage).width(100.percent).height(200).margin(10).objectFit(ImageFit.Contain)

                Image(
                    this.editedImage.getOrDefault(
                    {=> createPixelMap(InitializationOptions(size: Size(height: 1, width: 1)))})).width(100.percent).
                    height(200).margin(10).objectFit(ImageFit.Contain)
            }.width(100.percent)
        }.height(100.percent)
    }
}
