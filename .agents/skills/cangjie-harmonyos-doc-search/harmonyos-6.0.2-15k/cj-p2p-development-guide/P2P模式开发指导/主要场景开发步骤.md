## 主要场景开发步骤

### 创建/删除P2P群组

1. import需要的WI-FI模块。
2. 开启设备的WI-FI。
3. 需要SystemCapability.Communication.WiFi.P2P系统能力。
4. 创建/删除P2P群组。

示例代码：

```cangjie
import kit.ConnectivityKit.*
import ohos.base.*

// 创建群组，将当前设备当做GO使用时，需要该步骤
// netId：-1表示创建临时组，下次和已连接过的设备连接，需要重新进行GO协商，以及wps秘钥协商;
//        -2表示创建永久组，下次和已连接过的设备连接，不需要重新进行GO和wps秘钥协商;

let recvP2pPersistentGroupChangeFunc = WifiCallback0 {
    =>
    AppLog.info("p2p persistent group change receive event")
    // 永久组创建好后需要处理的业务
}

// 创建永久组，需要注册永久组状态改变事件回调
onP2pPersistentGroupChange(recvP2pPersistentGroupChangeFunc)
try {
    let config = WifiP2PConfig (
        "00:11:22:33:44:55",
        -2,
        "12345678",
        "testGroup",
        GroupOwnerBand.GO_BAND_AUTO,
        deviceAddressType: DeviceAddressType.REAL_DEVICE_ADDRESS
    )
    createGroup(config)
} catch(e: BusinessException) {
    AppLog.error("failed: ${e}")
}

// 删除群组
try {
    removeGroup()
} catch(e: BusinessException) {
    AppLog.error("failed: ${e}")
}
```

错误码请参见[WIFI错误码](../../../API_Reference/source_zh_cn/errorcodes/cj-errorcode-wifi-manager.md)。

### 建立P2P连接

1. import需要的WI-FI模块。
2. 开启设备的WI-FI。
3. 需要SystemCapability.Communication.WiFi.P2P系统能力。
4. 注册"p2pPeerDeviceChange"事件回调，并在回调实现中执行P2P连接。
5. 开始P2P设备发现。

示例代码：

```cangjie
import kit.ConnectivityKit.*
import ohos.base.*
import std.{sync.*, time.*}

let recvP2pConnectionChangeFunc = WifiCallback1<WifiP2pLinkedInfo> {
    info: WifiP2pLinkedInfo =>
    AppLog.info("p2p connection change receive event: ${info}")
    let linkInfo = try {
        getP2pLinkedInfo()
    } catch (e: BusinessException) {
        AppLog.error("failed to get getP2pLinkedInfo: ${e}")
    }
    AppLog.info("get getP2pLinkedInfo: ${linkInfo}")
    // 添加P2P连接成功或者失败场景的业务处理
}

// P2P连接完成，会调用"p2pConnectionChange"事件回调
onP2pConnectionChange(recvP2pConnectionChangeFunc)

let recvP2pPeerDeviceChangeFunc = WifiCallback1<Array<WifiP2pDevice>>({ result =>
    AppLog.info("p2p peer device change receive event: ${result}")
    let devices: Array<WifiP2pDevice> = try {
        getP2pPeerDevices()
    } catch(e: BusinessException) {
        AppLog.error("failed to get getP2pPeerDevices: ${e}")
        []
    }
    AppLog.info("get peer devices: ${devices}")
    for(device in devices) {
        if (device.deviceName == "my_test_device") {
            AppLog.info("p2p connect to test device: ${device.deviceAddress}")
            let config = WifiP2PConfig(device.deviceAddress, -2, "", "", GroupOwnerBand.GO_BAND_AUTO)
            p2pConnect(config)
        }
    }
})

// P2P扫描结果上报时会调用"p2pPeerDeviceChange"事件回调
onP2pPeerDeviceChange(recvP2pPeerDeviceChangeFunc)

Timer.once(Duration.second * 125, { => offP2pConnectionChange(callback: recvP2pConnectionChangeFunc) })
Timer.once(Duration.second * 125, { => offP2pPeerDeviceChange(callback: recvP2pPeerDeviceChangeFunc) })

// 开始发现P2P设备，即，开始P2P扫描
AppLog.info("start discover devices")
startDiscoverDevices()
```

错误码请参见[WIFI错误码](../../../API_Reference/source_zh_cn/errorcodes/cj-errorcode-wifi-manager.md)。
