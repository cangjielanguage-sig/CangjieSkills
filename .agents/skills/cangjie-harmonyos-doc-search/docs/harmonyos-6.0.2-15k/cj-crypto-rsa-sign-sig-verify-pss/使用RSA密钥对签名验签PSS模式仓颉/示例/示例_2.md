func verifyMessagePSS() {
    // 完整的明文被拆分为input1和input2。
    let plan1 = "This is Sign test plan1"
    let plan2 = "This is Sign test plan2"
    let input1: DataBlob = DataBlob(plan1.toArray())
    let input2: DataBlob = DataBlob(plan2.toArray())
    // 获得RSA密钥对密钥参数对象。
    let rsaKeyPairSpec = genRsa2048KeyPairSpec()
    // 构造RSA密钥对生成器。
    let rsaGeneratorSpec = createAsyKeyGeneratorBySpec(rsaKeyPairSpec)
    // sign和verfiy均支持RSA密钥带长度/不带长度的写法。
    let signer = createSign("RSA|PSS|SHA256|MGF1_SHA256")
    let verifyer = createVerify("RSA2048|PSS|SHA256|MGF1_SHA256")
    let keyPair = rsaGeneratorSpec.generateKeyPair()
    signer.initialize(keyPair.priKey)
    // 在签名初始化后，对PSS参数进行set和get操作。
    let setN: Int32 = 32
    signer.setSignSpec(SignSpecItem.PSS_SALT_LEN_NUM, setN)
    var saltLen = signer.getSignSpec(SignSpecItem.PSS_SALT_LEN_NUM)
    AppLog.info("SaltLen == " + saltLen.toString())
    var tf = signer.getSignSpec(SignSpecItem.PSS_TRAILER_FIELD_NUM)
    AppLog.info("trailer field == " + tf.toString())
    var md = signer.getSignSpec(SignSpecItem.PSS_MD_NAME_STR)
    AppLog.info("md == " + md.toString())
    var mgf = signer.getSignSpec(SignSpecItem.PSS_MGF_NAME_STR)
    AppLog.info("mgf == " + mgf.toString())
    var mgf1Md = signer.getSignSpec(SignSpecItem.PSS_MGF1_MD_STR)
    AppLog.info("mgf1Md == " + mgf1Md.toString())
    signer.update(input1)
    let signMessageBlob = signer.sign(input2)
    // 在验签初始化前，对PSS参数进行set和get操作。
    verifyer.setVerifySpec(SignSpecItem.PSS_SALT_LEN_NUM, setN)
    saltLen = verifyer.getVerifySpec(SignSpecItem.PSS_SALT_LEN_NUM)
    AppLog.info("SaltLen == " + saltLen.toString())
    tf = verifyer.getVerifySpec(SignSpecItem.PSS_TRAILER_FIELD_NUM)
    AppLog.info("trailer field == " + tf.toString())
    md = verifyer.getVerifySpec(SignSpecItem.PSS_MD_NAME_STR)
    AppLog.info("md == " + md.toString())
    mgf = verifyer.getVerifySpec(SignSpecItem.PSS_MGF_NAME_STR)
    AppLog.info("mgf == " + mgf.toString())
    mgf1Md = verifyer.getVerifySpec(SignSpecItem.PSS_MGF1_MD_STR)
    verifyer.initialize(keyPair.pubKey)
    verifyer.update(input1)
    let verifyResult = verifyer.verify(input2, signMessageBlob)
    if (verifyResult == true) {
        AppLog.info('verify success')
    } else {
        AppLog.error('verify failed')
    }
}
