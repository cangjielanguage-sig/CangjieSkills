## 完整示例

参考以下示例，完整地播放一首音乐，实现起播后3s暂停，暂停3s重新播放的效果。

```cangjie
import kit.MediaKit.*
import kit.CoreFileKit.*
import kit.AudioKit.*
import ohos.base.*

//初始化context，可以通过MainAbility获取AbilityContext
var ctx = None<UIAbilityContext>

class AVPlayerDemo {
    private var count: Int64 = 0
    private var isSeek: Bool = true // 用于区分模式是否支持seek操作
    private var fileSize: Int64 = -1
    private var fd: Int32 = 0
    // 注册avplayer回调函数
    func setAVPlayerCallback(avPlayer: AVPlayer) {
        // seek操作结果回调函数
        avPlayer.on(AVPlayerCallbackType.seekDone, SeekDoneCallBack())
        // error回调监听函数,当avPlayer在操作过程中出现错误时调用 reset接口触发重置流程
        avPlayer.on(AVPlayerCallbackType.error, ErrCallBack(avPlayer))
        // 状态机变化回调函数
        avPlayer.on(AVPlayerCallbackType.stateChange, StateChangeCallback(avPlayer, count, isSeek))
    }

    // 以下demo为使用fs文件系统打开沙箱地址获取媒体文件地址并通过url属性进行播放示例
    func avPlayerUrlDemo() {
        // 创建avPlayer实例对象
        var avPlayer: AVPlayer = createAVPlayer()
        // 创建状态机变化回调函数
        this.setAVPlayerCallback(avPlayer)
        var fdPath = "fd://"

        var pathDir = ctx.getOrThrow().filesDirectory
        var path = pathDir + "/01.mp3"
        // 打开相应的资源文件地址获取fd，并为url赋值触发initialized状态机上报
        var file = FileFs.open(path)
        fdPath = fdPath + "" + file.fd.toString()
        this.isSeek = true // 支持seek操作
        avPlayer.url = fdPath
    }

    // 以下demo为使用资源管理接口获取打包在HAP内的媒体资源文件并通过fdSrc属性进行播放示例
    func avPlayerFdSrcDemo() {
        // 创建avPlayer实例对象
        var avPlayer: AVPlayer = createAVPlayer()
        // 创建状态机变化回调函数
        this.setAVPlayerCallback(avPlayer)
        // 通过AbilityContext的resourceManager成员的getRawFd接口获取媒体资源播放地址
        // 返回类型为AVFileDescriptor,fd为HAP包fd地址，offset为媒体资源偏移量，length为播放长度
        var fileDescriptor = ctx.getOrThrow().resourceManager.getRawFd("01.mp3")
        var avFileDescriptor: AVFileDescriptor = AVFileDescriptor(fileDescriptor.fd, fileDescriptor.offset,
            fileDescriptor.length)
        this.isSeek = true // 支持seek操作

        // 为fdSrc赋值触发initialized状态机上报
        avPlayer.fdSrc = avFileDescriptor
    }

    // 以下demo为使用fs文件系统打开沙箱地址获取媒体文件地址并通过dataSrc属性进行播放(seek模式)示例
    func avPlayerDataSrcSeekDemo() {
        // 创建avPlayer实例对象
        var avPlayer: AVPlayer = createAVPlayer()
        // 创建状态机变化回调函数
        this.setAVPlayerCallback(avPlayer)
        // dataSrc播放模式的的播放源地址，当播放为Seek模式时fileSize为播放文件的具体大小，下面会对fileSize赋值
        var src: AVDataSrcDescriptor = AVDataSrcDescriptor(
            -1,
            avPlayerDataSrcSeekDemoCallback(this.fd, this.fileSize)
        )
        // 通过AbilityContext获取沙箱地址filesDir
        var pathDir = ctx.getOrThrow().filesDirectory
        var path = pathDir + "/01.mp3"
        var file = FileFs.open(path)
        this.fd = file.fd
        // 获取播放文件的大小
        this.fileSize = FileFs.stat(path).size
        src.fileSize = this.fileSize
        this.isSeek = true // 支持seek操作
        avPlayer.dataSrc = src
    }
