## 示例

应用侧代码：

```cangjie
// xxx.cj
internal import ohos.component.Button
import ohos.state_macro_manage.rawfile
import kit.ArkWeb.*
import kit.UIKit.{Web, BusinessException, PromptAction}
import kit.LocalizationKit.{__GenerateResource__}

@Entry
@Component
class EntryView {
    let webController = WebviewController()

    public func aboutToAppear(): Unit {
        try {
            WebviewController.setWebDebuggingAccess(true);
        } catch (e: BusinessException) {
            AppLog.error("ErrorCode: ${e.code},  Message: ${e.message}");
        }
    }

    func build() {
        Column {
            Web(src: @rawfile("index.html"), controller: webController).onControllerAttached(
                {
                => AppLog.info("controller attached")
            }).onLoadIntercept(
                {
                    event =>
                    AppLog.info("onLoadIntercept url: ${event.getRequestUrl()}")
                    AppLog.info("isMainFrame: ${event.isMainFrame()}")
                    AppLog.info("isRedirect: ${event.isRedirect()}")
                    AppLog.info("isRequestGesture: ${event.isRequestGesture()}")
                    // 返回true表示阻止此次加载，否则允许此次加载
                    return false
                }
            ).onPageBegin({
                event => AppLog.info("onPageBegin url: ${event.url}")
            }).onPageEnd({
                event => AppLog.info("onPageBegin url: ${event.url}")
            }).onDisAppear({
                => PromptAction.showToast(
                    message: "The web is hidden",
                    duration: 2000
                )
            })
        }
    }
}
```
