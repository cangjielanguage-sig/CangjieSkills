@Entry
@Component
class EntryView {
    let webController = WebviewController()
    let mainUrl: String = 'https://www.example.com'
    let textIputController: TextInputController = TextInputController()
    @State
    var inputText: String = 'https://www.example.com'

    func build() {
        Column {
            Row {
                Column {
                    TextInput(text: inputText, placeholder: mainUrl, controller: textIputController).id("input_url").
                        height(40).margin(5).borderColor(Color.BLUE).onChange {
                        value => this.inputText = value
                    }.margin(10)
                    Button(ButtonOptions(shape: ButtonType.Capsule)) {
                        Text("Go")
                    }.onClick {
                        evt => webController.loadUrl(this.inputText);
                    }.margin(10)

                    Button(ButtonOptions(shape: ButtonType.Capsule)) {
                        Text("addAdsBlockAllowedList")
                    }.onClick {
                        evt =>
                        // addAdsBlockAllowedList和addAdsBlockAllowedList配套使用，设置网页级的广告过滤开关
                        AdsBlockManager.addAdsBlockDisallowedList(["example.com"])

                        let arrAllowedDomainSuffixes: Array<String> = ["news.example.com", "sport.example.com"]
                        AdsBlockManager.addAdsBlockAllowedList(arrAllowedDomainSuffixes)
                    }.margin(10)
                }
                Web(src: mainUrl, controller: webController).onControllerAttached {
                    => try {
                        webController.enableAdsBlock(true)
                    } catch (e: BusinessException) {
                        AppLog.error("ErrorCode: ${e.code},  Message: ${e.message}")
                    }
                }
            }
        }
    }
}
```

需要注意的是，AdsBlockManager的DisallowedList和AllowedList列表不会持久化，因此重启应用后会重置为空。

如果Web组件未通过[enableAdsBlock()](../../API_Reference/source_zh_cn/apis/ArkWeb/cj-apis-webview.md#func-enableadsblockbool)接口开启广告过滤功能，上述接口设置在此Web组件中将不起作用。
