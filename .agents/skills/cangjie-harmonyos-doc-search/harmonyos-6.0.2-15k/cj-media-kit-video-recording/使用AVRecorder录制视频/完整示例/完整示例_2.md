func stopRecordingProcess() {
        if (this.avRecorder.isSome()) {
            var realAvRecorder = this.avRecorder.getOrThrow()
            // 1. 停止录制
            if (realAvRecorder.state.toString() == "started" || realAvRecorder.state.toString() == "paused") { // 仅在started或者paused状态下调用stop为合理状态切换
                realAvRecorder.stop()
                this.stopCameraOutput()
            }
            // 2.重置
            realAvRecorder.reset()
            // 3.释放录制实例
            realAvRecorder.release()
            // 4.文件录制完成后，关闭fd,实现略
            FileFs.close(this.fileFd)
            // 5.释放相机相关实例
            this.releaseCamera()
        }
    }

    // 安全控件保存媒体资源至图库
    func saveRecorderAsset() {
        var phAccessHelper = getPhotoAccessHelper(ctx.getOrThrow())
        // 需要确保uriPath对应的资源存在
        this.uriPath = FileUri.getUriFromPath(this.filePath) // 获取录制文件的uri，用于安全控件保存至图库
        var assetChangeRequest: MediaAssetChangeRequest = MediaAssetChangeRequest.createVideoAssetRequest(
            ctx.getOrThrow(), this.uriPath)
        phAccessHelper.applyChanges(assetChangeRequest)
    }

    // 一个完整的【开始录制-暂停录制-恢复录制-停止录制】示例
    func videoRecorderDemo() {
        this.startRecordingProcess() // 开始录制

        // 用户此处可以自行设置录制时长，例如通过设置休眠阻止代码执行
        this.pauseRecordingProcess() //暂停录制
        this.resumeRecordingProcess() // 恢复录制
        this.stopRecordingProcess() // 停止录制

        // 安全控件保存媒体资源至图库
        this.saveRecorderAsset()
    }
