## 图片编辑类应用实现图片编辑页面

1. 在DevEco Studio工程中手动新建一个PhotoEditorExtensionAbility。

    在工程Module对应的cangjie目录下，右键选择“New > File”，新建一个.cj文件，如examplePhotoEditorAbility.cj。

2. 在ExamplePhotoEditorAbility.ets中重写onCreate、onForeground、onBackground、onDestroy和onStartContentEditing的生命周期回调。

    其中，需要在onStartContentEditing中加载入口页面文件index.cj，并将session、uri、实例对象等保存在LocalStorage中传递给页面。

    ```cangjie
    import ohos.state_manage.LocalStorage
    import kit.AbilityKit.{PhotoEditorExtensionAbility, PhotoEditorExtensionContext, UIExtensionContentSession, Want}
    import kit.UIKit.{AppLog}
    import ohos.component.sendKeyEvent

    let PHOTO_EDITOR_ABILITY_REGISTER_RESULT = PhotoEditorExtensionAbility.registerCreator("ExamplePhotoEditorAbility",
        {=> ExamplePhotoEditorAbility()})
    let localStorage = LocalStorage()
    var globalContext: ?PhotoEditorExtensionContext = None

    class ExamplePhotoEditorAbility <: PhotoEditorExtensionAbility {
        public override func onCreate(): Unit {
            AppLog.info("ExamplePhotoEditorAbility OnCreated.")
        }

        // 获取图片，加载页面并将需要的参数传递给页面
        public override func onStartContentEditing(uri: String, want: Want, session: UIExtensionContentSession): Unit {
            globalContext = this.context
            AppLog.info("onStartContentEditing uri: ${uri}")
            localStorage.setOrCreate<UIExtensionContentSession>("session", session)
            localStorage.setOrCreate<String>("uri", uri)
            session.loadContent("PhotoEditorView")
        }

        public override func onForeground(): Unit {
            AppLog.info("ExamplePhotoEditorAbility onForeground.")
        }

        public override func onBackground(): Unit {
            AppLog.info("ExamplePhotoEditorAbility onBackground.")
        }

        public override func onDestroy(): Unit {
            AppLog.info("ExamplePhotoEditorAbility onDestroy.")
        }
    }
    ```

                )
            } catch (e: BusinessException) {
                AppLog.error("Open file failed: ${e.message}")
                return None
            }

                originalImage = pixmap
                return pixmap
            } catch (e: BusinessException) {
                AppLog.error("ReadImage failed: ${e.message}")
            } finally {
                FileFs.close(file)
            }
            return None
        }
