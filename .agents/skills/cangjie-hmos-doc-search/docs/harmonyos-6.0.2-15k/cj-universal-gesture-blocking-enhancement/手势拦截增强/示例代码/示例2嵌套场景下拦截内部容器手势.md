### 示例2（嵌套场景下拦截内部容器手势）

本示例通过将参数exposeInnerGesture设置为true，实现了一级Tabs容器在嵌套二级Tabs的场景下，能够屏蔽二级Tabs内置Swiper的滑动手势，从而触发一级Tabs内置Swiper滑动手势的功能。

开发者自行定义变量来记录内层Tabs的索引值，通过该索引值判断当滑动达到内层Tabs的边界处时，触发回调返回屏蔽使外层Tabs产生滑动手势。

<!-- run -->

```cangjie
package ohos_app_cangjie_entry

import ohos.base.*
import ohos.component.*
import ohos.state_manage.*
import ohos.state_macro_manage.*

@Entry
@Component
class EntryView {
    @State
    var currentIndex: Int32 = 0
    @State
    var selectedIndex: Int32 = 0
    @State
    var fontColor: UInt32 = 0x182431
    @State
    var selectedFontColor: UInt32 = 0x007DFF
    var innerSelectedIndex: Int32 = 0 // 记录内层Tabs索引
    var controller: TabsController = TabsController()

    @Builder
    func tabBuilder(index: Int32, name: String) {
        Column {
            if (this.selectedIndex == index) {
                Text(name).fontColor(this.selectedFontColor).fontSize(16).fontWeight(FontWeight.W500).lineHeight(22).
                    margin(top: 17, bottom: 7)
                Divider().strokeWidth(2).color(0x007DFF).opacity(1)
            } else {
                Text(name).fontColor(this.fontColor).fontSize(16).fontWeight(FontWeight.W400).lineHeight(22).margin(
                    top: 17, bottom: 7)
                Divider().strokeWidth(2).color(0x007DFF).opacity(0)
            }
        }.width(100.percent)
    }

    func build() {
        Column {
            Tabs(BarPosition.Start, this.controller, this.currentIndex) {
                TabContent() {
                    Column().width(100.percent).height(100.percent).backgroundColor(Color.GREEN)
                }.tabBar() {
                    bind(this.tabBuilder, this)(0, "green")
                }
                TabContent() {
                    Tabs() {
                        TabContent() {
                            Column().width(100.percent).height(100.percent).backgroundColor(Color.BLUE)
                        }.tabBar("blue")

                        TabContent() {
                            Column().width(100.percent).height(100.percent).backgroundColor(Color.PINK)
                        }.tabBar("pink")
                    }.onAnimationStart(
                        {
                        index: Int32, targetIndex: Int32, event: TabsAnimationEvent => this.innerSelectedIndex = targetIndex
                    }).onGestureRecognizerJudgeBegin(
                        { // 在识别器即将要成功时，根据当前组件状态，设置识别器使能状态
                            event: BaseGestureEvent, current: GestureRecognizer, others: Array<GestureRecognizer> =>
                            var target = current.getEventTargetInfo()
                            var isPanGesture = match (current.getType()) {
                                case GestureTypes.PAN_GESTURE => true
                                case _ => false
                            }
                            if (isPanGesture && current.isBuiltIn()) {
                                AppLog.info("onGestureRecognizerJudgeBegin child")
                                if (event is PanGestureEvent) {
                                    AppLog.info("onGestureRecognizerJudgeBegin child PAN_GESTURE")
                                    var panEvent = (event as PanGestureEvent).getOrThrow()
                                    if (panEvent.velocityX < 0.0 && this.innerSelectedIndex == 1) { // 内层Tabs滑动到尽头
                                        AppLog.info("onGestureRecognizerJudgeBegin child reject end")
                                        return GestureJudgeResult.REJECT
                                    }
                                    if (panEvent.velocityX > 0.0 && this.innerSelectedIndex == 0) { // 内层Tabs滑动到开头
                                        AppLog.info("onGestureRecognizer child reject begin")
                                        return GestureJudgeResult.REJECT
                                    }
                                }
                            }
                            return GestureJudgeResult.CONTINUE
                        },
                        true
                    )
                }.tabBar() {
                    bind(this.tabBuilder, this)(1, "blue and pink")
                }
                TabContent() {
                    Column().width(100.percent).height(100.percent).backgroundColor(Color.BROWN)
                }.tabBar() {
                    bind(this.tabBuilder, this)(2, "brown")
                }
            }.onAnimationStart(
                {
                index: Int32, targetIndex: Int32, event: TabsAnimationEvent => this.selectedIndex = targetIndex
            }) // 切换动画开始时触发该回调。目标页签显示下划线。
        }
    }
}
```
