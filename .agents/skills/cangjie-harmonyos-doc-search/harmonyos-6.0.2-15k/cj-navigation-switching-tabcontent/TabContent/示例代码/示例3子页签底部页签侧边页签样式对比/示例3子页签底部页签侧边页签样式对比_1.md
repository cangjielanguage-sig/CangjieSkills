### 示例3（子页签/底部页签/侧边页签样式对比）

本示例使用了SubTabBarStyle、BottomTabBarStyle实现了子页签、底部页签和侧边页签。

<!-- run -->

```cangjie
package ohos_app_cangjie_entry

import kit.UIKit.*
import std.collection.*
import kit.LocalizationKit.*
import ohos.state_macro_manage.*

@Entry
@Component
class EntryView {
    func build() {
        Column(5.vp) {
            Text("子页签样式")
            Column() {
                Tabs(BarPosition.Start) {
                    TabContent() {
                        Column().width(100.percent).height(100.percent).backgroundColor(0xE67C92)
                    }.tabBar(SubTabBarStyle("Pink")).onWillShow({
                        => AppLog.info("Pink will show")
                    }).onWillHide({
                        => AppLog.info("Pink will hide")
                    })

                    TabContent() {
                        Column().width(100.percent).height(100.percent).backgroundColor(0xFFBF00)
                    }.tabBar(SubTabBarStyle("Yellow")).onWillShow({
                        => AppLog.info("Yellow will show")
                    }).onWillHide({
                        => AppLog.info("Yellow will hide")
                    })

                    TabContent() {
                        Column().width(100.percent).height(100.percent).backgroundColor(0x007DFF)
                    }.tabBar(SubTabBarStyle("Blue")).onWillShow({
                        => AppLog.info("Blue will show")
                    }).onWillHide({
                        => AppLog.info("Blue will hide")
                    })

                    TabContent() {
                        Column().width(100.percent).height(100.percent).backgroundColor(0x00CB87)
                    }.tabBar(SubTabBarStyle("Green")).onWillShow({
                        => AppLog.info("Green will show")
                    }).onWillHide({
                        => AppLog.info("Green will hide")
                    })
                }.vertical(false).scrollable(true).barMode(BarMode.Fixed).onChange(
                    {
                    index: Int32 => AppLog.info("${index}")
                }).width(100.percent).backgroundColor(0xF1F3F5)
            }.width(100.percent).height(200)
            Text("底部页签样式")
            Column() {
                Tabs(BarPosition.End) {
                    TabContent() {
                        Column().width(100.percent).height(100.percent).backgroundColor(0xE67C92)
                    }.tabBar(BottomTabBarStyle(@r(app.media.startIcon), "Pink")).onWillShow(
                        {
                        => AppLog.info("Pink will show")
                    }).onWillHide({
                        => AppLog.info("Pink will hide")
                    })

                    TabContent() {
                        Column().width(100.percent).height(100.percent).backgroundColor(0xFFBF00)
                    }.tabBar(BottomTabBarStyle(@r(app.media.startIcon), "Yellow")).onWillShow(
                        {
                        => AppLog.info("Yellow will show")
                    }).onWillHide({
                        => AppLog.info("Yellow will hide")
                    })

                    TabContent() {
                        Column().width(100.percent).height(100.percent).backgroundColor(0x007DFF)
                    }.tabBar(BottomTabBarStyle(@r(app.media.startIcon), "Blue")).onWillShow(
                        {
                        => AppLog.info("Blue will show")
                    }).onWillHide({
                        => AppLog.info("Blue will hide")
                    })

                    TabContent() {
                        Column().width(100.percent).height(100.percent).backgroundColor(0x00CB87)
                    }.tabBar(BottomTabBarStyle(@r(app.media.startIcon), "Green")).onWillShow(
                        {
                        => AppLog.info("Green will show")
                    }).onWillHide({
                        => AppLog.info("Green will hide")
                    })
                }.vertical(false).scrollable(true).barMode(BarMode.Fixed).onChange(
                    {
                    index: Int32 => AppLog.info("${index}")
                }).width(100.percent).backgroundColor(0xF1F3F5)
            }.width(100.percent).height(200)
            Text("侧边页签样式")
            Column() {
                Tabs(BarPosition.Start) {
                    TabContent() {
                        Column().width(100.percent).height(100.percent).backgroundColor(0xE67C92)
                    }.tabBar(BottomTabBarStyle(@r(app.media.startIcon), "Pink")).onWillShow(
                        {
                        => AppLog.info("Pink will show")
                    }).onWillHide({
                        => AppLog.info("Pink will hide")
                    })

                    TabContent() {
                        Column().width(100.percent).height(100.percent).backgroundColor(0xFFBF00)
                    }.tabBar(BottomTabBarStyle(@r(app.media.startIcon), "Yellow")).onWillShow(
                        {
                        => AppLog.info("Yellow will show")
                    }).onWillHide({
                        => AppLog.info("Yellow will hide")
                    })

                    TabContent() {
                        Column().width(100.percent).height(100.percent).backgroundColor(0x007DFF)
                    }.tabBar(BottomTabBarStyle(@r(app.media.startIcon), "Blue")).onWillShow(
                        {
                        => AppLog.info("Blue will show")
                    }).onWillHide({
                        => AppLog.info("Blue will hide")
                    })
