### 示例代码

#### @StorageProp

<!-- run -->

```cangjie
package ohos_app_cangjie_entry

import kit.UIKit.*
import ohos.state_macro_manage.*

let a = AppStorage.setOrCreate<Int64>("PropA", 47)

@Entry
@Component
class EntryView {
    @StorageProp["PropA"]
    let storageProp1: Int64 = 1
    func build() {
        Column(15) {
            Button("Parent from AppStorage ${storageProp1}").onClick {
                =>
                // 只能通过操作AppStorage来改变PropA的值
                AppStorage.set<Int64>("PropA", 48)
            }
        }
    }
}
```

#### @StorageLink

<!-- run -->

```cangjie
package ohos_app_cangjie_entry

import kit.UIKit.*
import ohos.state_macro_manage.*

let a = AppStorage.setOrCreate("Name", "Cangjie")

@Entry
@Component
class EntryView {
    @StorageLink["Name"]
    var name: String = ""
    func build() {
        Row() {
            Column(10) {
                Text(name)
                // 修改会同步到其他组件中
                Button("Add").onClick {
                    e => name += "1"
                }
                Son()
            }
        }
    }
}

@Component
class Son {
    @StorageLink["Name"]
    var name: String = ""
    func build() {
        Column(10) {
            Text("son: " + name)
            // 修改会同步到其他组件中
            Button("Son Add").onClick {
                e => name += "2"
            }
        }
    }
}
```
