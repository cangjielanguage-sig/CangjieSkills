## 完整示例

Context获取方式请参见：[获取UIAbility的上下文信息](../../application-models/cj-uiability-usage.md#获取uiability的上下文信息)。

```cangjie
import kit.CameraKit.*
import kit.ImageKit.*
import kit.BasicServicesKit.*
import kit.AbilityKit.*
import kit.CoreFileKit.*
import kit.MediaLibraryKit.*
import std.time.*
import ohos.base.*

class CameraStatusCallBack <: Callback1Argument<CameraStatusInfo> {
    public open func invoke(cameraStatusInfo: CameraStatusInfo): Unit {
        AppLog.info("camera : ${cameraStatusInfo.camera.cameraId}")
        AppLog.info("status: ${cameraStatusInfo.status}")
    }
}

class InputErrorCallBack <: Callback1Argument<BusinessException> {
    public open func invoke(error: BusinessException): Unit {
        AppLog.error("Camera input error code: ${error.code}")
    }
}

class PhotoSessionErrorCallBack <: Callback1Argument<BusinessException> {
    public open func invoke(error: BusinessException): Unit {
        AppLog.error("Capture session error code: ${error.code}")
    }
}

class PreviewOutputErrorCallBack <: Callback1Argument<BusinessException> {
    public open func invoke(error: BusinessException): Unit {
        AppLog.error("Preview output error code: ${error.code}")
    }
}

func cameraShootingCase(context: UIAbilityContext, surfaceId: String): Unit {
    // 创建CameraManager对象
    let cameraManager: CameraManager = getCameraManager(context)
    // 监听相机状态变化
    cameraManager.on(CameraCallbackType.CameraStatus, CameraStatusCallBack())

    // 获取相机列表
    let cameraArray: Array<CameraDevice> = cameraManager.getSupportedCameras()
    if (cameraArray.size <= 0) {
        AppLog.error("cameraManager.getSupportedCameras error")
        return
    }

    for (index in 0..cameraArray.size) {
        AppLog.info("cameraId : + ${cameraArray[index].cameraId}") // 获取相机ID
        AppLog.info("cameraPosition : + ${cameraArray[index].cameraPosition}") // 获取相机位置
        AppLog.info("cameraType : + ${cameraArray[index].cameraType}") // 获取相机类型
        AppLog.info("connectionType : + ${cameraArray[index].connectionType}") // 获取相机连接类型
    }

    // 创建相机输入流
    let cameraInput: CameraInput = cameraManager.createCameraInput(cameraArray[0])

    // 监听cameraInput错误信息
    let cameraDevice: CameraDevice = cameraArray[0]
    cameraInput.on(CameraCallbackType.CameraError, cameraDevice, InputErrorCallBack())

    // 打开相机
    cameraInput.open()

    // 获取支持的模式类型
    let sceneModes: Array<SceneMode> = cameraManager.getSupportedSceneModes(cameraArray[0])
    let isSupportPhotoMode: Bool = sceneModes.indexOf(SceneMode.NORMAL_PHOTO).isSome()
    if (!isSupportPhotoMode) {
        AppLog.error('photo mode not support')
        return
    }

    //创建会话
    let photoSession: PhotoSession = (cameraManager.createSession(SceneMode.NORMAL_PHOTO) as PhotoSession).getOrThrow()

    // 监听session错误信息
    photoSession.on(CameraCallbackType.CameraError, PhotoSessionErrorCallBack())

    // 查询Preconfig能力
    let isPreconfigSupport = photoSession.canPreconfig(PreconfigType.PRECONFIG_1080P)
    if (!isPreconfigSupport) {
        AppLog.error('PhotoSession canPreconfig check fail.')
        return
    }

    // 配置Preconfig
    photoSession.preconfig(PreconfigType.PRECONFIG_1080P)

    // 创建预览输出流,其中参数 surfaceId 参考上文 XComponent 组件，预览流为XComponent组件提供的surface
    let previewOutput: PreviewOutput = cameraManager.createPreviewOutput(surfaceId)
    // 监听预览输出错误信息
    previewOutput.on(CameraCallbackType.CameraError, PreviewOutputErrorCallBack())

    // 创建拍照输出流
    let photoOutput: PhotoOutput = cameraManager.createPhotoOutput()

    // 开始配置会话
    photoSession.beginConfig()

    // 向会话中添加相机输入流
    photoSession.addInput(cameraInput)

    // 向会话中添加预览输出流
    photoSession.addOutput(previewOutput)
