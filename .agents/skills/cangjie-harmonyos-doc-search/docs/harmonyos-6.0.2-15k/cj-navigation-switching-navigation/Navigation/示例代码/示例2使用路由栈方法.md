### 示例2（使用路由栈方法）

该示例主要演示NavPathStack中方法的使用。

<!-- run -->

```cangjie
package ohos_app_cangjie_entry

import kit.UIKit.*
import ohos.state_macro_manage.*

@Builder
func pageMap(name: String) {
    if (name == "pageOne") {
        PageOne()
    } else {
        PageTwo()
    }
}

@Entry
@Component
class EntryView {
    var pageInfos: NavPathStack = NavPathStack()
    public func build() {
        Navigation(pageInfos) {
            Column() {
                Button("pushPath", ButtonOptions(shape: ButtonType.Capsule)) //将name指定的NavDestination页面信息入栈
                    .width(80.percent).height(40).onClick {
                    pageInfos.pushPath(NavPathInfo("pageOne", "pageOne test"))
                }
            }
        }.title("Navigation").navDestination(bind<String>(pageMap, this))
    }
}

@Component
class PageOne {
    var pageInfos: NavPathStack = NavPathStack()

    public func build() {
        NavDestination() {
            Column() {
                Text("Page One").margin(20)

                Button("pushPathByName", ButtonOptions(shape: ButtonType.Capsule)).width(80.percent).height(40).margin(
                    20).onClick {
                    => this.pageInfos.pushPathByName("pageTwo", "pageTwo test")
                }

                Button("singletonLaunchMode", ButtonOptions(shape: ButtonType.Capsule)).width(80.percent).height(40).
                    margin(20).onClick {
                    => this.pageInfos.pushPath(NavPathInfo("PageOne", ""),
                        options: NavigationOptions(launchMode: LaunchMode.MoveToTopSingleTon))
                }

                Button("popToName", ButtonOptions(shape: ButtonType.Capsule)).width(80.percent).height(40).margin(20).
                    onClick {
                    =>
                    this.pageInfos.popToName("pageTwo")
                    AppLog.info("popToName: pageTwo")
                }

                Button("popToIndex", ButtonOptions(shape: ButtonType.Capsule)).width(80.percent).height(40).margin(20).
                    onClick {
                    =>
                    this.pageInfos.popToIndex(1)
                    AppLog.info("popToIndex: 1")
                }

                Button("moveToTop", ButtonOptions(shape: ButtonType.Capsule)).width(80.percent).height(40).margin(20).
                    onClick {
                    =>
                    this.pageInfos.moveToTop("pageTwo")
                    AppLog.info("moveToTop: pageTwo")
                }

                Button("moveIndexToTop", ButtonOptions(shape: ButtonType.Capsule)).width(80.percent).height(40).margin(
                    20).onClick {
                    =>
                    this.pageInfos.moveIndexToTop(1)
                    AppLog.info("moveIndexToTop: 1")
                }

                Button("get", ButtonOptions(shape: ButtonType.Capsule)).width(80.percent).height(40).margin(20).onClick {
                    =>
                    AppLog.info("获取栈中所有NavDestination页面的名称: ${this.pageInfos.getAllPathName().toString()}")
                    AppLog.info(
                        "获取index指定的NavDestination页面的参数信息: ${this.pageInfos.getParamByIndex(1).toString()}")
                    AppLog.info(
                        "获取全部名为name的NavDestination页面的参数信息: ${this.pageInfos.getParamByName("pageTwo").toString()}"
                    )
                    AppLog.info(
                        "获取全部名为name的NavDestination页面的位置索引: ${this.pageInfos.getIndexByName("pageONE").toString()}"
                    )
                    AppLog.info("获取栈大小: ${this.pageInfos.size().toString()}")
                }

                Button("clear").onClick {
                    => this.pageInfos.clear()
                }
            }
        }.onReady {
            context => pageInfos = context.pathStack
        }.onBackPressed {
            pageInfos.pop()
            true
        }
    }
}

@Component
class PageTwo {
    var pageInfos: NavPathStack = NavPathStack()

    public func build() {
        NavDestination() {
            Column() {
                Text("Page Two").width(100.percent).height(30).margin(60)

                Button("pushPathByName", ButtonOptions(shape: ButtonType.Capsule)).width(80.percent).height(40).margin(
                    20).onClick {
                    => this.pageInfos.pushPathByName("pageOne", "")
                }
            }
        }.onReady {
            context => pageInfos = context.pathStack
        }.onBackPressed {
            pageInfos.pop()
            true
        }
    }
}
```
