package ohos_app_cangjie_entry

import kit.UIKit.*
import ohos.state_macro_manage.*

@Entry
@Component
class EntryView {
    func build() {
        Column() {
            Text("margin and padding:").fontSize(12).fontColor(0xCCCCCC).width(90.percent)
            Row() {
                // 宽度80 ,高度80 ,外边距20（蓝色区域），内边距10（白色区域）
                Row() {
                    Row().size(width: 100.percent, height: 100.percent).backgroundColor(Color.GREEN)
                }.width(80).height(80).padding(10).margin(20).backgroundColor(Color.WHITE)
            }.backgroundColor(Color.BLUE)

            Text("constraintSize").fontSize(12).fontColor(0xCCCCCC).width(90.percent)
            Text(
                "this is a Text.this is a Text.this is a Text.this is a Text.this is a Text.this is a Text.this is a Text.this is a Text.this is a Text.this is a Text.this is a Text.this is a Text.this is a Text.this is a Text.this is a Text"
            ).width(90.percent).constraintSize(maxWidth: 200.vp)

            Text("layoutWeight").fontSize(12).fontColor(0xCCCCCC).width(90.percent)
            // 父容器尺寸确定时，设置了layoutWeight的子元素在主轴布局尺寸按照权重进行分配，忽略本身尺寸设置。
            Row() {
                // 权重1，占主轴剩余空间1/3
                Text("layoutWeight(1)").size(width: 30.percent, height: 110.vp).backgroundColor(0xFFEFD5).textAlign(
                    TextAlign.Center).layoutWeight(1)
                // 权重2，占主轴剩余空间2/3
                Text("layoutWeight(2)").size(width: 30.percent, height: 110.vp).backgroundColor(0xF5DEB3).textAlign(
                    TextAlign.Center).layoutWeight(2)
                // 未设置layoutWeight属性，组件按照自身尺寸渲染
                Text("no layoutWeight").size(width: 30.percent, height: 110.vp).backgroundColor(0xD2B48C).textAlign(
                    TextAlign.Center)
            }.size(width: 90.percent, height: 140.vp).backgroundColor(0xAFEEEE)
            Text("calc:").fontSize(12).fontColor(0xCCCCCC).width(90.percent)
            Text("calc test").fontSize(50).fontWeight(FontWeight.Bold).backgroundColor(0xFFFAF0).textAlign(
                TextAlign.Center).margin(50).size(width: 90.percent, height: 10.percent)
        }.width(100.percent).margin(top: 5)
    }
}
