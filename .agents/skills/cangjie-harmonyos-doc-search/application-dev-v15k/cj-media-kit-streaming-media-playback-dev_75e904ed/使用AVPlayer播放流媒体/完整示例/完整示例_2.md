class StateChangeCallback <: OnAVPlayerStateChangeHandle {
    let avPlayer: AVPlayer
    let surfaceID: String
    public init(avPlayer: AVPlayer, surfaceID: String) {
        this.avPlayer = avPlayer
        this.surfaceID = surfaceID
    }
    public open func invoke(state: AVPlayerState, reason: StateChangeReason): Unit {
        match (state) {
            case AVPlayerState.idle => // 成功调用reset接口后触发该状态机上报
                AppLog.info("AVPlayer state idle called.")
                avPlayer.release() // 调用release接口销毁实例对象
            case AVPlayerState.initialized => // avplayer 设置播放源后触发该状态上报
                AppLog.info("AVPlayer state initialized called.")
                avPlayer.surfaceId = this.surfaceID // 设置显示画面，当播放的资源为纯音频时无需设置
                avPlayer.prepare()
            case AVPlayerState.prepared => // prepare调用成功后上报该状态机
                AppLog.info("AVPlayer state prepared called.")
                avPlayer.play() // 调用播放接口开始播放
            case AVPlayerState.playing => // play成功调用后触发该状态机上报
                AppLog.info("AVPlayer state playing called.")
            case AVPlayerState.paused => // pause成功调用后触发该状态机上报
                AppLog.info("AVPlayer state paused called.")
            case AVPlayerState.completed => // 播放结束后触发该状态机上报
                AppLog.info("AVPlayer state completed called.")
                avPlayer.stop() //调用播放结束接口
            case AVPlayerState.stopped => // stop接口成功调用后触发该状态机上报
                AppLog.info("AVPlayer state stopped called.")
                avPlayer.reset() // 调用reset接口初始化avplayer状态
            case AVPlayerState.released => AppLog.info("AVPlayer state released called.")
            case _ => AppLog.info("AVPlayer state unknown called.")
        }
    }
}
