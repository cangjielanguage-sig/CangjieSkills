## 计算设备旋转角度

当前可通过调用[once\<T>(SensorId, Callback1Argument\<T>) where T <: Response](../../../API_Reference/source_zh_cn/apis/SensorServiceKit/cj-apis-sensor.md#func-oncetsensorid-callback1argumentt-where-t--response)获取一次重力传感器在x、y、z三个方向上的数据，计算得出设备旋转角度deviceDegree，示例如下所示。

如果无法获得重力传感器数据，需要申请重力传感器权限ohos.permission.ACCELEROMETER。权限申请请参见[声明权限](../../../Dev_Guide/security/AccessToken/cj-declare-permissions.md#声明权限)，如何获取传感器数据请参见[传感器开发指导](../../device/sensor/cj-sensor-guidelines.md)。

```cangjie
import kit.SensorServiceKit.*
import ohos.base.*
import std.math.*
import std.math.numeric.*

var gravity: Option<Int64> = None

func getRealData(data: GravityResponse): Int64 {
    var getDeviceDegree: Int64 = 0
    AppLog.info("Succeeded in invoki e. X-coordinate component: ${data.x}")
    AppLog.info("Succeeded in invoking once. Y-coordinate component: ${data.y}")
    AppLog.info("Succeeded in invoking once. Z-coordinate component: ${data.z}")
    let x = data.x
    let y = data.y
    let z = data.z
    if ((x * x + y * y) * 3.0 < z * z) {
        return getDeviceDegree
    } else {
        let sd: Float32 = atan2(y, -x)
        let sc = round(Decimal(sd) / Decimal(3.141592653589 * 180.0))
        getDeviceDegree = 90 - sc.toInt64()
        getDeviceDegree = if (getDeviceDegree >= 0) {
            getDeviceDegree % 360
        } else {
            getDeviceDegree % 360 + 360
        }
    }
    return getDeviceDegree
}

func getRealData(data: AccelerometerResponse): Int64 {
    var getDeviceDegree: Int64 = 0
    AppLog.info("Succeeded in invoki e. X-coordinate component: ${data.x}")
    AppLog.info("Succeeded in invoking once. Y-coordinate component: ${data.y}")
    AppLog.info("Succeeded in invoking once. Z-coordinate component: ${data.z}")
    let x = data.x
    let y = data.y
    let z = data.z
    if ((x * x + y * y) * 3.0 < z * z) {
        return getDeviceDegree
    } else {
        let sd: Float32 = atan2(y, -x)
        let sc = round(Decimal(sd) / Decimal(3.141592653589 * 180.0))
        getDeviceDegree = 90 - sc.toInt64()
        getDeviceDegree = if (getDeviceDegree >= 0) {
            getDeviceDegree % 360
        } else {
            getDeviceDegree % 360 + 360
        }
    }
    return getDeviceDegree
}

class GravityCallback <: Callback1Argument<GravityResponse> {
    public open func invoke(data: GravityResponse): Unit {
        gravity = getRealData(data)
    }
}

class AccelerometerCallback <: Callback1Argument<AccelerometerResponse> {
    public open func invoke(data: AccelerometerResponse): Unit {
        gravity = getRealData(data)
    }
}

func getGravity(): Unit {
    var isSupported = false
    let data = getSensorList()
    for (i in data) {
        if (i.sensorId == SensorId.GRAVITY) {
            isSupported = true
            break
        }
    }
    if (isSupported == true) {
        once(SensorId.GRAVITY, GravityCallback())
    } else {
        once(SensorId.ACCELEROMETER, AccelerometerCallback())
    }
}
```
