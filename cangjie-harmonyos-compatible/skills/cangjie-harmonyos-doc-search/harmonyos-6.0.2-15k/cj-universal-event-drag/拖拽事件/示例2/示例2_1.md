import ohos.image.PixelMap

import ohos.image.InitializationOptions

import ohos.image.ScaleMode

@Entry
@Component
class EntryView {
    @State
    var numbers: ArrayList<String> = ArrayList<String>(["one", "two", "three", "four", "five", "six"])
    @State
    var dragItem: String = ""

    var dragStartInfo: String = ""

    @State
    var bananaVisible: Visibility = Visibility.Visible
    @State
    var select: Int32 = 0

    func pixelMapBuilder() {
        Column() {
            Text(this.dragItem).width(50.percent).height(60).fontSize(16).borderRadius(10).textAlign(TextAlign.Center).
                backgroundColor(0xFF0000)
        }
    }
    func build(): Unit {
        Column(30) {
            Text(this.dragStartInfo).width(100.percent).height(30).fontSize(16)
            Text(this.dragListInfo).width(100.percent).height(30).fontSize(16)
            Text("There are three Text elements here").fontSize(12).fontColor(0xCCCCCC).width(90.percent).textAlign(

                        this.dragStartInfo = "apple, " + "onDragStart info:" + evt.extraParams + "," +

                        this.appleVisible = Visibility.Hidden
                        return DragItemInfo(builder: bind(pixelMapBuilder, this), extraInfo: "button1")
                    }
                )

                Button("orange").width(25.percent).height(35).fontSize(16).backgroundColor(0xAFEEEE).visibility(
                    this.orangeVisible).onDragStart(

                        this.dragItem = "orange"
                        this.dragStartInfo = "orange, " + "onDragStart info:" + evt.extraParams + "," +

                        this.orangeVisible = Visibility.Hidden
                        pixelMapBuilder
                    }

                        let opts = InitializationOptions(scaleMode: ScaleMode.CENTER_CROP, size: size)
                        let pixelMap = createPixelMap(colors, opts)
                        return DragItemInfo(pixelMap: pixelMap, extraInfo: "button2")
                    }
                )
            }.border(width: 1.px).width(90.percent).padding(top: 10, bottom: 10).margin(10)
