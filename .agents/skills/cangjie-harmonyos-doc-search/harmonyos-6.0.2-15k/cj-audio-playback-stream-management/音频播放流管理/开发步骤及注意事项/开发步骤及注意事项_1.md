## 开发步骤及注意事项

1. 创建AudioStreamManager实例。
    在使用AudioStreamManager的API前，需要使用getStreamManager()创建一个AudioStreamManager实例。

    ```cangjie
    import kit.AudioKit.*

    let audioManager = getAudioManager()
    let audioStreamManager = audioManager.getStreamManager()
    ```

2. 使用[on(AudioStreamManagerCallbackType, Callback1Argument\<AudioRendererChangeInfoArray>)](../../../API_Reference/source_zh_cn/apis/AudioKit/cj-apis-multimedia-audio.md#func-onaudiostreammanagercallbacktype-callback1argumentaudiocapturerchangeinfoarray)监听音频播放流的变化。如果音频流监听应用需要在音频播放流状态变化、设备变化时获取通知，可以订阅该事件。

    ```cangjie
    import kit.AudioKit.*
    import ohos.base.Callback1Argument

    class AudioRendererChangeCallback <: Callback1Argument<AudioRendererChangeInfoArray> {
        public AudioRendererChangeCallback(let f: (AudioRendererChangeInfoArray) -> Unit) {}
        public func invoke(infoArray: AudioRendererChangeInfoArray) {
            f(infoArray)
        }
    }

    func subscribe() {
        let callback: (AudioRendererChangeInfoArray) -> Unit = {
            infoArray: AudioRendererChangeInfoArray => for (i in 0..infoArray.size) {
                let info = infoArray[i];
                AppLog.info("## RendererChange on is called for ${i} ##");
                AppLog.info("StreamId for ${i} is: ${info.streamId}");
                AppLog.info("Stream ${i} is: ${info.rendererInfo.usage}");
                AppLog.info("Flag ${i} is: ${info.rendererInfo.rendererFlags}");
                let deviceDescriptors = info.deviceDescriptors
                for (j in 0..deviceDescriptors.size) {
                    AppLog.info("Id: ${i} : ${deviceDescriptors[j].id}");
                    AppLog.info("Type: ${i} : ${deviceDescriptors[j].deviceType}");
                    AppLog.info("Role: ${i} : ${deviceDescriptors[j].deviceRole}");
                    AppLog.info("Name: ${i} : ${deviceDescriptors[j].name}");
                    AppLog.info("Address: ${i} : ${deviceDescriptors[j].address}");
                    AppLog.info("SampleRates: ${i} : ${deviceDescriptors[j].sampleRates[0]}");
                    AppLog.info("ChannelCount ${i} : ${deviceDescriptors[j].channelCounts[0]}");
                    AppLog.info("ChannelMask: ${i} : ${deviceDescriptors[j].channelMasks}");
                }
            }
        }
        audioStreamManager.on(AudioStreamManagerCallbackType.RENDERER_CHANGE, AudioRendererChangeCallback(callback))
    }
    ```

3. （可选）使用off(AudioStreamManagerCallbackType)取消监听音频播放流变化。

    ```cangjie
    audioStreamManager.off(AudioStreamManagerCallbackType.RENDERER_CHANGE);
    AppLog.info("RendererChange Off is called.");
    ```

4. （可选）使用getCurrentAudioRendererInfoArray()获取所有音频播放流的信息。

    该接口可获取音频播放流唯一ID、音频播放客户端的UID、音频状态以及音频播放器的其他信息。

    > **说明：**
    >
    > 对所有音频流状态进行监听的应用需要[声明权限](../../security/AccessToken/cj-declare-permissions.md)ohos.permission.USE_BLUETOOTH，否则无法获得实际的设备名称和设备地址信息，查询到的设备名称和设备地址（蓝牙设备的相关属性）将为空字符串。
