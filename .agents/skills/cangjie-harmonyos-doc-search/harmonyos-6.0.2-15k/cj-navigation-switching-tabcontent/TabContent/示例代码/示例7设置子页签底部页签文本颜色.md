### 示例7（设置子页签/底部页签文本颜色）

本示例通过labelStyle中的unselectedColor和selectedColor改变底部页签以及子页签的文本颜色。

通过iconStyle中的unselectedColor和selectedColor改变底部页签的图标颜色。

<!-- run -->

```cangjie
package ohos_app_cangjie_entry

import kit.UIKit.*
import std.collection.*
import kit.LocalizationKit.*
import ohos.state_macro_manage.*

@Entry
@Component
class EntryView {
    func build() {
        Column(5.vp) {
            Text("子页签样式")
            Column() {
                Tabs(BarPosition.Start) {
                    TabContent() {
                        Column().width(100.percent).height(100.percent).backgroundColor(0xE67C92)
                    }.tabBar(
                        SubTabBarStyle("Pink").labelStyle(LabelStyle(unselectedColor: 0xff0000, selectedColor: 0x00CB87)
                    ))

                    TabContent() {
                        Column().width(100.percent).height(100.percent).backgroundColor(0xFFBF00)
                    }.tabBar(
                        SubTabBarStyle("Yellow").labelStyle(
                        LabelStyle(unselectedColor: 0xff0000, selectedColor: 0x00CB87)))

                    TabContent() {
                        Column().width(100.percent).height(100.percent).backgroundColor(0x007DFF)
                    }.tabBar(
                        SubTabBarStyle("Blue").labelStyle(LabelStyle(unselectedColor: 0xff0000, selectedColor: 0x00CB87)
                    ))

                    TabContent() {
                        Column().width(100.percent).height(100.percent).backgroundColor(0x00CB87)
                    }.tabBar(
                        SubTabBarStyle("Green").labelStyle(
                        LabelStyle(unselectedColor: 0xff0000, selectedColor: 0x00CB87)))
                }.vertical(false).scrollable(true).barMode(BarMode.Fixed).onChange(
                    {
                    index: Int32 => AppLog.info("${index}")
                }).width(100.percent).backgroundColor(0xF1F3F5)
            }.width(100.percent).height(200)
            Text("底部页签样式")
            Column() {
                Tabs(BarPosition.End) {
                    TabContent() {
                        Column().width(100.percent).height(100.percent).backgroundColor(0xE67C92)
                    }.tabBar(
                        BottomTabBarStyle(@r(app.media.star_off), "Pink").labelStyle(
                        LabelStyle(unselectedColor: 0xff0000, selectedColor: 0x00CB87)).iconStyle(
                        TabBarIconStyle(unselectedColor: 0xff0000, selectedColor: 0x00CB87)))

                    TabContent() {
                        Column().width(100.percent).height(100.percent).backgroundColor(0xFFBF00)
                    }.tabBar(
                        BottomTabBarStyle(@r(app.media.star_off), "Yellow").labelStyle(
                        LabelStyle(unselectedColor: 0xff0000, selectedColor: 0x00CB87)).iconStyle(
                        TabBarIconStyle(unselectedColor: 0xff0000, selectedColor: 0x00CB87)))

                    TabContent() {
                        Column().width(100.percent).height(100.percent).backgroundColor(0x007DFF)
                    }.tabBar(
                        BottomTabBarStyle(@r(app.media.star_off), "Blue").labelStyle(
                        LabelStyle(unselectedColor: 0xff0000, selectedColor: 0x00CB87)).iconStyle(
                        TabBarIconStyle(unselectedColor: 0xff0000, selectedColor: 0x00CB87)))

                    TabContent() {
                        Column().width(100.percent).height(100.percent).backgroundColor(0x00CB87)
                    }.tabBar(
                        BottomTabBarStyle(@r(app.media.star_off), "Green").labelStyle(
                        LabelStyle(unselectedColor: 0xff0000, selectedColor: 0x00CB87)).iconStyle(
                        TabBarIconStyle(unselectedColor: 0xff0000, selectedColor: 0x00CB87)))
                }.vertical(false).scrollable(true).barMode(BarMode.Fixed).onChange(
                    {
                    index: Int32 => AppLog.info("${index}")
                }).width(100.percent).backgroundColor(0xF1F3F5)
            }.width(100.percent).height(200)
        }.width(100.percent)
    }
}
```
