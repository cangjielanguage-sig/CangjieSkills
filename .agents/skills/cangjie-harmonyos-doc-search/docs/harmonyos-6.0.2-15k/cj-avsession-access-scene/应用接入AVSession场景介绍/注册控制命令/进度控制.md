### 进度控制

应用如果支持进度显示，进一步也可以支持进度控制。应用需要响应seek的控制命令，那么当用户在播控中心的界面上进行拖动操作时，应用就会收到对应的回调。参考实现：

```cangjie
import ohos.base.*
import kit.AVSessionKit.*
import kit.BasicServicesKit.*

class SeekCallBack <: Callback1Argument<Int64> {
    SeekCallBack(let f: (Int64) -> Unit) {}
    public func invoke(arg1: Int64): Unit {
        f(arg1)
    }
}

let context = Global.getStageContext() // 获取Context应用上下文请参见 [API 参考] -> [API参考概述] -> [开发说明] -> [仓颉示例代码说明]
// 假设已经创建了一个session，如何创建session可以参考之前的案例。
let `type`: AVSessionType = SESSION_TYPE_AUDIO
let session = createAVSession(context, 'SESSION_NAME', `type`)

session.on(Seek, SeekCallBack({position => AppLog.info('on seek , the time is ${position}')}))
// 由于应用内seek可能会触发较长的缓冲等待，可以先把状态设置为 Buffering。
let playbackState = AVPlaybackState(PlaybackState.PLAYBACK_STATE_BUFFERING, // 缓冲状态。
    None, None, None,
    None, None, None, None, None, None, None, None, None, None)
// 应用响应seek命令，使用应用内播放器完成seek实现。
try {
    session.setAVPlaybackState(playbackState)
    AppLog.info('SetAVPlaybackState successfully')
} catch (err: BusinessException) {
    AppLog.error('Failed to set AVPlaybackState. Code: ${err.code}, message: ${err.message}')
}

try {
    // 应用内更新新的位置后，也需要同步更新状态给系统。
    playbackState.state = PlaybackState.PLAYBACK_STATE_PLAY // 播放状态。
    playbackState.position = PlaybackPosition(playbackState
        .position
        .getOrThrow()
        .elapsedTime, // 已经播放的位置，以ms为单位。
        //此处缺一个能获取时间的函数                     // 应用更新当前位置的时间戳，以ms为单位。
        )
    session.setAVPlaybackState(playbackState)

    AppLog.info('SetAVPlaybackState successfully')
} catch (err: BusinessException) {
    AppLog.error('Failed to set AVPlaybackState. Code: ${err.code}, message: ${err.message}')
}
```
