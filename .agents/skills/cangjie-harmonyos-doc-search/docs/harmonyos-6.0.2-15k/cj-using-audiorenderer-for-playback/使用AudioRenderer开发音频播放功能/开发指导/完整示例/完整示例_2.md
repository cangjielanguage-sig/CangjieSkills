// 停止渲染。
func stop() {
    if (let Some(renderer) <- audioRenderer) {
        match (renderer.state) {
            // 只有渲染器状态为running或paused的时候才可以停止。
            case AudioState.STATE_PAUSED | AudioState.STATE_RUNNING =>
                try {
                    // 停止渲染。
                    renderer.stop()
                    AppLog.info("Renderer stop success.")
                } catch (e: BusinessException) {
                    AppLog.error("Renderer stop failed")
                }
            case _ => AppLog.error("Renderer is not running or paused.")
        }
    }
}

// 销毁实例，释放资源。
func release() {
    if (let Some(renderer) <- audioRenderer) {
        match (renderer.state) {
            // 渲染器状态不是released状态，才能release。
            case AudioState.STATE_RELEASED =>
                AppLog.error("Renderer already released")
            case _ =>
                try {
                    // 释放资源。
                    renderer.release()
                    AppLog.info("Renderer release success.")
                } catch (e: BusinessException) {
                    AppLog.error("Renderer release failed")
                }
        }
    }
}
