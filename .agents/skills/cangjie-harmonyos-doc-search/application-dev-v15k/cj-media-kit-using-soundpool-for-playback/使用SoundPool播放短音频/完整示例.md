## 完整示例

下面展示了使用SoundPool进行低时延播放的完整示例代码。

```cangjie
import kit.MediaKit.*
import kit.CoreFileKit.*
import kit.AudioKit.*
import ohos.base.*

var soundPool: ?SoundPool = None
var streamId: Int32 = 0
var soundId: Int32 = 0
// audioRenderInfo中的参数usage取值为STREAM_USAGE_UNKNOWN，STREAM_USAGE_MUSIC，STREAM_USAGE_MOVIE，
// STREAM_USAGE_AUDIOBOOK时，SoundPool播放短音时为混音模式，不会打断其他音频播放。
var audioRendererInfo: AudioRendererInfo = AudioRendererInfo(
    StreamUsage.STREAM_USAGE_MUSIC,
    1
)
var playParameters: PlayParameters = PlayParameters(
    loop: 3, // 循环3次
    rate: AudioRendererRate.RENDER_RATE_NORMAL, // 正常倍速
    leftVolume: 0.5f32, // range = 0.0-1.0
    rightVolume: 0.5f32, // range = 0.0-1.0
    priority: 0, // 最低优先级
)
var uri: String = ""

func create() {
    //创建soundPool实例
    soundPool = createSoundPool(5, audioRendererInfo)
    //注册监听
    loadCallback()
    finishPlayCallback()
    setErrorCallback()
    // 加载音频资源
    // "/test_01.mp3" 作为样例，使用时需要传入文件对应路径。
    var file = FileFs.open("/test_01.mp3", mode: READ_ONLY.mode)
    uri = "fd://" + (file.fd).toString()
    soundId = soundPool.getOrThrow().load(uri)
}

func loadCallback() {
    // 加载完成回调
    match (soundPool) {
        case Some(v) => v.on(SoundPoolEvent.LoadCompleted, SoundPoolEventCallback())
        case _ => ()
    }
}
//设置播放完成监听
func finishPlayCallback() {
    // 播放完成回调'playFinished'，当音频播放完毕时，会触发注册的回调。
    match (soundPool) {
        case Some(v) =>
            // 可进行下次播放
            v.on(SoundPoolEvent.PlayFinished, FinishPlayCallback())
        case _ => ()
    }
}

//设置错误类型监听
func setErrorCallback() {
    match (soundPool) {
        case Some(v) => v.on(SoundPoolEvent.EventError, ErrCallBack())
        case _ => ()
    }
}

func PlaySoundPool() {
    if (soundPool.isNone()) {
        return
    }
    var realSoundPool = soundPool.getOrThrow()
    // 开始播放，这边play也可带播放的参数PlayParameters，请在音频资源加载完毕，即收到loadComplete回调之后再执行play操作
    streamId = realSoundPool.play(soundId, params: playParameters)
    // 设置循环播放次数
    realSoundPool.setLoop(streamId, 2) // 播放3次

    // 设置对应流的优先级
    realSoundPool.setPriority(streamId, 1)
    // 设置音量
    realSoundPool.setVolume(streamId, 0.5f32, 0.5f32)
}

func release() {
    if (soundPool.isNone()) {
        return
    }
    var realSoundPool = soundPool.getOrThrow()
    // 终止指定流的播放
    realSoundPool.stop(streamId)
    // 卸载音频资源
    realSoundPool.unload(soundId)
    //关闭监听
    setOffCallback()
    // 释放SoundPool
    realSoundPool.release()
}
//关闭监听
func setOffCallback() {
    if (soundPool.isNone()) {
        return
    }
    var realSoundPool = soundPool.getOrThrow()
    realSoundPool.off(SoundPoolEvent.LoadCompleted)
    realSoundPool.off(SoundPoolEvent.PlayFinished)
    realSoundPool.off(SoundPoolEvent.EventError)
}

//on 回调类型
class SoundPoolEventCallback <: Callback1Argument<Int32> {
    public init() {}
    public open func invoke(soundID: Int32): Unit {
        AppLog.info("loadComplete, soundId: ${soundId}")
    }
}

// finishPlayCallback() 使用的回调类型
class FinishPlayCallback <: Callback0Argument {
    public init() {}
    public open func invoke() {
        AppLog.info("receive play finished message")
    }
}

class ErrCallBack <: Callback1Argument<BusinessException> {
    public init() {}
    // 根据实际业务实现错误上报回调逻辑
    public open func invoke(err: BusinessException): Unit {
        AppLog.info("处理异常情况")
    }
}
```
