// 编辑图片功能实现
                            this.originalImage?.rotate(90.0)
                            let packOpts = PackingOption("image/jpeg", 98)
                            try {
                                // 调用saveEditedContentWithImage保存图片
                                let context = getContext()
                                let data = context.saveEditedContentWithImage(
                                    originalImage.getOrThrow({=> BusinessException(401, "None image")}), packOpts)

                                if (data.resultCode == 0) {
                                    AppLog.info("Save succeed.")
                                }
                                this.newWant = data.want
                                // data.want.uri存有编辑过图片的uri
                                let imagePixMap = this.readImageByUri(this.newWant?.uri ?? "")
                                editedImage = imagePixMap
                            } catch (e: BusinessException) {
                                AppLog.error("saveContentEditingWithImage failed: ${e.message}")
                            }
                        }
                    ).margin(10)

                    Button("terminateSelfWithResult").onClick(
                        {
                            evt =>
                            AppLog.info("Finish the current editing.")

                            let session = localStorage.get<UIExtensionContentSession>("session")
                            // 关闭并回传修改结果给调用方
                            session?.terminateSelfWithResult(AbilityResult(0, this.newWant ?? Want()))
                        }
                    ).margin(10)

                    Image(
                        this.originalImage.getOrDefault(
                        {=> createPixelMap(InitializationOptions(size: Size(height: 1, width: 1)))})).width(100.percent).
                        height(200).margin(10).objectFit(ImageFit.Contain)

                    Image(
                        this.editedImage.getOrDefault(
                        {=> createPixelMap(InitializationOptions(size: Size(height: 1, width: 1)))})).width(100.percent).
                        height(200).margin(10).objectFit(ImageFit.Contain)
                }.width(100.percent)
            }.height(100.percent)
        }
    }
