## 开发步骤

1. 导入接口。

    ```cangjie
    import kit.CameraKit.*
    import kit.ArkGraphics2D.*
    import kit.BasicServicesKit.*
    import kit.MediaKit.*
    import kit.AbilityKit.*
    import kit.MediaLibraryKit.*
    import kit.CoreFileKit.*
    import ohos.base.*
    import std.time.*
    ```

2. 获取预览、录像的配置项。

    HDR录像的输出格式需要设置成10bit的CAMERA_FORMAT_YCRCB_P010。具体参考[setColorSpace](../../../API_Reference/source_zh_cn/apis/CameraKit/cj-apis-multimedia-camera.md#func-setcolorspacecolorspace)。

    > **说明：**
    >
    > 预览流与录像输出流的分辨率的宽高比要保持一致，如果设置XComponent组件中的Surface显示区域宽高比为1920:1080 = 16:9，则需要预览流中的分辨率的宽高比也为16:9，如分辨率选择640:360，或960:540，或1920:1080，以此类推。

    ```cangjie
    func getPreviewProfile(previewProfiles: Array<Profile>, width: UInt32, height: UInt32): Option<Profile> {
        for (profile in previewProfiles) {
            if (profile.format == CameraFormat.CAMERA_FORMAT_YCRCB_P010 && profile.size.width == width &&
                profile.size.height == height) {
                return profile
            }
        }
        return None
    }

    func getVideoProfile(videoProfiles: Array<VideoProfile>, width: UInt32, height: UInt32): Option<VideoProfile> {
        for (profile in videoProfiles) {
            if (profile.format == CameraFormat.CAMERA_FORMAT_YCRCB_P010 && profile.size.width == width &&
                profile.size.height == height) {
                return profile
            }
        }
        return None
    }
    ```

3. 查询是否支持视频防抖。

    HDR录像需要支持视频防抖。

    ```cangjie
    func isVideoStabilizationModeSupported(session: VideoSession, mode: VideoStabilizationMode): Bool {
        var isSupported: Bool = false
        try {
            isSupported = session.isVideoStabilizationModeSupported(mode)
        } catch (error: BusinessException) {
            // 失败返回错误码error.code并处理
            AppLog.error("The isVideoStabilizationModeSupported call failed. error code: ${error.code}")
        }
        return isSupported
    }
    ```

4. 设置视频防抖。

    ```cangjie
    func setVideoStabilizationMode(session: VideoSession): Bool {
        let mode: VideoStabilizationMode = VideoStabilizationMode.AUTO
        // 查询是否支持视频防抖
        var isSupported: Bool = isVideoStabilizationModeSupported(session, mode)
        if (isSupported) {
            AppLog.info("setVideoStabilizationMode: ${mode}")
            // 设置视频防抖
            session.setVideoStabilizationMode(mode)
            let activeVideoStabilizationMode = session.getActiveVideoStabilizationMode()
            AppLog.info("activeVideoStabilizationMode: ${activeVideoStabilizationMode}")
        } else {
            AppLog.info("videoStabilizationMode: ${mode} is not support")
        }
        return isSupported
    }
    ```

5. 查询支持的色彩空间。

    ```cangjie
    func getSupportedColorSpaces(session: VideoSession): Array<ColorSpace> {
        var colorSpaces: Array<ColorSpace> = []
        try {
            colorSpaces = session.getSupportedColorSpaces()
        } catch (error: BusinessException) {
            AppLog.error("The getSupportedColorSpaces call failed. error code: ${error.code}")
        }
        return colorSpaces
    }
    ```

6. 设置色彩空间。

    如果是SDR录像色彩空间需要设置为BT709_LIMIT，如果是HDR录像色彩空间需要设置为BT2020_HLG_LIMIT。具体参考[setColorSpace](../../../API_Reference/source_zh_cn/apis/CameraKit/cj-apis-multimedia-camera.md#func-setcolorspacecolorspace)。
